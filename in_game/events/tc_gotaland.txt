# Content for Småland, Västergötland, Gotland and Bohuslän. Sweden owns all of this land from 1337 except Bohuslän, so
# these should fire in nearly every game unless it has been lost. .8 waits for Bohuslän to be conquered.

# Bondefred: about ten border peaces between Småland and Blekinge, 1505-25, because war disrupted their trade. The

	type = country_event

# The ban on exporting oxen, which helped set off the Dacke rising of 1542-43. Vanilla has no Dacke content. No rebels,
# so the rising is a large unrest modifier in Tioharad, the size of vanilla's cruel_overlord (0.5). The ban was meant
# to feed the miners and smiths of Bergslagen, so enforcing it also helps the mines there (swe_oxen_for_the_mines).
swe_gotaland.3 = {
	type = country_event
	title = swe_gotaland.3.title
	desc = swe_gotaland.3.desc
	outcome = neutral
	historical_info = swe_gotaland.3.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1538.1.1
		to = 1560.1.1
		monthly_chance = 3 # Median late 1539
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:vaxjo
	}

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
	}

	option = {
		name = swe_gotaland.3.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		custom_tooltip = {
			text = swe_gotaland.3.a_tt
			every_owned_location = {
				limit = { province_definition = province_definition:tioharad_province }
				add_location_modifier = { modifier = swe_oxen_ban years = 10 }
				# The anger stays in this region, in place of a country-wide estate penalty
				every_pop = {
					limit = { pop_type = pop_type:peasants }
					add_pop_satisfaction = pop_satisfaction_severe_penalty
				}
			}
		}
		custom_tooltip = {
			text = swe_gotaland.3.a_mines_tt
			every_owned_location = {
				limit = {
					OR = {
						province_definition = province_definition:dalarna_province
						province_definition = province_definition:vastmanland_province
					}
					OR = {
						raw_material = goods:iron
						raw_material = goods:copper
					}
				}
				add_location_modifier = { modifier = swe_oxen_for_the_mines years = 10 }
			}
		}
	}

	option = {
		name = swe_gotaland.3.b
		ai_chance = {
			base = 25
		}

		custom_tooltip = {
			text = swe_gotaland.3.b_tt
			every_owned_location = {
				limit = { province_definition = province_definition:tioharad_province }
				add_location_modifier = { modifier = swe_oxen_trade years = 25 }
			}
		}
	}
}

# Visby against the rural harbours (Lerbom, Mellan två riken). The Swedish crown banned trade at the rural harbours in
# 1646 to bring it all to Visby, and lifted the ban three years later after widespread complaints. Danish kings had
# tried the same, as the historical info says, but this event is the Swedish ban. The modifiers last 10 years, vanilla's
# median for a temporary situation. Anger stays on the island's peasants, in place of a country-wide estate penalty.
swe_gotaland.6 = {
	type = country_event
	title = swe_gotaland.6.title
	desc = swe_gotaland.6.desc
	outcome = neutral
	historical_info = swe_gotaland.6.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1644.1.1
		to = 1680.1.1
		monthly_chance = 3 # Median late 1645
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:visby
		OR = {
			owns = location:slite
			owns = location:hoborg
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:peasants_estate }
	}

	option = {
		name = swe_gotaland.6.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		location:visby = {
			add_location_modifier = { modifier = swe_visby_staple years = 10 }

			base = 25
		}

		custom_tooltip = {

# Jonas Alströmer's manufactories at Alingsås, privilege of 22 June 1724 (SBL, Jonas Alströmer), refused renewal in
# 1765-66, so about 40 years. Needs a town: the real Alingsås had town rights from 1619, the cloth bonus needs a building
# that needs town rank, and rural locations hold almost no burghers. No rank is forced, so this fires only where Alingsås
# has been raised, which is why the window stays open until the privileges ran out. Vanilla's flavor_swe.34 is the
# national version under the Hats and calls him Ahlströmer; this stays local. His workers came from abroad, London
# among other places (vastsverige.com), hence the migration.
swe_gotaland.7 = {
	type = country_event
	title = swe_gotaland.7.title
	desc = swe_gotaland.7.desc
	outcome = neutral
	historical_info = swe_gotaland.7.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1722.1.1
		to = 1765.1.1
		monthly_chance = 3 # Median late 1723 if it is already a town
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:allingsas
		location:allingsas = {
			OR = {
				location_rank = location_rank:town
				location_rank = location_rank:city
				location_rank = location_rank:megalopolis
			}
		}
	}

	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_cloth.dds"

	option = {
		name = swe_gotaland.7.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -4 }
		location:allingsas = {
			add_location_modifier = { modifier = swe_alingsas_manufaktur years = 40 }
		}
		if = {
			limit = {
				location:london = {
					has_owner = yes
					dominant_culture ?= culture:english
				}
			}
			add_migration = {
				# The source's owner, as in flavor_ven.2: owner = this creates nothing from a location we don't own
				owner = location:london.owner
				from = location:london.province_definition
				to_location = location:allingsas
				culture = culture:english
				type = pop_type:burghers
				amount = 0.02 # Per month, so 0.48 in all, against 875 workers in 1746 with the Swedes they trained
				months = 24
			}
		}
	}

	option = {
		name = swe_gotaland.7.b
		ai_chance = {
			base = 25
		}

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
	}
}

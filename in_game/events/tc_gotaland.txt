namespace = swe_gotaland

# Content for Småland and Västergötland. Sweden owns all of this land from 1337, so these should fire in
# nearly every game unless it has been lost.

# Bondefred: about ten border peaces between Småland and Blekinge, 1505-25, because war disrupted their trade. The
# treaty of August 1505 was confirmed in 1510, and Svante Nilsson spared Ronneby in 1506 (Cederholm). Manpower as in
# swe_jamtland_tings, whose peasants also kept out of the kings' wars.
swe_gotaland.1 = {
	type = country_event
	title = swe_gotaland.1.title
	desc = swe_gotaland.1.desc
	outcome = neutral
	historical_info = swe_gotaland.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1500.1.1
		to = 1530.1.1
		monthly_chance = 3 # Median late 1501
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:vaxjo
	}

	illustration_tags = {
		10 = happy
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:burghers_estate }
	}

	option = {
		name = swe_gotaland.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		custom_tooltip = {
			text = swe_gotaland.1.a_tt
			every_owned_location = {
				limit = { province_definition = province_definition:tioharad_province }
				add_location_modifier = { modifier = swe_varend_border_peace years = 20 }
			}
		}
	}

	option = {
		name = swe_gotaland.1.b
		ai_chance = {
			base = 25
		}

		custom_tooltip = {
			text = swe_gotaland.1.b_tt
			every_owned_location = {
				limit = { province_definition = province_definition:tioharad_province }
				add_location_modifier = { modifier = swe_border_peace_forbidden years = 10 }
			}
		}
	}
}

# The ban on exporting oxen, which helped set off the Dacke rising of 1542-43. Vanilla has no Dacke content. No rebels,
# so the rising is a large unrest modifier in Tioharad, the size of vanilla's cruel_overlord (0.5). The ban was meant
# to feed the miners and smiths of Bergslagen, so enforcing it also helps the mines there (swe_oxen_for_the_mines).
swe_gotaland.3 = {
	type = country_event
	title = swe_gotaland.3.title
	desc = swe_gotaland.3.desc
	outcome = neutral
	historical_info = swe_gotaland.3.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1538.1.1
		to = 1560.1.1
		monthly_chance = 3 # Median late 1539
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:vaxjo
	}

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
	}

	option = {
		name = swe_gotaland.3.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		custom_tooltip = {
			text = swe_gotaland.3.a_tt
			every_owned_location = {
				limit = { province_definition = province_definition:tioharad_province }
				add_location_modifier = { modifier = swe_oxen_ban years = 10 }
				# The anger stays in this region, in place of a country-wide estate penalty
				every_pop = {
					limit = { pop_type = pop_type:peasants }
					add_pop_satisfaction = pop_satisfaction_severe_penalty
				}
			}
		}
		custom_tooltip = {
			text = swe_gotaland.3.a_mines_tt
			every_owned_location = {
				limit = {
					OR = {
						province_definition = province_definition:dalarna_province
						province_definition = province_definition:vastmanland_province
					}
					OR = {
						raw_material = goods:iron
						raw_material = goods:copper
					}
				}
				add_location_modifier = { modifier = swe_oxen_for_the_mines years = 10 }
			}
		}
	}

	option = {
		name = swe_gotaland.3.b
		ai_chance = {
			base = 25
		}

		custom_tooltip = {
			text = swe_gotaland.3.b_tt
			every_owned_location = {
				limit = { province_definition = province_definition:tioharad_province }
				add_location_modifier = { modifier = swe_oxen_trade years = 25 }
			}
		}
	}
}

# The knallar of Sjuhäradsbygden, peddling from the mid-1500s against the town monopoly, with special laws from the
# 1680s. Skene and Bogesund stand in for the district.
swe_gotaland.4 = {
	type = country_event
	title = swe_gotaland.4.title
	desc = swe_gotaland.4.desc
	outcome = neutral
	historical_info = swe_gotaland.4.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1560.1.1
		to = 1700.1.1
		monthly_chance = 3 # Median late 1561
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:skene
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:burghers_estate }
	}

	option = {
		name = swe_gotaland.4.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		location:skene = {
			add_location_modifier = { modifier = swe_knallar years = 100 }
		}
		location:bogesund = {
			if = {
				limit = { owner ?= root }
				add_location_modifier = { modifier = swe_knallar years = 100 }
			}
		}
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_bonus }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_penalty }
		change_societal_value = { type = mercantilism_vs_free_trade value = societal_value_minor_move_to_right }
	}

	option = {
		name = swe_gotaland.4.b
		ai_chance = {
			base = 25
		}

		location:skene = {
			add_location_modifier = { modifier = swe_town_monopoly_enforced years = 10 }
		}
		location:bogesund = {
			if = {
				limit = { owner ?= root }
				add_location_modifier = { modifier = swe_town_monopoly_enforced years = 10 }
			}
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
		change_societal_value = { type = mercantilism_vs_free_trade value = societal_value_minor_move_to_left }
	}
}

# Jonas Alströmer's manufactories at Alingsås, privileged in 1724 and refused renewal in 1765-66 (SBL). Needs a town,
# since the cloth bonus needs a town building and rural locations hold almost no burghers. Vanilla's flavor_swe.34 is
# the national version. His workers came from abroad, London among other places (vastsverige.com), hence the migration.
swe_gotaland.7 = {
	type = country_event
	title = swe_gotaland.7.title
	desc = swe_gotaland.7.desc
	outcome = neutral
	historical_info = swe_gotaland.7.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1722.1.1
		to = 1765.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_gotaland_7
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:allingsas
		location:allingsas = {
			OR = {
				location_rank = location_rank:town
				location_rank = location_rank:city
				location_rank = location_rank:megalopolis
			}
		}
	}

	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_cloth.dds"

	option = {
		name = swe_gotaland.7.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -4 }
		location:allingsas = {
			add_location_modifier = { modifier = swe_alingsas_manufaktur years = 40 }
		}
		if = {
			limit = {
				location:london = {
					has_owner = yes
					dominant_culture ?= culture:english
				}
			}
			add_migration = {
				# The source's owner, since owner = this creates nothing from a location we don't own
				owner = location:london.owner
				from = location:london.province_definition
				to_location = location:allingsas
				culture = culture:english
				type = pop_type:burghers
				amount = 0.02 # Per month
				months = 24
			}
		}
	}

	option = {
		name = swe_gotaland.7.b
		ai_chance = {
			base = 25
		}

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
	}
}

# Nya Lödöse, 1473. From the 1450s the Norwegians and Danes took a toll at Bohus, and Sten Sture the Elder moved
# Lödöse, the realm's only port to the west, to the mouth of the Säveån (Lödöse museum; sv.wiki Nya Lödöse). Needs
# Bohus (konghalla) held by neither us nor a subject. swe_western_gate sits on lodose from game start and leaves with
# the town. The new town is a modifier, not a rename, since vanilla's flavor_swe.41 renames utlandet to Göteborg.
swe_gotaland.11 = {
	type = country_event
	title = swe_gotaland.11.title
	desc = swe_gotaland.11.desc
	outcome = neutral
	historical_info = swe_gotaland.11.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1471.9.1
		to = 1600.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_gotaland_11
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:lodose
		owns = location:utlandet
		NOT = { owns = location:konghalla }
		location:konghalla = {
			NOT = { owner ?= { is_subject_of = root } }
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
	}

	option = {
		name = swe_gotaland.11.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -4 }
		location:lodose = {
			if = {
				limit = { has_location_modifier = swe_western_gate }
				remove_location_modifier = swe_western_gate
			}
			change_prosperity = prosperity_mild_penalty
		}
		location:utlandet = {
			add_location_modifier = { modifier = swe_nya_lodose months = -1 }
			change_development = development_mild_bonus
			change_prosperity = prosperity_mild_bonus
		}
		# The burghers of old Lödöse were ordered to move. Laborers, since rural Lödöse holds almost no burghers
		add_migration = {
			owner = this
			from_location = location:lodose
			to_location = location:utlandet
			type = pop_type:laborers
			amount = 0.02
			months = 24
		}
	}

	option = {
		name = swe_gotaland.11.b
		ai_chance = {
			base = 25
		}

		location:lodose = {
			add_location_modifier = { modifier = swe_bohus_toll years = 20 }
		}
	}
}

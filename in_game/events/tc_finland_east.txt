# Content for inland and eastern Finland. Sweden starts with Viborg and southern Tavastland, but Savolax and
# Kajanaland have no landed owner in 1337. So .2, .3, .4 and .8 should fire in nearly every game, .7 in any game
# where Sweden turns Protestant, while .1, .5 and .6 wait for Sweden to own the location and fire at whatever date the
# land is settled.

# Olofsborg stands on the Russian side of the Nöteborg line (1475), and about 1,000 Russians tried to move the
# border posts in 1499. It doesn't build the castle, since the player can already do that by hand.
swe_finland_east.1 = {
	type = country_event
	title = swe_finland_east.1.title
	desc = swe_finland_east.1.desc
	outcome = neutral
	historical_info = swe_finland_east.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1475.1.1
		to = 1600.1.1
		monthly_chance = 3 # Median about two years once Sweden owns Olofsborg
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:olofsborg
	}

	illustration_tags = {
		10 = armed
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:peasants_estate }
	}

	option = {
		name = swe_finland_east.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -2 }
		location:olofsborg = {
			add_location_modifier = { modifier = swe_olofsborg_line years = 20 }
		}
		add_prestige = prestige_mild_bonus
	}

	option = {
		name = swe_finland_east.1.b
		ai_chance = {
			base = 25
		}

		location:olofsborg = {
			add_location_modifier = { modifier = swe_olofsborg_quiet_border years = 10 }
		}
	}
}

# Gustav Vasa's 1542 claim to uninhabited wilderness, with the Häme-Savo boundary judgments of 1415, 1446 and 1452
# (Matikainen 2022) as background. A migration modifier rather than add_migration, since Sweden may not own this
# land yet. Skips lakes and foreign-held locations. Same is_ownable and has_owner checks as vanilla. The tax-free years
# cost gold, and the hunting families' anger stays with the peasants of the claimed land, not the whole estate.
swe_finland_east.4 = {
	type = country_event
	title = swe_finland_east.4.title
	desc = swe_finland_east.4.desc
	outcome = neutral
	historical_info = swe_finland_east.4.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1540.1.1
		to = 1600.1.1
		monthly_chance = 3 # Median late 1541
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:tavastehus
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
	}

	option = {
		name = swe_finland_east.4.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		# Some years of tax freedom for every settler who clears a farm
		change_gold_effect = { scale = -4 }
		custom_tooltip = {
			text = swe_finland_east.4.a_tt
			province_definition:savolax_province = {
				every_location_in_province_definition = {
					limit = {
						is_ownable = yes
						OR = {
							has_owner = no
							owner ?= root
						}
					}
					add_location_modifier = { modifier = swe_crown_wilderness years = 25 }
				}
			}
			province_definition:kajanaland_province = {
				every_location_in_province_definition = {
					limit = {
						is_ownable = yes
						OR = {
							has_owner = no
							owner ?= root
						}
					}
					add_location_modifier = { modifier = swe_crown_wilderness years = 25 }
				}
			}
			# Only the north of Tavastland, settled from Savo: Laukaa (with Jyväskylä) split from Rautalampi in 1628,
			# Viitasaari followed in 1635 and Saarijärvi in 1639 (en.wiki). The south is old farmland around Tavastehus.
			location:jyvaskyla = {
				if = {
					limit = {
						OR = {
							has_owner = no
							owner ?= root
						}
					}
					add_location_modifier = { modifier = swe_crown_wilderness years = 25 }
				}
			}
			location:saarijarvi = {
				if = {
					limit = {
						OR = {
							has_owner = no
							owner ?= root
						}
					}
					add_location_modifier = { modifier = swe_crown_wilderness years = 25 }
				}
			}
			location:viitasaari = {
				if = {
					limit = {
						OR = {
							has_owner = no
							owner ?= root
						}
					}
					add_location_modifier = { modifier = swe_crown_wilderness years = 25 }
				}
			}
			location:laukas = {
				if = {
					limit = {
						OR = {
							has_owner = no
							owner ?= root
						}
					}
					add_location_modifier = { modifier = swe_crown_wilderness years = 25 }
				}
			}
		}
		# The hunting families' anger stays in the land claimed, in place of a country-wide estate penalty
		custom_tooltip = {
			text = swe_finland_east.4.a_peasants_tt
			every_owned_location = {
				limit = { swe_crown_wilderness_lands_trigger = yes }
				every_pop = {
					limit = { pop_type = pop_type:peasants }
					add_pop_satisfaction = pop_satisfaction_mild_penalty
				}
			}
		}
	}

	option = {
		name = swe_finland_east.4.b
		ai_chance = {
			base = 25
		}

		custom_tooltip = {
			text = swe_finland_east.4.b_tt
			every_owned_location = {
				limit = { swe_crown_wilderness_lands_trigger = yes }
				every_pop = {
					limit = { pop_type = pop_type:peasants }
					add_pop_satisfaction = pop_satisfaction_weak_bonus
				}
			}
		}
	}
}

# Johannes Messenius: sentenced to death in July 1616 for conspiring with Sigismund and the Jesuits, then sent to
# unfinished Kajaneborg in October, where he wrote Scondia Illustrata. Waits for Sweden to own Kajana.
swe_finland_east.5 = {
	type = country_event
	title = swe_finland_east.5.title
	desc = swe_finland_east.5.desc
	outcome = neutral
	historical_info = swe_finland_east.5.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1614.1.1
		to = 1660.1.1
		monthly_chance = 3 # Median late 1615, or about two years after Sweden takes Kajana
	}

	trigger = {
		has_or_had_tag = SWE
		# The charge was plotting with the Jesuits, which only a Protestant realm would bring
		is_protestant = yes
		owns = location:kajana
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
		# The sentence is ours to commute
		ruler_or_regent ?= { save_scope_as = target_character }
	}

	option = {
		name = swe_finland_east.5.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		# He wrote Scondia Illustrata in prison. No artist, as in flavor_swe.40, so no new character.
		location:kajana = {
			create_art = {
				quality = 70
				type = work_of_art_type:chronicle
				key = swe_scondia_illustrata
			}
		}
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }
	}

	option = {
		# Not .b: that key's hash collides with Oikoumenē's xuzhou_province.silesian_dialect
		name = swe_finland_east.5.b_pardon
		ai_chance = {
			base = 25
		}

		add_prestige = prestige_mild_bonus
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_penalty }
	}
}

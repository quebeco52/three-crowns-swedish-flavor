namespace = swe_towns

# Town foundings that rename a location. Values from vanilla's flavor_dan.43 (Christianstad) and flavor_swe.41
# (Göteborg). None of them change the location rank.

# Gustav Vasa founds Helsingfors at the mouth of the Vanda (12 June 1550) to take the Gulf of Finland trade from Reval,
# and orders the burghers of Ekenäs, Borgå, Raumo and Ulvsby to move. If Sweden owns Reval the text changes; that
# variant, a German Reval trading on its own account, is an inference, since Reval became Swedish only in 1561.
swe_towns.1 = {
	type = country_event
	title = swe_towns.1.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { owns = location:tallinn }
				desc = swe_towns.1.desc_tallinn
			}
			triggered_desc = {
				desc = swe_towns.1.desc
			}
		}
	}
	outcome = neutral
	historical_info = swe_towns.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1548.1.1
		to = 1600.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_towns_1
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:helsinge
		OR = {
			owns = location:ekenas
			owns = location:porvoo
			owns = location:raumo
			owns = location:bjorneborg
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
	}

	option = {
		name = {
			text = swe_towns.1.a
			trigger = { NOT = { owns = location:tallinn } }
		}
		name = {
			text = swe_towns.1.a_tallinn
			trigger = { owns = location:tallinn }
		}
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -4 }
		location:helsinge = {
			rename_location = helsingfors
			change_development = development_mild_bonus
			change_prosperity = prosperity_mild_bonus
		}
		# Ekenäs and Borgå (Nyland), Raumo and Ulvsby (in bjorneborg, Satakunta). The trigger needs only one of the
		# four, so each move checks its own town. Only into a town or city, since burghers sent to a rural location
		# vanish within a month.
		if = {
			limit = {
				location:helsinge = { NOT = { location_rank = location_rank:rural_settlement } }
			}
			if = {
				limit = { owns = location:ekenas }
				add_migration = {
					owner = this
					from_location = location:ekenas
					to_location = location:helsinge
					type = pop_type:burghers
					amount = 0.02
					months = 24
				}
			}
			if = {
				limit = { owns = location:porvoo }
				add_migration = {
					owner = this
					from_location = location:porvoo
					to_location = location:helsinge
					type = pop_type:burghers
					amount = 0.02
					months = 24
				}
			}
			if = {
				limit = { owns = location:raumo }
				add_migration = {
					owner = this
					from_location = location:raumo
					to_location = location:helsinge
					type = pop_type:burghers
					amount = 0.02
					months = 24
				}
			}
			if = {
				limit = { owns = location:bjorneborg }
				add_migration = {
					owner = this
					from_location = location:bjorneborg
					to_location = location:helsinge
					type = pop_type:burghers
					amount = 0.02
					months = 24
				}
			}
		}
		# The anger stays with the burghers ordered to move
		custom_tooltip = {
			text = swe_towns.1.a_tt
			swe_pop_satisfaction_in = { location = location:ekenas type = burghers value = pop_satisfaction_severe_penalty }
			swe_pop_satisfaction_in = { location = location:porvoo type = burghers value = pop_satisfaction_severe_penalty }
			swe_pop_satisfaction_in = { location = location:raumo type = burghers value = pop_satisfaction_severe_penalty }
			swe_pop_satisfaction_in = { location = location:bjorneborg type = burghers value = pop_satisfaction_severe_penalty }
		}
	}

	option = {
		name = swe_towns.1.b
		ai_chance = {
			base = 25
		}

		custom_tooltip = {
			text = swe_towns.1.b_tt
			swe_pop_satisfaction_in = { location = location:ekenas type = burghers value = pop_satisfaction_weak_bonus }
			swe_pop_satisfaction_in = { location = location:porvoo type = burghers value = pop_satisfaction_weak_bonus }
			swe_pop_satisfaction_in = { location = location:raumo type = burghers value = pop_satisfaction_weak_bonus }
			swe_pop_satisfaction_in = { location = location:bjorneborg type = burghers value = pop_satisfaction_weak_bonus }
		}
	}
}

# Karl IX founds a town by Korsholm castle (2 October 1606). It was first called Mussor and
# renamed Vasa after the royal house a few years later, so the name follows the ruling house.
swe_towns.3 = {
	type = country_event
	title = swe_towns.3.title
	desc = swe_towns.3.desc
	outcome = neutral
	historical_info = swe_towns.3.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1604.1.1
		to = 1660.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_towns_3
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:korsholm
	}

	illustration_tags = {
		10 = happy
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:peasants_estate }
	}

	option = {
		name = swe_towns.3.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -2 }
		if = {
			limit = {
				exists = global_var:swe_vasa_house
				ruler ?= { dynasty ?= global_var:swe_vasa_house }
			}
			location:korsholm = { rename_location = vasa }
		}
		else = {
			location:korsholm = { rename_location = mussor }
		}
		location:korsholm = {
			change_development = development_mild_bonus
			change_prosperity = prosperity_mild_bonus
		}
	}

	option = {
		name = swe_towns.3.b
		ai_chance = {
			base = 25
		}
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_weak_bonus }
	}
}

# Borås, 1620-22. The district complained to Gustav II Adolf of how the neighbouring towns treated its peddlers, and he
# told them to stop trading or found a town, chartered in 1621-22 (sv.wiki Borås, citing Millqvist 1991). skene stands
# in for Borås, as it does for the peddlers' district in swe_gotaland.4, which this follows but doesn't need. Option b
# repeats swe_gotaland.4.b, with a market for Skara. swe_boras_founded picks swe_towns.5's text.
swe_towns.8 = {
	type = country_event
	title = swe_towns.8.title
	desc = swe_towns.8.desc
	outcome = neutral
	historical_info = swe_towns.8.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1619.6.1
		to = 1700.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_towns_8
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:skene
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:burghers_estate }
	}

	option = {
		name = swe_towns.8.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -4 }
		location:skene = {
			rename_location = boras
			if = {
				limit = { has_location_modifier = swe_knallar }
				remove_location_modifier = swe_knallar
			}
			if = {
				limit = { has_location_modifier = swe_town_monopoly_enforced }
				remove_location_modifier = swe_town_monopoly_enforced
			}
			add_location_modifier = { modifier = swe_boras_privileges years = 20 }
			swe_grant_market = yes
			change_development = development_mild_bonus
			change_prosperity = prosperity_mild_bonus
		}
		custom_tooltip = {
			text = swe_towns.8.a_tt
			every_owned_location = {
				limit = { province_definition = province_definition:vastergotland_province }
				every_pop = {
					limit = { pop_type = pop_type:burghers }
					add_pop_satisfaction = pop_satisfaction_weak_penalty
				}
			}
		}
		set_variable = swe_boras_founded
	}

	option = {
		name = swe_towns.8.b
		ai_chance = {
			base = 25
		}

		location:skene = {
			if = {
				limit = { has_location_modifier = swe_knallar }
				remove_location_modifier = swe_knallar
			}
			if = {
				limit = { NOT = { has_location_modifier = swe_town_monopoly_enforced } }
				add_location_modifier = { modifier = swe_town_monopoly_enforced years = 10 }
			}
		}
		location:bogesund = {
			if = {
				limit = { owner ?= root }
				if = {
					limit = { has_location_modifier = swe_knallar }
					remove_location_modifier = swe_knallar
				}
				if = {
					limit = { NOT = { has_location_modifier = swe_town_monopoly_enforced } }
					add_location_modifier = { modifier = swe_town_monopoly_enforced years = 10 }
				}
			}
		}
		# The peddlers' trade goes to the chartered town, Skara, the only town in the province at the start
		location:skara = {
			if = {
				limit = { owner ?= root }
				swe_grant_market = yes
			}
		}
		custom_tooltip = {
			text = swe_towns.8.b_tt
			every_owned_location = {
				limit = { province_definition = province_definition:vastergotland_province }
				every_pop = {
					limit = { pop_type = pop_type:burghers }
					add_pop_satisfaction = pop_satisfaction_weak_bonus
				}
			}
		}
	}
}

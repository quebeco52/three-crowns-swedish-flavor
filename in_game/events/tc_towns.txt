namespace = swe_towns

# Town foundings that renamed a location. Balanced against flavor_dan.43 (Christianstad) and
# flavor_swe.41 (Göteborg), but none of them change the location rank: that stays the player's
# decision. Each one is optional and only needs Sweden to own the location at the time.

# Gustav Vasa founds Helsingfors at the mouth of the Vanda (12 June 1550) to take the Gulf of
# Finland trade from Reval, and orders the burghers of Ekenäs, Borgå, Raumo and Ulvsby to move.
# Sweden often holds Reval by then, so owning it changes the text rather than blocking the event. That variant,
# a German Reval that trades on its own account, is an inference, not history: Reval became Swedish only in 1561.
swe_towns.1 = {
	type = country_event
	title = swe_towns.1.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { owns = location:tallinn }
				desc = swe_towns.1.desc_tallinn
			}
			triggered_desc = {
				desc = swe_towns.1.desc
			}
		}
	}
	outcome = neutral
	historical_info = swe_towns.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1548.1.1
		to = 1600.1.1
		monthly_chance = 3 # Median late 1549
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:helsinge
		OR = {
			owns = location:ekenas
			owns = location:porvoo
			owns = location:raumo
			owns = location:bjorneborg
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
	}

	option = {
		name = {
			text = swe_towns.1.a
			trigger = { NOT = { owns = location:tallinn } }
		}
		name = {
			text = swe_towns.1.a_tallinn
			trigger = { owns = location:tallinn }
		}
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -4 }
		location:helsinge = {
			rename_location = helsingfors
			change_development = development_mild_bonus
			change_prosperity = prosperity_mild_bonus
		}
		# Ekenäs and Borgå (Nyland), Raumo and Ulvsby (in bjorneborg, Satakunta)
		# The trigger only needs one of the four, so each move checks its own town. Per month, so 0.48 from each at most.
		# A rural source has no burghers to give, and a town usually holds only one to four.
		if = {
			limit = { owns = location:ekenas }
			add_migration = {
				owner = this
				from_location = location:ekenas
				to_location = location:helsinge
				type = pop_type:burghers
				amount = 0.02
				months = 24
			}
		}
		if = {
			limit = { owns = location:porvoo }
			add_migration = {
				owner = this
				from_location = location:porvoo
				to_location = location:helsinge
				type = pop_type:burghers
				amount = 0.02
				months = 24
			}
		}
		if = {
			limit = { owns = location:raumo }
			add_migration = {
				owner = this
				from_location = location:raumo
				to_location = location:helsinge
				type = pop_type:burghers
				amount = 0.02
				months = 24
			}
		}
		if = {
			limit = { owns = location:bjorneborg }
			add_migration = {
				owner = this
				from_location = location:bjorneborg
				to_location = location:helsinge
				type = pop_type:burghers
				amount = 0.02
				months = 24
			}
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_severe_penalty }
	}

	option = {
		name = swe_towns.1.b
		ai_chance = {
			base = 25
		}

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
	}
}

# Duke Karl, the later Karl IX, charters the market at Tingvalla as Karlstad (March 1584),
# the first town of Värmland.
swe_towns.2 = {
	type = country_event
	title = swe_towns.2.title
	desc = swe_towns.2.desc
	outcome = neutral
	historical_info = swe_towns.2.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1582.1.1
		to = 1640.1.1
		monthly_chance = 3 # Median late 1583
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:tingvalla
	}

	illustration_tags = {
		10 = happy
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:peasants_estate }
	}

	option = {
		name = swe_towns.2.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -2 }
		location:tingvalla = {
			rename_location = karlstad
			change_development = development_mild_bonus
			change_prosperity = prosperity_mild_bonus
		}
		# The privilege letter of 15 March 1584 called out all Värmland to help build two bridges, so the
		# anger stays in Värmland, in place of a country-wide estate penalty
		custom_tooltip = {
			text = swe_towns.2.a_tt
			every_owned_location = {
				limit = { province_definition = province_definition:varmland_province }
				every_pop = {
					limit = { pop_type = pop_type:peasants }
					add_pop_satisfaction = pop_satisfaction_weak_penalty
				}
			}
		}
	}

	option = {
		name = swe_towns.2.b
		ai_chance = {
			base = 25
		}
		custom_tooltip = {
			text = swe_towns.2.b_tt
			every_owned_location = {
				limit = { province_definition = province_definition:varmland_province }
				every_pop = {
					limit = { pop_type = pop_type:peasants }
					add_pop_satisfaction = pop_satisfaction_weak_bonus
				}
			}
		}
	}
}

# Karl IX founds a town by Korsholm castle (2 October 1606). It was first called Mussor and
# renamed Vasa after the royal house a few years later, so the name follows the ruling house.
swe_towns.3 = {
	type = country_event
	title = swe_towns.3.title
	desc = swe_towns.3.desc
	outcome = neutral
	historical_info = swe_towns.3.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1604.1.1
		to = 1660.1.1
		monthly_chance = 3 # Median late 1605
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:korsholm
	}

	illustration_tags = {
		10 = happy
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:peasants_estate }
	}

	option = {
		name = swe_towns.3.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -2 }
		if = {
			limit = {
				exists = global_var:swe_vasa_house
				ruler ?= { dynasty ?= global_var:swe_vasa_house }
			}
			location:korsholm = { rename_location = vasa }
		}
		else = {
			location:korsholm = { rename_location = mussor }
		}
		location:korsholm = {
			change_development = development_mild_bonus
			change_prosperity = prosperity_mild_bonus
		}
	}

	option = {
		name = swe_towns.3.b
		ai_chance = {
			base = 25
		}
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_weak_bonus }
	}
}

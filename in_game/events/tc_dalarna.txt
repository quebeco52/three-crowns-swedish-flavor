namespace = swe_dalarna

# Events for Dalarna. Sweden owns all of it from 1337 except Idre, so these should fire in nearly every game.

# Klockupproret, the bell revolt of 1531-33, over the bell tax voted at Örebro and Uppsala in 1531 (sv.wiki
# Klockupproret; en.wiki Dalecarlian rebellions). Floda stands in for Ål and Gagnef. Vanilla's flavor_swe.4 covers
# the Reformation side.
swe_dalarna.1 = {
	type = country_event
	title = swe_dalarna.1.title
	desc = swe_dalarna.1.desc
	outcome = neutral
	historical_info = swe_dalarna.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1529.1.1
		to = 1545.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_dalarna_1
	}

	trigger = {
		has_or_had_tag = SWE
		has_ruler = yes
		owns = location:falun
		owns = location:leksand
		owns = location:stora_tuna
	}

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:clergy_estate }
		ruler_or_regent ?= { save_scope_as = target_character }
	}

	option = {
		name = swe_dalarna.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		custom_tooltip = {
			text = swe_dalarna.1.a_tt
			every_owned_location = {
				limit = { province_definition = province_definition:dalarna_province }
				add_location_modifier = { modifier = swe_klockskatt years = 10 }
				# The anger stays in Dalarna
				every_pop = {
					limit = { pop_type = pop_type:peasants }
					add_pop_satisfaction = pop_satisfaction_mild_penalty
				}
			}
		}
		change_gold_effect = { scale = 4 }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_weak_penalty }
	}

	option = {
		name = swe_dalarna.1.b
		ai_chance = {
			base = 25
		}

		change_gold_effect = { scale = 2 }
		custom_tooltip = {
			text = swe_dalarna.1.b_tt
			every_owned_location = {
				limit = { province_definition = province_definition:dalarna_province }
				every_pop = {
					limit = { pop_type = pop_type:peasants }
					add_pop_satisfaction = pop_satisfaction_weak_bonus
				}
			}
		}
	}
}

# Magnus Eriksson's privilege letter of 17 February 1347 to the copper miners of Kopparberget (sv.wiki; SGU;
# sverigeshistoria.se). It pays in people through migration, not in copper output, which falun_base already raises
# by 400%.
swe_dalarna.2 = {
	type = country_event
	title = swe_dalarna.2.title
	desc = swe_dalarna.2.desc
	outcome = neutral
	historical_info = swe_dalarna.2.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1345.1.1
		to = 1400.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_dalarna_2
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:falun
	}

	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_copper.dds"

	option = {
		name = swe_dalarna.2.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		location:falun = {
			add_location_modifier = { modifier = swe_kopparberg_charter years = 50 }
		}
	}

	option = {
		name = swe_dalarna.2.b
		ai_chance = {
			base = 25
		}

		change_gold_effect = { scale = 2 }
		location:falun = {
			add_location_modifier = { modifier = swe_kopparberg_bailiff years = 20 }
		}
		custom_tooltip = {
			text = swe_dalarna.2.b_tt
			location:falun = {
				every_pop = {
					limit = { pop_type = pop_type:peasants }
					add_pop_satisfaction = pop_satisfaction_weak_penalty
				}
			}
		}
	}
}

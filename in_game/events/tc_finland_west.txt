# Content for western Finland: Satakunta, Nyland, Österbotten and Åland. Sweden owns every location these events
# need from 1337, so they should fire in nearly every game unless the land has been lost.

# Albrekt's letter of 12 July 1367 ordered a fortress in "Satagundia" taken down as too burdensome, and a new one
# built in a better place. Some historians read this as Kumo and Korsholm, which is disputed. Modifiers rather than
# a building, so no free castle.
swe_finland_west.1 = {
	type = country_event
	title = swe_finland_west.1.title
	desc = swe_finland_west.1.desc
	outcome = neutral
	historical_info = swe_finland_west.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1365.1.1
		to = 1385.1.1
		monthly_chance = 1 # Plus 3% a month from tc_pulse_finland_west_1 (tc_event_pulses.txt). Median late 1366
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:bjorneborg
		owns = location:korsholm
	}

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:nobles_estate }
	}

	option = {
		name = swe_finland_west.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		# A castle pulled down and built again, at the cost of moving a town (swe_finland_west.7)
		change_gold_effect = { scale = -4 }
		location:bjorneborg = {
			add_location_modifier = { modifier = swe_satakunta_relieved years = 15 }
		}
		location:korsholm = {
			add_location_modifier = { modifier = swe_korsholm_coast_watch years = 50 }
		}
	}

	option = {
		name = swe_finland_west.1.b
		ai_chance = {
			base = 25
		}

		location:bjorneborg = {
			add_location_modifier = { modifier = swe_kumo_castle_burden years = 20 }
		}
	}
}

# The Club War began at Kyrö on 25 November 1596, and Österbotten's joint complaint of 27 December 1599 asked Duke

# full Club War: the disaster version is still in the backlog. Fiefs followed from 1608 (Jakob De la Gardie, Pedersöre
# and Nykarleby), most in 1650-52, and Karl XI reclaimed them on 10 December 1674 (K. V. Åkerblom's chronology in Den
# österbottniska byn, as printed by the Swedish Finn Historical Society). Single source: the 1674 date, six years
# before the Great Reduction, was doubted in the 2026-11-10 review, but no other source turned up either way.

# Peasant sailing: limited by the town law of c.1350 and the ordinance of 1614, probably never strictly enforced,
# and lifted by the decree of 3 December 1765. Locations follow Jutikkala's figures for c.1800: Åland (276 ship
# part-owners) and the coast from Rimito to Sjundeå (141). Sauvo-Karuna had none, so it's left out. Option b runs
# 50 years, not to the decree of 1765, so that looking the other way doesn't outlast every other choice in the mod
# (review 2026-11-10). Option a sends the trade to Åbo for 20 years, so enforcing the law moves trade rather than only
# destroying it. The quarrel stays on these coasts, in place of country-wide estate changes.
swe_finland_west.5 = {
	type = country_event
	title = swe_finland_west.5.title
	desc = swe_finland_west.5.desc
	outcome = neutral
	historical_info = swe_finland_west.5.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1614.1.1
		to = 1700.1.1
		monthly_chance = 1 # Plus 3% a month from tc_pulse_finland_west_5 (tc_event_pulses.txt). Median late 1615
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:kastelholm
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:burghers_estate }
	}

	option = {
		name = swe_finland_west.5.a
		ai_chance = {
			base = 25
		}

		location:kastelholm = {
			add_location_modifier = { modifier = swe_skipper_ban years = 20 }
		}
		location:ekenas = {
			if = {
				limit = { owner ?= root }
				add_location_modifier = { modifier = swe_skipper_ban years = 20 }
			}
		}
		location:sjundea = {
			if = {
				limit = { owner ?= root }
				add_location_modifier = { modifier = swe_skipper_ban years = 20 }
			}
		}
		# The farmers' trade goes through the staple towns, Åbo first among them
		location:abo = {
			if = {
				limit = { owner ?= root }
				add_location_modifier = { modifier = swe_skipper_staple years = 20 }
			}
		}
		# The quarrel stays on these coasts, in place of country-wide estate changes
		custom_tooltip = {
			text = swe_finland_west.5.a_tt
			swe_pop_satisfaction_in = { location = location:abo type = burghers value = pop_satisfaction_mild_bonus }
			swe_pop_satisfaction_in = { location = location:kastelholm type = peasants value = pop_satisfaction_mild_penalty }
			swe_pop_satisfaction_in = { location = location:ekenas type = peasants value = pop_satisfaction_mild_penalty }
			swe_pop_satisfaction_in = { location = location:sjundea type = peasants value = pop_satisfaction_mild_penalty }
		}
		change_societal_value = { type = mercantilism_vs_free_trade value = societal_value_minor_move_to_left }
	}

	option = {
		name = swe_finland_west.5.b
		historical_option = yes
		ai_chance = {
			base = 75
		}

		location:kastelholm = {
			add_location_modifier = { modifier = swe_peasant_skippers years = 50 }
		}
		location:ekenas = {
			if = {
				limit = { owner ?= root }
				add_location_modifier = { modifier = swe_peasant_skippers years = 50 }
			}
		}
		location:sjundea = {
			if = {
				limit = { owner ?= root }
				add_location_modifier = { modifier = swe_peasant_skippers years = 50 }
			}
		}
		custom_tooltip = {
			text = swe_finland_west.5.b_tt
			swe_pop_satisfaction_in = { location = location:kastelholm type = peasants value = pop_satisfaction_mild_bonus }
			swe_pop_satisfaction_in = { location = location:ekenas type = peasants value = pop_satisfaction_mild_bonus }
			swe_pop_satisfaction_in = { location = location:sjundea type = peasants value = pop_satisfaction_mild_bonus }
			swe_pop_satisfaction_in = { location = location:abo type = burghers value = pop_satisfaction_weak_penalty }
		}
		change_societal_value = { type = mercantilism_vs_free_trade value = societal_value_minor_move_to_right }
	}
}

# The ironworks of western Nyland, chartered to spare Sweden's forests: Fiskars 1649, Antskog, Billnäs and Fagervik.
# Sjundeå stands in for them, so Ekenäs keeps its fish. Iron brings it under swe_bergskollegium's location bonus
# without a new tier. Same change_raw_material as vanilla's flavor_swe.39.
swe_finland_west.6 = {
	type = country_event
	title = swe_finland_west.6.title
	desc = swe_finland_west.6.desc
	outcome = neutral
	historical_info = swe_finland_west.6.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1640.1.1
		to = 1720.1.1
		monthly_chance = 1 # Plus 3% a month from tc_pulse_finland_west_6 (tc_event_pulses.txt). Median late 1641
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:sjundea
	}

	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_iron.dds"

	option = {
		name = swe_finland_west.6.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -2 }
		location:sjundea = {
			change_raw_material = goods:iron
			# Finnish ore was poor and soon ran short, so the same cap as Sala in flavor_swe.38
			if = {
				limit = { rgo_level > 1 }
				change_max_raw_material_workers = {
					value = rgo_level
					subtract = 1
					multiply = -1
				}
			}
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
	}

	option = {
		name = swe_finland_west.6.b
		ai_chance = {
			base = 25
		}

		custom_tooltip = {
			text = swe_finland_west.6.b_tt
			every_owned_location = {
				limit = { province_definition = province_definition:nyland_province }
				add_location_modifier = { modifier = swe_nyland_forests years = 20 }
			}
		}
	}
}

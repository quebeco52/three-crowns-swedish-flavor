namespace = swe_finland_west

# Content for western Finland: Satakunta, Nyland, Österbotten and Åland. Sweden owns every location these events
# need from 1337, so they should fire in nearly every game unless the land has been lost.

# Albrekt's letter of 12 July 1367 ordered a fortress in "Satagundia" taken down as too burdensome, and a new one
# built in a better place. Some historians read this as Kumo and Korsholm, which is disputed. Modifiers rather than
# a building, so no free castle.
swe_finland_west.1 = {
	type = country_event
	title = swe_finland_west.1.title
	desc = swe_finland_west.1.desc
	outcome = neutral
	historical_info = swe_finland_west.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1365.1.1
		to = 1385.1.1
		monthly_chance = 3 # Median late 1366
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:bjorneborg
		owns = location:korsholm
	}

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:nobles_estate }
	}

	option = {
		name = swe_finland_west.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -2 }
		location:bjorneborg = {
			add_location_modifier = { modifier = swe_satakunta_relieved years = 25 }
		}
		location:korsholm = {
			add_location_modifier = { modifier = swe_korsholm_coast_watch years = 50 }
		}
	}

	option = {
		name = swe_finland_west.1.b
		ai_chance = {
			base = 25
		}

		location:bjorneborg = {
			add_location_modifier = { modifier = swe_kumo_castle_burden years = 25 }
		}
	}
}

# Peasant sailing: limited by the town law of c.1350 and the ordinance of 1614, probably never strictly enforced,
# and lifted by the decree of 3 December 1765. Locations follow Jutikkala's figures for c.1800: Åland (276 ship
# part-owners) and the coast from Rimito to Sjundeå (141). Sauvo-Karuna had none, so it's left out. Option B's 150
# years ends around 1765.
swe_finland_west.5 = {
	type = country_event
	title = swe_finland_west.5.title
	desc = swe_finland_west.5.desc
	outcome = neutral
	historical_info = swe_finland_west.5.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1614.1.1
		to = 1700.1.1
		monthly_chance = 3 # Median late 1615
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:kastelholm
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:burghers_estate }
	}

	option = {
		name = swe_finland_west.5.a
		ai_chance = {
			base = 25
		}

		location:kastelholm = {
			add_location_modifier = { modifier = swe_skipper_ban years = 50 }
		}
		location:ekenas = {
			if = {
				limit = { owner ?= root }
				add_location_modifier = { modifier = swe_skipper_ban years = 50 }
			}
		}
		location:sjundea = {
			if = {
				limit = { owner ?= root }

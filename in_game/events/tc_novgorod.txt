namespace = swe_novgorod

# The Novgorod protectorate. De la Gardie took Novgorod in July 1611, and its leading men accepted Swedish
# protection and asked for a son of Karl IX as their prince. Sweden held the city until Stolbovo (1617),
# when it gave Novgorod back for Ingria and Kexholm. Fires whenever Sweden holds an Orthodox Novgorod that
# is not its capital, not only in the Time of Troubles, which only changes the description.
# Protection turns the Novgorod lands into a march, vanilla's frontier subject, made the way vanilla
# flavor_fra.1101 makes an appanage; Kexholm and Ingria stay Swedish. Keeping the city instead gives
# claims on whatever part of Kexholm and Ingria others still hold, the price Sweden asked at Stolbovo.
swe_novgorod.1 = {
	type = country_event
	title = swe_novgorod.1.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					any_country = { has_active_disaster = disaster_type:time_of_troubles }
				}
				desc = swe_novgorod.1.desc_troubles
			}
			triggered_desc = {
				desc = swe_novgorod.1.desc
			}
		}
	}
	outcome = neutral
	historical_info = swe_novgorod.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 10 # Median about half a year after Sweden takes the city
	}

	trigger = {
		has_or_had_tag = SWE
		is_subject = no
		location:novgorod ?= {
			owner = root
			is_capital = no
			dominant_religion = religion:orthodox
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:clergy_estate }
		location:novgorod = {
			save_scope_as = swe_novgorod_city
			dominant_culture = { save_scope_as = swe_novgorod_culture }
			dominant_religion = { save_scope_as = swe_novgorod_religion }
		}
	}

	option = {
		name = swe_novgorod.1.a
		historical_option = yes
		# A march can never be annexed, so the AI leans towards keeping the city
		ai_chance = {
			base = 40
		}

		scope:swe_novgorod_city = {
			create_country_from_location = {
				overlord = root
				subject_type = subject_type:march
				save_scope_as = swe_novgorod_state
			}
		}
		custom_tooltip = {
			text = swe_novgorod.1.a_tt
			every_owned_location = {
				limit = {
					swe_novgorod_lands_trigger = yes
					is_capital = no
					dominant_religion = religion:orthodox
				}
				change_location_owner = scope:swe_novgorod_state
			}
		}
	}

	option = {
		name = swe_novgorod.1.b
		ai_chance = {
			base = 60
		}

		custom_tooltip = swe_novgorod.1.b_tt
		swe_claim_kexholm_ingria_effect = yes
	}

	# The march from swe_novgorod.1.a: the city's own culture and faith, and Novgorod's name and arms if that
	# country is gone. Effects as in vanilla flavor_hab.81 and fall_of_delhi.txt (release_bahmanis_effect).
	# Not in the option: while its tooltip is built, scope:swe_novgorod_state is still the city, and the ruler
	# and family effects log errors on it for as long as the popup is open. Scope used here like flavor_maj.
	after = {
		if = {
			limit = { exists = scope:swe_novgorod_state }
			scope:swe_novgorod_state = {
				change_government_type = government_type:monarchy
				change_culture = scope:swe_novgorod_culture
				change_religion = scope:swe_novgorod_religion
				change_culture_for_ruler_and_family = { country = scope:swe_novgorod_state culture = scope:swe_novgorod_culture }
				change_religion_for_ruler_and_family = { country = scope:swe_novgorod_state religion = scope:swe_novgorod_religion }
				if = {
					limit = {
						NOT = { country_exists = c:NOV }
					}
					change_country_name = NOV
					change_country_adjective = NOV_ADJ
					change_country_color = map_NOV
					change_country_flag = NOV
				}
				every_owned_location = {
					add_core = scope:swe_novgorod_state
					change_integration_level = core
				}
			}
		}
	}
}

namespace = swe_finland

# Finland at Mora. When Håkan Magnusson was elected beside his father Magnus Eriksson in February 1362, men
# from Finland took part in the royal election at the Stones of Mora for the first time, as one of the
# lands of the realm. Fires once at that date, or later, as soon as Sweden holds Åbo while it still
# elects its monarchs. Granting it adds swe_finland_at_mora to Sweden's locations in Finland.
swe_finland.1 = {
	type = country_event
	title = swe_finland.1.title
	desc = swe_finland.1.desc
	outcome = neutral
	historical_info = swe_finland.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1362.2.1
		to = 1837.1.1
		monthly_chance = 100
	}

	trigger = {
		has_or_had_tag = SWE
		government_type = government_type:monarchy
		has_estate_privilege = estate_privilege:swe_konungabalken
		owns = location:abo
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:nobles_estate }
	}

	option = {
		name = swe_finland.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		custom_tooltip = {
			text = swe_finland.1.a_tt
			every_owned_location = {
				limit = { area = area:finland_area }
				add_location_modifier = { modifier = swe_finland_at_mora years = modifier_duration_years_very_long }
			}
		}
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_weak_bonus }
	}

	option = {
		name = swe_finland.1.b
		ai_chance = {
			base = 25
		}

		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_weak_bonus }
	}
}

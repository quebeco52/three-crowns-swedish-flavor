namespace = swe_norway

# Events for Norwegian land that Sweden takes. Sweden claimed the Arctic coast for a generation and never held it,
# so this fires whenever Sweden owns the land, not by date.

# The Arctic coast. Swedish bailiffs and birkarlar taxed the Sámi up to the coast from the mid-1500s. Duke Karl
# claimed two-thirds of the Sámi tax after Teusina (1595), Karl IX gave the claim up at Knäred (1613), and Karl X
# Gustav's negotiators gave up Nordland and Vardøhus at Roskilde (Favorin; Odén, Scandia). Settling the coast is the
# non-historical option.
swe_norway.2 = {
	type = country_event
	title = swe_norway.2.title
	desc = swe_norway.2.desc
	outcome = neutral
	historical_info = swe_norway.2.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 15
	}

	trigger = {
		has_or_had_tag = SWE
		any_owned_location = {
			province_definition = province_definition:finnmark_province
			is_coastal = yes
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
	}

	option = {
		name = swe_norway.2.a
		historical_option = yes
		ai_chance = {
			base = 60
		}

		custom_tooltip = {
			text = swe_norway.2.a_tt
			every_owned_location = {
				limit = { province_definition = province_definition:finnmark_province }
				add_location_modifier = { modifier = swe_finnmark_bailiffs years = 20 }
				# The people who pay the tax: fishermen and farmers, and the Sámi where they are still tribesmen
				every_pop = {
					limit = {
						OR = {
							pop_type = pop_type:peasants
							pop_type = pop_type:tribesmen
						}
					}
					add_pop_satisfaction = pop_satisfaction_weak_penalty
				}
			}
		}
	}

	option = {
		name = swe_norway.2.b
		ai_chance = {
			base = 40
		}

		custom_tooltip = {
			text = swe_norway.2.b_tt
			every_owned_location = {
				limit = {
					province_definition = province_definition:finnmark_province
					is_coastal = yes
				}
				add_location_modifier = { modifier = swe_finnmark_settlers years = 20 }
			}
		}
	}
}

namespace = swe_svealand

# Content for Svealand. Sweden owns all of this land from 1337, so these should fire in nearly every game unless it has
# been lost.

# Enköping's Twentieth-Day Market. Gustav Vasa confirmed the town's privileges in 1521 but banned bargaining on the ice,
# probably with little effect (Upplandsmuseet report 2006:24, citing Ljung 1963).
swe_svealand.1 = {
	type = country_event
	title = swe_svealand.1.title
	desc = swe_svealand.1.desc
	outcome = neutral
	historical_info = swe_svealand.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1515.1.1
		to = 1560.1.1
		monthly_chance = 3 # Median late 1516
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:enkoping
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:peasants_estate }
	}

	option = {
		name = swe_svealand.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		location:enkoping = {
			add_location_modifier = { modifier = swe_enkoping_ice_ban years = 15 }
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
	}

	option = {
		name = swe_svealand.1.b
		ai_chance = {
			base = 25
		}

		location:enkoping = {
			add_location_modifier = { modifier = swe_enkoping_ice_market years = 50 }
		}
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_bonus }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_penalty }
	}
}

# The flood of 1661, when about 3,000 men dug through at Södertälje to drain Mälaren (Sjöfartsverket PM 2013). The cut
# fell into disrepair, so its modifier runs out. A real map canal isn't possible: vanilla's canals need predefined
# locations, and Mälaren is a lake.
swe_svealand.2 = {
	type = country_event
	title = swe_svealand.2.title
	desc = swe_svealand.2.desc
	outcome = neutral
	historical_info = swe_svealand.2.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1659.1.1
		to = 1700.1.1
		monthly_chance = 3 # Median late 1660
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:sodertalje
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:burghers_estate }
	}

	option = {
		name = swe_svealand.2.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -4 }
		location:sodertalje = {
			add_location_modifier = { modifier = swe_sodertalje_cut years = 25 }
		}
		# The diggers come from the lake's own provinces, so the anger stays there, in place of a
		# country-wide estate penalty
		custom_tooltip = {
			text = swe_svealand.2.a_tt
			every_owned_location = {
				limit = {
					OR = {
						province_definition = province_definition:uppland_province
						province_definition = province_definition:sodermanland_province
						province_definition = province_definition:vastmanland_province
					}
				}
				every_pop = {
					limit = { pop_type = pop_type:peasants }
					add_pop_satisfaction = pop_satisfaction_mild_penalty
				}
			}
		}
		add_prestige = prestige_mild_bonus
	}

	option = {
		name = swe_svealand.2.b
		ai_chance = {
			base = 25
		}

		custom_tooltip = {
			text = swe_svealand.2.b_tt
			location:stockholm = {
				if = {
					limit = { owner ?= root }
					change_prosperity = prosperity_mild_penalty
				}
			}
			location:enkoping = {
				if = {
					limit = { owner ?= root }
					change_prosperity = prosperity_mild_penalty
				}
			}
			location:strangnas = {
				if = {
					limit = { owner ?= root }
					change_prosperity = prosperity_mild_penalty
				}
			}
			location:vasteras = {
				if = {
					limit = { owner ?= root }
					change_prosperity = prosperity_mild_penalty
				}
			}
			location:sodertalje = {
				change_prosperity = prosperity_mild_penalty
			}
		}
	}
}

# Reinhold Rademacher's manufactory at Eskilstuna, from 1654. Vanilla has no location for Eskilstuna, so Strängnäs stands
# in. It doesn't need Riga. The tools bonus only works if the location already has a tools building, which needs town
# rank, so the migration and burgher power carry it otherwise.
swe_svealand.3 = {
	type = country_event
	title = swe_svealand.3.title
	desc = swe_svealand.3.desc
	outcome = neutral
	historical_info = swe_svealand.3.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1650.1.1
		to = 1700.1.1
		monthly_chance = 3 # Median late 1651
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:strangnas
	}

	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_tools.dds"

	option = {
		name = swe_svealand.3.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -4 }
		location:strangnas = {
			add_location_modifier = { modifier = swe_rademacher_forges years = 25 }
		}
		# His smiths were immigrants. The owner is the source's owner, as in flavor_ven.2, so Sweden needn't hold Riga
		if = {
			limit = {
				location:riga = { has_owner = yes }
			}
			add_migration = {
				owner = location:riga.owner
				from = location:riga.province_definition
				to_location = location:strangnas
				culture = culture:german_baltic
				type = pop_type:burghers
				amount = 0.01 # Per month, so 0.24 in all: only 20 forges were ever built
				months = 24
			}
		}
	}

	option = {
		name = swe_svealand.3.b
		ai_chance = {
			base = 25
		}

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
	}
}

# Uppsala Cathedral. Begun in the 1270s and consecrated in June 1435 by Archbishop Olof Larsson, still without its
# towers (SBL, Olaus Laurentii; Sundquist, Årsboken Uppland 1962; Bengtsson, Upplandsmuseets rapporter 2011:04). The
# Church consecrated it, not the king: Erik of Pomerania had fought the archbishop's election and was in Denmark. No
# royal money is attested after Magnus Eriksson's letters of 1331 (DS 2875, a third of the tithe) and 1338 (DS 3384,
# the masons' privilege), so the cathedral stands either way and paying is the unhistorical option.
swe_svealand.4 = {
	type = country_event
	title = swe_svealand.4.title
	desc = swe_svealand.4.desc
	outcome = neutral
	historical_info = swe_svealand.4.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1433.6.1
		to = 1447.1.1
		monthly_chance = 3 # Median mid-1435
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:uppsala
		religion = religion:catholic
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
		ruler ?= { save_scope_as = target_character }
	}

	option = {
		name = swe_svealand.4.a
		ai_chance = {
			base = 40
		}

		change_gold_effect = { scale = -5 }
		location:uppsala = {
			create_art = {
				quality = 85
				type = work_of_art_type:monument
				key = swe_uppsala_domkyrka
			}
		}
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_weak_bonus }
	}

	option = {
		name = swe_svealand.4.b
		historical_option = yes
		ai_chance = {
			base = 60
		}

		location:uppsala = {
			create_art = {
				quality = 65
				type = work_of_art_type:monument
				key = swe_uppsala_domkyrka
			}
		}
	}
}

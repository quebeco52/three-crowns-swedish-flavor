# Enköping's Twentieth-Day Market. Gustav Vasa confirmed the town's privileges in 1521 but banned bargaining on the ice

	type = country_event
	title = swe_svealand.1.title
	desc = swe_svealand.1.desc
	outcome = neutral
	historical_info = swe_svealand.1.historical_info

		tag = SWE
		tag = SCA
		from = 1519.1.1
		to = 1560.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_svealand_1
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:enkoping
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:peasants_estate }

			base = 75

# Reinhold Rademacher's manufactory at Eskilstuna, from 1654; Strängnäs stands in for Eskilstuna. The tools bonus needs
# a tools building, which needs town rank, so a rural Strängnäs gets only the migration and building efficiency.
swe_svealand.3 = {
	type = country_event
	title = swe_svealand.3.title
	desc = swe_svealand.3.desc
	outcome = neutral
	historical_info = swe_svealand.3.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1652.1.1
		to = 1700.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_svealand_3
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:strangnas
	}

	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_tools.dds"

	option = {
		name = swe_svealand.3.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -4 }
		location:strangnas = {
			add_location_modifier = { modifier = swe_rademacher_forges years = 25 }
		}
		# His smiths were immigrants. The migration's owner is Riga's owner, so Sweden needn't hold Riga. Only into a
		# town or city, since burghers sent to a rural location vanish within a month.
		if = {
			limit = {
				location:riga = { has_owner = yes }
				location:strangnas = { NOT = { location_rank = location_rank:rural_settlement } }
			}
			add_migration = {
				owner = location:riga.owner
				from = location:riga.province_definition
				to_location = location:strangnas
				culture = culture:german_baltic
				type = pop_type:burghers
				amount = 0.01 # Per month. Small, since only 20 forges were ever built
				months = 24
			}
		}
	}

	option = {
		name = swe_svealand.3.b
		ai_chance = {
			base = 25
		}

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
	}
}

# Uppsala's new charter. After the church meeting of 1593 asked for it, Duke Karl and the council named professors and
# sealed the university's privileges in 1595 (uu.se; Annerstedt, Upsala universitets historia I). Option a builds the
# university if Uppsala has none (an event's construct_building ignores town rank), or else pays the chairs, and sets
# swe_uppsala_charter for the clergy literacy in swe_uppsala_mote. The chairs are paid in an else branch, not a second
# has_building check, since has_building is true from the day construction starts.
swe_svealand.5 = {
	type = country_event
	title = swe_svealand.5.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					location:uppsala = { has_building = building_type:university }
				}
				desc = swe_svealand.5.desc
			}
			triggered_desc = {
				desc = swe_svealand.5.desc_found
			}
		}
	}
	outcome = neutral
	historical_info = swe_svealand.5.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1593.1.1
		to = 1640.1.1 # Late enough for a late Reformation
		monthly_chance = 1 # Mostly fired by tc_pulse_svealand_5
	}

	trigger = {
		has_or_had_tag = SWE
		religion = religion:lutheran
		owns = location:uppsala
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
		ruler_or_regent ?= { save_scope_as = target_character }
	}

	option = {
		name = swe_svealand.5.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		if = {
			limit = {
				NOT = {
					location:uppsala = { has_building = building_type:university }
				}
			}
			location:uppsala = {
				construct_building = {
					building_type = building_type:university
					cost_multiplier = 0.50
					cost_multiplier_reason = "academic_cheap_sponsorship"
				}
			}
		}
		else = {
			location:uppsala = {
				change_development = development_severe_bonus
				change_prosperity = prosperity_severe_bonus
			}
			# The tithes of the chairs' parishes no longer reach the treasury
			change_gold_effect = { scale = -3 }
		}
		add_research_progress = research_progress_mild_bonus
		set_variable = swe_uppsala_charter
		custom_tooltip = swe_svealand.5.a_tt
	}

	option = {
		name = swe_svealand.5.b
		ai_chance = {
			base = 25
		}

		# The tithes stay with the Crown
		change_gold_effect = { scale = 2 }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_weak_penalty }
	}
}

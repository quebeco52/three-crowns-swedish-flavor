namespace = swe_church

# The Uppsala Synod of 1593 adopts the Augsburg Confession and grants swe_uppsala_mote.
# Fires under any Lutheran ruler: vanilla's Västerås event (flavor_swe.4) converts the ruler with
# the country, so a Catholic king like Sigismund almost never happens. When the ruler or heir is
# not Lutheran, only the description changes.
# Declining keeps the privilege unlocked for later.
swe_church.1 = {
	type = country_event
	title = swe_church.1.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					ruler ?= { religion != religion:lutheran }
				}
				desc = swe_church.1.desc_ruler
			}
			triggered_desc = {
				trigger = {
					heir ?= { religion != religion:lutheran }
				}
				desc = swe_church.1.desc_heir
			}
			triggered_desc = {
				desc = swe_church.1.desc
			}
		}
	}
	outcome = neutral
	historical_info = swe_church.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1591.6.1
		to = 1700.1.1 # Late enough for a late Reformation
		monthly_chance = 3 # Median spring 1593, ~98% by 1602
	}

	trigger = {
		has_or_had_tag = SWE
		religion = religion:lutheran
		owns = location:uppsala
		NOT = { has_estate_privilege = estate_privilege:swe_uppsala_mote }
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
		# The same order as the descriptions. The synod alone shows no one.
		if = {
			limit = {
				ruler ?= { religion != religion:lutheran }
			}
			ruler = { save_scope_as = target_character }
		}
		else_if = {
			limit = {
				heir ?= { religion != religion:lutheran }
			}
			heir = { save_scope_as = target_character }
		}
	}

	option = {
		name = swe_church.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		unlock_estate_privilege_effect = { type = swe_uppsala_mote }
		grant_estate_privilege = estate_privilege:swe_uppsala_mote
		change_societal_value = { type = spiritualist_vs_humanist value = societal_value_minor_move_to_left }
	}

	option = {
		name = swe_church.1.b
		ai_chance = {
			base = 25
		}

		unlock_estate_privilege_effect = { type = swe_uppsala_mote }
		custom_tooltip = swe_church.1.b_tt
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_penalty }
	}
}

# The schools in the early Reformation. Gustav Vasa's open letter of 1533 says the town schools that once had two
# hundred pupils or more had barely fifty, and some none (Evertsson, Svensk Teologisk Kvartalskrift 2019, citing GR 1533;
# Annerstedt 1877, 58). The Crown answered with the parishes' money (St Olof's penny for the Uppsala schoolmaster in
# 1534, fines for withheld tithe in 1540), took the best pupils for its own offices (GR 1544, the demand on Agricola) and
# left the rebuilding to the cathedral chapters. So the choice is where the few scholars go: the towns' schools, the
# Crown's cabinet, or the clergy's goodwill. Fires once vanilla's flavor_swe.4 has made Sweden Lutheran. Option a reaches
# towns and cities of the old realm only (swe_old_realm_trigger), since rural locations hold almost no burghers.
swe_church.2 = {
	type = country_event
	title = swe_church.2.title
	desc = swe_church.2.desc
	outcome = neutral
	historical_info = swe_church.2.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1531.1.1
		to = 1570.1.1 # Late enough for a late Reformation
		monthly_chance = 3 # Median early 1533
	}

	trigger = {
		has_or_had_tag = SWE
		religion = religion:lutheran
		any_owned_location = {
			swe_old_realm_trigger = yes
			NOT = { location_rank = location_rank:rural_settlement }
		}
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:burghers_estate }
		ruler_or_regent ?= { save_scope_as = target_character }
	}

	option = {
		name = swe_church.2.a
		historical_option = yes
		ai_chance = {
			base = 50
		}

		custom_tooltip = {
			text = swe_church.2.a_tt
			every_owned_location = {
				limit = {
					swe_old_realm_trigger = yes
					NOT = { location_rank = location_rank:rural_settlement }
				}
				add_location_modifier = { modifier = swe_stadsskola years = 20 }
			}
		}
		# Every parish of the realm owed the tithe, so the whole estate pays
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_weak_penalty }
	}

	option = {
		name = swe_church.2.b
		ai_chance = {
			base = 25
		}

		add_country_modifier = { modifier = swe_chancery_scholars years = 10 }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_weak_penalty }
	}

	option = {
		name = swe_church.2.c
		ai_chance = {
			base = 25
		}

		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }
	}
}

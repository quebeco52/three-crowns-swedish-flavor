namespace = swe_council

# The German half of the town councils (swe_tyska_radet), from game start to swe_svenska_radman_decision.

# The Käpplinge murders (17 June 1389). With King Albrecht captured at Åsle, the German party in
# Stockholm, the Hättebröder, burned Swedish burghers they took for traitors on Käpplingeholmen.
# Fires around that date, in a war, while the German half of the council still sits. It needs neither a
# German king, since nothing in the game puts one on the Swedish throne, nor a German majority, since
# Stockholm's is thin at game start. Punishing them cuts the German burghers of Stockholm by a quarter,
# enough at game start to open swe_svenska_radman_decision.
swe_council.1 = {
	type = country_event
	title = swe_council.1.title
	desc = swe_council.1.desc
	outcome = neutral
	historical_info = swe_council.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1387.1.1
		to = 1420.1.1
		monthly_chance = 3 # Median early 1389 if Sweden is at war
	}

	trigger = {
		has_or_had_tag = SWE
		at_war = yes
		owns = location:stockholm
		location:stockholm = {
			has_location_modifier = swe_tyska_radet
		}
	}

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:nobles_estate }
	}

	option = {
		name = swe_council.1.a
		historical_option = yes
		ai_chance = {
			base = 50
		}

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_penalty }
	}

	option = {
		name = swe_council.1.b
		ai_chance = {
			base = 50
		}

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_penalty }
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_bonus }
		location:stockholm = {
			every_pop = {
				limit = {
					pop_type = pop_type:burghers
					culture = { has_culture_group = culture_group:german_group }
				}
				add_pop_size = {
					value = pop_size
					multiply = -0.25
				}
			}
		}
	}
}

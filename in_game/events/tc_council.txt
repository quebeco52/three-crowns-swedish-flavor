# The Käpplinge murders, 1389. With King Albrekt captured at Åsle, the Hättebröder, Stockholm's German party,
# burned Swedish burghers they took for traitors. Fires in a war while the German half of the council still sits.
# Punishing them cuts Stockholm's German burghers by a quarter, enough at game start to open
# swe_svenska_radman_decision.
swe_council.1 = {
	type = country_event
	title = swe_council.1.title
	desc = swe_council.1.desc
	outcome = neutral
	historical_info = swe_council.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1387.1.1
		to = 1420.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_council_1
	}

	trigger = {
		has_or_had_tag = SWE
		at_war = yes
		owns = location:stockholm
		location:stockholm = {
			has_location_modifier = swe_tyska_radet
		}
	}

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:nobles_estate }
		ruler_or_regent ?= { save_scope_as = target_character }
	}

	option = {
		name = swe_council.1.a
		historical_option = yes
		ai_chance = {
			base = 50
		}

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_severe_penalty }
	}

	option = {
		name = swe_council.1.b
		ai_chance = {
			base = 50
		}

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_severe_penalty }
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_bonus }
		location:stockholm = {
			every_pop = {
				limit = {
					pop_type = pop_type:burghers
					culture = { has_culture_group = culture_group:german_group }
				}
				add_pop_size = {
					value = pop_size
					multiply = -0.25
				}
			}
		}
	}
}

# Gustav Vasa's crown farms (avelsgårdar), a programme of autumn 1555 (Nyström, Scandia 1936). It was carried out
# mostly in Finland, where about 50 new estates took about 270 farms (Haggrén 2016). The farms go to the forts we own
# when it fires.
swe_council.3 = {
	type = country_event
	title = swe_council.3.title
	desc = swe_council.3.desc
	outcome = neutral
	historical_info = swe_council.3.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1553.11.1
		to = 1575.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_council_3
	}

	trigger = {
		has_or_had_tag = SWE
		any_owned_location = { swe_crown_castle_trigger = yes }
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:nobles_estate }
		ruler_or_regent ?= { save_scope_as = target_character }
	}

	option = {
		name = swe_council.3.a
		historical_option = yes
		ai_chance = {
			base = 60
		}

		custom_tooltip = {
			text = swe_council.3.a_tt
			every_owned_location = {
				limit = { swe_crown_castle_trigger = yes }
				add_location_modifier = { modifier = swe_avelsgard years = 15 }
				# The anger stays at the castles
				every_pop = {
					limit = { pop_type = pop_type:peasants }
					add_pop_satisfaction = pop_satisfaction_mild_penalty
				}
			}
		}
	}

	# In 1556 Hälsingland, Medelpad and Ångermanland offered more rent to be spared, and paid a butter tax instead
	option = {
		name = swe_council.3.b
		ai_chance = {
			base = 40
		}

		change_gold_effect = { scale = 1 }
	}
}

# Landskapens fänikor. At the Riksdag of Västerås in 1544 Gustav Vasa was allowed to keep the peasant bands raised in
# the Dacke war as standing provincial fänikor of about 500 (Söderberg 1944; Högman). Before this the Crown hired
# companies paid by a tax in lieu of service (Neuding Skoog 2018). Fires in the first years of peace after a war,
# until vanilla's allotment system in 1637; swe_vasa.3 covers the Riksdag itself.

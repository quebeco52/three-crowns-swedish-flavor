namespace = swe_council

# Government and justice. .1 concerns the German half of the town councils (swe_tyska_radet), from game start to
# swe_svenska_radman_decision.

# The Käpplinge murders (17 June 1389). With King Albrecht captured at Åsle, the German party in
# Stockholm, the Hättebröder, burned Swedish burghers they took for traitors on Käpplingeholmen.
# Fires around that date, in a war, while the German half of the council still sits. It needs neither a
# German king, since nothing in the game puts one on the Swedish throne, nor a German majority, since
# Stockholm's is thin at game start. Punishing them cuts the German burghers of Stockholm by a quarter,
# enough at game start to open swe_svenska_radman_decision.
swe_council.1 = {
	type = country_event
	title = swe_council.1.title
	desc = swe_council.1.desc
	outcome = neutral
	historical_info = swe_council.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1387.1.1
		to = 1420.1.1
		monthly_chance = 3 # Median early 1389 if Sweden is at war
	}

	trigger = {
		has_or_had_tag = SWE
		at_war = yes
		owns = location:stockholm
		location:stockholm = {
			has_location_modifier = swe_tyska_radet
		}
	}

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:nobles_estate }
		ruler ?= { save_scope_as = target_character }
	}

	option = {
		name = swe_council.1.a
		historical_option = yes
		ai_chance = {
			base = 50
		}

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_severe_penalty }
	}

	option = {
		name = swe_council.1.b
		ai_chance = {
			base = 50
		}

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_severe_penalty }
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_bonus }
		location:stockholm = {
			every_pop = {
				limit = {
					pop_type = pop_type:burghers
					culture = { has_culture_group = culture_group:german_group }
				}
				add_pop_size = {
					value = pop_size
					multiply = -0.25
				}
			}
		}
	}
}

# Karl Knutsson's räfst, November 1453 to March 1454 (SBL, Karl Knutsson). The modifier goes on the judges' eight stops
# rather than the whole realm. No gold, since vanilla's perform_reduction already takes estate money by hand. The clergy
# penalty rests on the chapters' protest of 15 January 1454. Whether the räfst really pushed back the church's gains is
# disputed, so the text doesn't say it did. The description names nobody, because any ruler can hold the throne by now.
swe_council.2 = {
	type = country_event
	title = swe_council.2.title
	desc = swe_council.2.desc
	outcome = neutral
	historical_info = swe_council.2.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1451.1.1
		to = 1470.1.1
		monthly_chance = 3 # Median late 1452
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:uppsala
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:clergy_estate }
		ruler ?= { save_scope_as = target_character }
	}

	option = {
		name = swe_council.2.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		custom_tooltip = {
			text = swe_council.2.a_tt
			every_owned_location = {
				limit = {
					OR = {
						this = location:vaxjo
						this = location:jonkoping
						this = location:skara
						this = location:orebro
						this = location:vasteras
						this = location:uppsala
						this = location:strangnas
						this = location:linkoping
					}
				}
				add_location_modifier = { modifier = swe_rafst years = 20 }
			}
		}
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_penalty }
	}

	option = {
		name = swe_council.2.b
		ai_chance = {
			base = 25
		}

		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_weak_bonus }
	}
}

namespace = swe_ostergotland

# Province and territory events for Östergötland. Sweden owns all of it from 1337, so these should fire in nearly
# every game unless it has been lost.

# The shrining of Katarina Ulfsdotter, Birgitta's daughter, at Vadstena on 1 August 1489. Sources: Sr Maria Diana
# O.SS.S. for katolskakyrkan.se (citing Aron Andersson 1981), sv.wikipedia Katarina av Vadstena, and the Birgitta
# database of the University of Oslo (citing Gillingstam, SBL). They differ on the first bull (Sixtus IV 1482 or
# Innocent VIII 1484), so the text says the 1480s. No source puts the regent at the ceremony, so the Crown's
# attendance in option a is an inference, not history. Vanilla's flavor_swe.46 and .48 cover Birgitta herself and
# make Vadstena a holy site, which the description reads. The shrine modifier carries no max control. No has_ruler:
# 1489 fell in Sten Sture's regency, and the text names nobody.
swe_ostergotland.1 = {
	type = country_event
	title = swe_ostergotland.1.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					location:vadstena = { is_location_holy_site_for = religion:catholic }
				}
				desc = swe_ostergotland.1.desc_holy_site
			}
			triggered_desc = {
				desc = swe_ostergotland.1.desc
			}
		}
	}
	outcome = neutral
	historical_info = swe_ostergotland.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1487.1.1
		to = 1520.1.1
		monthly_chance = 3 # Median late 1488
	}

	trigger = {
		has_or_had_tag = SWE
		religion = religion:catholic
		owns = location:vadstena
		NOT = { has_variable = swe_vadstena_kept_by_crown } # swe_ostergotland.2.b: Birgitta's house never came to Vadstena
	}

	illustration_tags = {
		10 = happy
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
		ruler_or_regent ?= { save_scope_as = target_character }
	}

	option = {
		name = swe_ostergotland.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -2 }
		location:vadstena = {
			add_location_modifier = { modifier = swe_katarina_shrine years = 20 }
		}
		add_legitimacy = legitimacy_weak_bonus
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }
	}

	option = {
		name = swe_ostergotland.1.b
		ai_chance = {
			base = 25
		}

		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_weak_penalty }
	}
}

# The Bridgettine order founded at Vadstena, about 1370. Magnus Eriksson and Blanka promised their manor at Vadstena to
# Birgitta's monastery in their Swedish testament of 1346 (Berglund, Uppsala University Press; SKBL; Magnus also wished
# to be buried there, Gillingstam in Scandia, citing DS 5 no. 4069). Rebuilding began in the late 1360s, and Urban V's
# bull of 5 August 1370 (Montefiascone) approved the rule and noted a monastery already rising at the manor (Berglund,
# citing SDHK 9751; UiO Birgitta database). The founding was Birgitta's and the approval Rome's; the Crown's act was
# letting the gift stand, which option a does. found_religious_order has no live vanilla use (tested by console
# 2026-11-12: it makes a working "Order of Vadstena"); the monastic type's founding-location check reads
# owner ?= scope:actor, so root saves it. A location has no religious_order link, so the new order is found with
# random_religious_order_in_location. Zeal 50 is the neutral point of vanilla's invite and expel AI; recent_order_service
# is what vanilla sets after every order action (+6 zeal target for five years). The order loses Vadstena when Sweden
# leaves the Catholic faith, and lives on only where other realms have invited it. Option b copies vanilla's refusal of
# a new order holding (religious_orders.2.c), and sets swe_vadstena_kept_by_crown, which blocks swe_ostergotland.1:
# Katarina's shrine belongs to her mother's house.
swe_ostergotland.2 = {
	type = country_event
	title = swe_ostergotland.2.title
	desc = swe_ostergotland.2.desc
	outcome = neutral
	historical_info = swe_ostergotland.2.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1368.8.1
		to = 1430.1.1
		monthly_chance = 3 # Median mid-1370
	}

	trigger = {
		has_or_had_tag = SWE
		religion = religion:catholic
		owns = location:vadstena
		location:vadstena = {
			has_building_with_at_least_one_level = monastery
			dominant_religion = religion:catholic
			location_can_host_order_holding_trigger = yes
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
		save_scope_as = actor # the monastic type's founding check reads owner ?= scope:actor
		ruler_or_regent ?= { save_scope_as = target_character }
		# A second portrait while vanilla's Birgitta (flavor_swe.46) lives
		character:swe_heliga_birgitta ?= {
			if = {
				limit = { is_alive = yes }
				save_scope_as = swe_birgitta
			}
		}
	}

	option = {
		name = swe_ostergotland.2.a
		historical_option = yes
		ai_chance = {
			base = 80
		}

		custom_tooltip = {
			text = swe_ostergotland.2.a_tt
			location:vadstena = {
				found_religious_order = {
					type = religious_order_type:monastic_order_type
					calling = calling:contemplatives_calling
					religion = religion:catholic
				}
				random_religious_order_in_location = { save_scope_as = swe_bridgettine_order }
				set_variable = { name = recent_order_service value = yes days = 1825 }
			}
			if = {
				limit = { exists = scope:swe_bridgettine_order }
				scope:swe_bridgettine_order = { set_religious_order_zeal = 50 }
				create_character = {
					female = yes
					min_age = 30
					max_age = 50
					culture = root.culture
					religion = religion:catholic
					estate = estate_type:clergy_estate
					birth_location = location:vadstena
					religious_order = scope:swe_bridgettine_order
					save_scope_as = swe_vadstena_abbess
				}
				scope:swe_bridgettine_order = { set_head_character = scope:swe_vadstena_abbess }
			}
		}
		add_prestige = prestige_weak_bonus
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_weak_bonus }
	}

	option = {
		name = swe_ostergotland.2.b
		ai_chance = {
			base = 20
		}

		add_political_influence = political_influence_extreme_bonus
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_penalty }
		set_variable = swe_vadstena_kept_by_crown # Blocks swe_ostergotland.1, Katarina's shrine at Birgitta's house
	}
}

# Krigsmanshuset, the house for crippled soldiers at Vadstena. Gustav II Adolf ordered it in the empty convent in 1622
# and gave 2,000 daler; the regency proclaimed the founding on 3 August 1637, and the first men moved into the nuns'
# ranges in the winter of 1640-41. Kristina's grant of farms on 12 November 1646, made after she came of age, is what
# older works give as the founding. Closed 1778-84. Sources: Erik Petersson, Sårade soldater (2016), through Kuvaja's
# review in Historisk Tidskrift för Finland and soldatreg.se (the residents' figures rest on him alone); the Riksdag's
# army pension report of 1913 (DBE82b3); Riksarkivet; Värmlands regementes historia; sv.wikipedia Vadstena kloster.
# Option b is sourced too: most crippled men stayed in their parishes on grain from the fund, and the house closed
# because a fund could help more men than its rooms held (1913 report). Gold and army tradition copy vanilla's
# parliaments.38.b, which pays the veterans. The text names nobody, as 1637 fell in Kristina's minority.
# Leaving the Catholic faith also ends the Bridgettine holding from swe_ostergotland.2. If .2.b kept the manor with
# the Crown, the convent never stood, so the desc speaks of the manor.
swe_ostergotland.3 = {
	type = country_event
	title = swe_ostergotland.3.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = swe_vadstena_kept_by_crown }
				desc = swe_ostergotland.3.desc_manor
			}
			triggered_desc = {
				desc = swe_ostergotland.3.desc
			}
		}
	}
	outcome = neutral
	historical_info = swe_ostergotland.3.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1635.8.1
		to = 1700.1.1
		monthly_chance = 3 # Median mid-1637
	}

	trigger = {
		has_or_had_tag = SWE
		NOT = { religion = religion:catholic }
		owns = location:vadstena
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:nobles_estate }
		ruler_or_regent ?= { save_scope_as = target_character }
	}

	option = {
		name = swe_ostergotland.3.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -2 }
		add_army_tradition = army_tradition_weak_bonus
		add_country_modifier = { modifier = swe_krigsmanshus years = 50 }
	}

	option = {
		name = swe_ostergotland.3.b
		ai_chance = {
			base = 25
		}

		change_gold_effect = { scale = -1 }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_weak_bonus }
	}
}

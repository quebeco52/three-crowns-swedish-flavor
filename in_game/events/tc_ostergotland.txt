namespace = swe_ostergotland

# Östergötland, which Sweden holds from 1337.

# The shrining of Katarina Ulfsdotter, Birgitta's daughter, at Vadstena on 1 August 1489 (katolskakyrkan.se; sv.wiki
# Katarina av Vadstena; UiO Birgitta database). No source puts the regent at the ceremony, so the Crown's part in
# option a is an inference. No has_ruler check, since 1489 fell in Sten Sture's regency.
swe_ostergotland.1 = {
	type = country_event
	title = swe_ostergotland.1.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					location:vadstena = { is_location_holy_site_for = religion:catholic }
				}
				desc = swe_ostergotland.1.desc_holy_site
			}
			triggered_desc = {
				desc = swe_ostergotland.1.desc
			}
		}
	}
	outcome = neutral
	historical_info = swe_ostergotland.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1487.1.1
		to = 1520.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_ostergotland_1
	}

	trigger = {
		has_or_had_tag = SWE
		religion = religion:catholic
		owns = location:vadstena
		NOT = { has_variable = swe_vadstena_kept_by_crown } # Set by swe_ostergotland.2.b, where Birgitta's house never came
	}

	illustration_tags = {
		10 = happy
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
		ruler_or_regent ?= { save_scope_as = target_character }
	}

	option = {
		name = swe_ostergotland.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -2 }
		location:vadstena = {
			add_location_modifier = { modifier = swe_katarina_shrine years = 20 }
		}
		add_legitimacy = legitimacy_weak_bonus
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }
	}

	option = {
		name = swe_ostergotland.1.b
		ai_chance = {
			base = 25
		}

		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_weak_penalty }
	}
}

# The Bridgettine order founded at Vadstena, about 1370. Magnus Eriksson and Blanka promised their manor there to
# Birgitta's monastery in 1346, and Urban V's bull of 5 August 1370 approved the rule (Berglund; SKBL; SDHK 9751). The
# Crown's act was letting the gift stand, which option a does. The order loses Vadstena when Sweden leaves the Catholic
# faith. Option b from vanilla's religious_orders.2.c.
# found_religious_order has no live vanilla use but works. A location has no religious_order link, so the new order is
# found with random_religious_order_in_location. Zeal 50 is the neutral point of vanilla's invite and expel AI, and
# vanilla sets recent_order_service after every order action.
swe_ostergotland.2 = {
	type = country_event
	title = swe_ostergotland.2.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					location:vadstena = { is_holding_of_any_religious_order = yes }
				}
				desc = swe_ostergotland.2.desc_order
			}
			triggered_desc = {
				desc = swe_ostergotland.2.desc
			}
		}
	}
	outcome = neutral
	historical_info = swe_ostergotland.2.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1368.8.1
		to = 1430.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_ostergotland_2
	}

	trigger = {
		has_or_had_tag = SWE
		religion = religion:catholic
		owns = location:vadstena
		location:vadstena = {
			has_building_with_at_least_one_level = monastery
			dominant_religion = religion:catholic
			# The half of location_can_host_order_holding_trigger that still applies. An existing order doesn't block
			# the event, since Sweden's only monasteries in 1337 (Vadstena, Alvastra, Riseberga) can all go to the
			# Cistercians. Option a moves that order out, as vanilla's reformation.15.b does.
			NOT = { is_capital = yes }
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
		save_scope_as = actor # the monastic type's founding check reads owner ?= scope:actor
		ruler_or_regent ?= { save_scope_as = target_character }
		# A second portrait while vanilla's Birgitta (flavor_swe.46) lives
		character:swe_heliga_birgitta ?= {
			if = {
				limit = { is_alive = yes }
				save_scope_as = swe_birgitta
			}
		}
	}

	option = {
		name = swe_ostergotland.2.a
		historical_option = yes
		ai_chance = {
			base = 80
		}

		if = {
			limit = {
				location:vadstena = { is_holding_of_any_religious_order = yes }
			}
			custom_tooltip = {
				text = swe_ostergotland.2.a_order_tt
				location:vadstena = {
					every_religious_order_in_location = {
						change_religious_order_zeal = -5
						save_scope_as = swe_vadstena_old_order
						location:vadstena = { remove_religious_order = scope:swe_vadstena_old_order }
					}
				}
			}
		}
		custom_tooltip = {
			text = swe_ostergotland.2.a_tt
			location:vadstena = {
				found_religious_order = {
					type = religious_order_type:monastic_order_type
					calling = calling:contemplatives_calling
					religion = religion:catholic
				}
				random_religious_order_in_location = { save_scope_as = swe_bridgettine_order }
				set_variable = { name = recent_order_service value = yes days = 1825 }
			}
			if = {
				limit = { exists = scope:swe_bridgettine_order }
				scope:swe_bridgettine_order = { set_religious_order_zeal = 50 }
				create_character = {
					female = yes
					min_age = 30
					max_age = 50
					culture = root.culture
					religion = religion:catholic
					estate = estate_type:clergy_estate
					birth_location = location:vadstena
					religious_order = scope:swe_bridgettine_order
					save_scope_as = swe_vadstena_abbess
				}
				scope:swe_bridgettine_order = { set_head_character = scope:swe_vadstena_abbess }
			}
		}
		add_prestige = prestige_weak_bonus
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_weak_bonus }
	}

	option = {
		name = swe_ostergotland.2.b
		ai_chance = {
			base = 20
		}

		add_political_influence = political_influence_extreme_bonus
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_penalty }
		set_variable = swe_vadstena_kept_by_crown # Blocks swe_ostergotland.1, Katarina's shrine at Birgitta's house
	}
}

# Krigsmanshuset, the house for crippled soldiers at Vadstena. Gustav II Adolf ordered it in the empty convent in 1622,
# the regency proclaimed the founding on 3 August 1637 (older works give Kristina's grant of 1646), and the first men
# moved into the nuns' ranges in the winter of 1640-41 (Petersson, Sårade soldater 2016; Riksdag army pension report
# 1913; sv.wiki Vadstena kloster). Option b follows the 1913 report: most crippled men stayed in their parishes on grain from the
# fund. Values from vanilla's parliaments.38.b. If .2.b kept the manor with the Crown, the desc speaks of the manor.
swe_ostergotland.3 = {
	type = country_event
	title = swe_ostergotland.3.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = swe_vadstena_kept_by_crown }
				desc = swe_ostergotland.3.desc_manor
			}
			triggered_desc = {
				desc = swe_ostergotland.3.desc
			}
		}
	}
	outcome = neutral
	historical_info = swe_ostergotland.3.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1635.8.1
		to = 1700.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_ostergotland_3
	}

	trigger = {
		has_or_had_tag = SWE
		NOT = { religion = religion:catholic }
		owns = location:vadstena
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:nobles_estate }
		ruler_or_regent ?= { save_scope_as = target_character }
	}

	option = {
		name = swe_ostergotland.3.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -2 }
		add_army_tradition = army_tradition_weak_bonus
		add_country_modifier = { modifier = swe_krigsmanshus years = 50 }
	}

	option = {
		name = swe_ostergotland.3.b
		ai_chance = {
			base = 25
		}

		change_gold_effect = { scale = -1 }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_weak_bonus }
	}
}

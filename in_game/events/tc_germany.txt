namespace = swe_germany

# Content for German land beyond Pomerania, which Sweden held only in war or as fiefs of the Empire. These fire
# whenever Sweden owns the land, not by date.

# The council of Bremen. Westphalia (1648) gave Sweden the duchy of Bremen without saying whether the city, free since
# the Linz diploma of 1646, went with it. The city swore homage to the king as duke in 1654 and gave up its immediacy
# at Habenhausen in 1666 until the end of the century, but was never brought into the duchy (Lundqvist 1893; Eichberg
# 1975; en.wiki Swedish wars on Bremen). swe_germany.3 covers the rival town Sweden built at the mouth of the river.
swe_germany.1 = {
	type = country_event
	title = swe_germany.1.title
	desc = swe_germany.1.desc
	outcome = neutral
	historical_info = swe_germany.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 15
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:bremen
		location:bremen = {
			NOT = { location_rank = location_rank:rural_settlement }
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
		ruler_or_regent ?= { save_scope_as = target_character }
	}

	# The homage of 1654
	option = {
		name = swe_germany.1.a
		historical_option = yes
		ai_chance = {
			base = 60
		}

		location:bremen = {
			add_location_modifier = { modifier = swe_bremen_homage years = modifier_duration_years_very_long }
		}
		custom_tooltip = {
			text = swe_germany.1.a_tt
			swe_pop_satisfaction_in = { location = location:bremen type = burghers value = pop_satisfaction_weak_penalty }
		}
	}

	# The city's freedoms, which it has kept under every lord
	option = {
		name = swe_germany.1.b
		ai_chance = {
			base = 40
		}

		location:bremen = {
			if = {
				limit = { NOT = { has_location_modifier = swe_baltic_town_privileges } }
				add_location_modifier = { modifier = swe_baltic_town_privileges years = modifier_duration_years_very_long }
			}
		}
		custom_tooltip = {
			text = swe_germany.1.b_tt
			swe_pop_satisfaction_in = { location = location:bremen type = burghers value = pop_satisfaction_weak_bonus }
		}
	}
}

# The Franconian sees. Gustav II Adolf's army stormed the Marienberg above Würzburg in October 1631. In June 1633, after
# the army mutinied for its arrears, Oxenstierna granted Bernhard of Saxe-Weimar the sees of Würzburg and Bamberg as the
# Duchy of Franconia, a fief of the Swedish crown, lost after Nördlingen in 1634 (Roberts; NDB; ADB). NDB alone leaves
# the fortresses out of the grant, so just Königshofen is held back here. The duchy is a modifier, not a vassal.
swe_germany.2 = {
	type = country_event
	title = swe_germany.2.title
	desc = swe_germany.2.desc
	outcome = neutral
	historical_info = swe_germany.2.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 15
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:wurzburg
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
		ruler_or_regent ?= { save_scope_as = target_character }
	}

	# The letter of donation, June 1633
	option = {
		name = swe_germany.2.a
		historical_option = yes
		ai_chance = {
			base = 60
		}

		change_gold_effect = { scale = 4 }
		custom_tooltip = {
			text = swe_germany.2.a_tt
			every_owned_location = {
				limit = { swe_franconia_trigger = yes }
				if = {
					limit = { NOT = { has_location_modifier = swe_war_contributions } }
					add_location_modifier = { modifier = swe_war_contributions years = 10 }
				}
				# The crown kept the fortress of Königshofen out of the duchy
				if = {
					limit = {
						NOT = { this = location:konigshofen }
						NOT = { has_location_modifier = swe_franconian_duchy }
					}
					add_location_modifier = { modifier = swe_franconian_duchy years = 20 }
				}
			}
		}
	}

	# The king's own government, October 1631
	option = {
		name = swe_germany.2.b
		ai_chance = {
			base = 40
		}

		custom_tooltip = {
			text = swe_germany.2.b_tt
			every_owned_location = {
				limit = {
					swe_franconia_trigger = yes
					NOT = { has_location_modifier = swe_franconian_government }
				}
				add_location_modifier = { modifier = swe_franconian_government years = 20 }
			}
		}
		# The officers stay unpaid
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_weak_penalty }
	}
}

# The mouth of the Geeste. Having failed to take Bremen in 1666, Sweden tried to win the quarrel with trade: Henrik Horn
# broke ground for Carlsburg in 1672, but the settlers didn't come, and Karl XI's economies had the works pulled down
# from 1683 (Eichberg 1975; en.wiki Carlsburg, Weser). Needs Bremen outside our realm, since the town was built to take
# its trade. The Swedish name sits on the carlsburg key, so it shows only once the town is founded.
swe_germany.3 = {
	type = country_event
	title = swe_germany.3.title
	desc = swe_germany.3.desc
	outcome = neutral
	historical_info = swe_germany.3.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 15
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:gestenthorpe
		NOT = { location:bremen.owner ?= { swe_is_swedish_realm_trigger = yes } }
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
		ruler_or_regent ?= { save_scope_as = target_character }
	}

	# Carlsburg, 1672
	option = {
		name = swe_germany.3.a
		historical_option = yes
		ai_chance = {
			base = 60
		}

		change_gold_effect = { scale = -4 }
		# Renamed first and the modifier added after, in the same scope
		location:gestenthorpe = {
			rename_location = carlsburg
			add_location_modifier = { modifier = swe_carlsburg years = 20 }
		}
	}

	# The marsh, as Karl XI's economies left it from 1683
	option = {
		name = swe_germany.3.b
		ai_chance = {
			base = 40
		}
	}
}

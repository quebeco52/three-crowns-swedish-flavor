namespace = swe_baltic

# The Baltic capitulations. As the Livonian Confederation broke up, its German knighthoods and towns
# submitted to Sweden on terms: Estonia and Reval in 1561, Riga in 1621, Livonia in the 1620s, Ösel in
# 1645. Each event fires whenever Sweden owns the chief town while Baltic Germans still live there, not
# by date. Confirming the privileges speeds integration and slows assimilation, as vanilla's
# lat_compromise does for the Greeks; Swedish law speeds assimilation through swe_forsvenskning.
# Nobody's culture changes at once. The variables mark the choice for later content.

# Estonia: the knighthoods of Harrien, Wierland and Jerwen and the town of Reval submitted to Erik XIV
# in 1561, and he confirmed their privileges. Ösel and Wiek (Rotalia) go with it: mainland Wiek was Swedish
# from the 1580s, and the island of Ösel was Danish from 1559 until Brömsebro (1645), when its knighthood kept
# the privileges it had held under Denmark.
swe_baltic.1 = {
	type = country_event
	title = swe_baltic.1.title
	desc = swe_baltic.1.desc
	outcome = neutral
	historical_info = swe_baltic.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 3 # Median about two years after Sweden takes Reval
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:tallinn
		location:tallinn = {
			any_pop = { culture = culture:german_baltic }
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:burghers_estate }
	}

	option = {
		name = swe_baltic.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		custom_tooltip = {
			text = swe_baltic.1.a_tt
			every_owned_location = {
				limit = {
					swe_estonia_trigger = yes
					NOT = { this = location:tallinn }
					NOT = { this = location:narva }
					NOT = { has_location_modifier = swe_baltic_privileges }
				}
				add_location_modifier = { modifier = swe_baltic_privileges years = modifier_duration_years_very_long }
			}
		}
		# Reval and Narva were Hanseatic towns under Lübeck law: their councils, not the knighthood, kept the privileges
		every_owned_location = {
			limit = {
				OR = {
					this = location:tallinn
					this = location:narva
				}
				NOT = { has_location_modifier = swe_baltic_town_privileges }
			}
			add_location_modifier = { modifier = swe_baltic_town_privileges years = modifier_duration_years_very_long }
		}
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_weak_bonus }
	}

	option = {
		name = swe_baltic.1.b
		ai_chance = {
			base = 25
		}

		custom_tooltip = {
			text = swe_baltic.1.b_tt
			every_owned_location = {
				limit = {
					swe_estonia_trigger = yes
					NOT = { dominant_culture = culture:swedish }
					NOT = { has_location_modifier = swe_forsvenskning }
				}
				add_location_modifier = { modifier = swe_forsvenskning years = 20 }
			}
		}
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_penalty }
	}
}

# The Russian staple. Stolbovo (1617) cut Russia off from the Baltic, and Sweden tried to draw Russia's
# trade with the West through Narva, Nyen and Reval; Narva became the main one. Fires from the age of
# the Reformation, while Sweden holds one of the three and whoever owns Novgorod has no Baltic port.
# Whichever of them Sweden holds then get swe_russian_transit, and the chosen staple also gets swe_russian_staple.
# Both run for 50 years even if Russia returns to the Baltic: a location modifier can't check that later.
# The prosperity of flavor_tre.5 (the trade port of Kerasous left to prosper).
swe_baltic.4 = {
	type = country_event
	title = swe_baltic.4.title
	desc = swe_baltic.4.desc
	outcome = positive
	historical_info = swe_baltic.4.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 3 # Median about two years after Russia loses its last Baltic port
	}

	trigger = {
		has_or_had_tag = SWE
		current_age_or_later = {
			age = age_4_reformation
		}
		OR = {
			owns = location:narva
			owns = location:neva
			owns = location:tallinn
		}
		swe_russia_shut_out_trigger = yes
	}

	illustration_tags = {
		10 = happy
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
		# The same "Russia" as swe_russian_port_trigger
		if = {
			limit = {
				location:novgorod.owner ?= { NOT = { swe_is_swedish_realm_trigger = yes } }
			}
			location:novgorod.owner = { save_scope_as = swe_russian_hinterland }
		}
		else = {
			location:moscow.owner ?= { save_scope_as = swe_russian_hinterland }
		}
		# The same whichever town becomes the staple
		every_owned_location = {
			limit = {
				swe_russian_transit_port_trigger = yes
				NOT = { has_location_modifier = swe_russian_transit }
			}
			add_location_modifier = { modifier = swe_russian_transit years = modifier_duration_years_very_long }
		}
	}

	# Narva
	option = {
		name = swe_baltic.4.a
		historical_option = yes
		trigger = {
			owns = location:narva
		}
		ai_chance = {
			base = 50
		}

		custom_tooltip = swe_baltic.4.transit_tt
		location:narva = {
			add_location_modifier = { modifier = swe_russian_staple years = modifier_duration_years_very_long }
			change_prosperity = prosperity_mild_bonus
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
	}

	# Nyen
	option = {
		name = swe_baltic.4.b
		trigger = {
			owns = location:neva
		}
		ai_chance = {
			base = 25
		}

		custom_tooltip = swe_baltic.4.transit_tt
		location:neva = {
			add_location_modifier = { modifier = swe_russian_staple years = modifier_duration_years_very_long }
			change_prosperity = prosperity_mild_bonus
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
	}

	# Reval
	option = {
		name = swe_baltic.4.c
		trigger = {
			owns = location:tallinn
		}
		ai_chance = {
			base = 25
		}

		custom_tooltip = swe_baltic.4.transit_tt
		location:tallinn = {
			add_location_modifier = { modifier = swe_russian_staple years = modifier_duration_years_very_long }
			change_prosperity = prosperity_mild_bonus
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
	}
}

# Courland. Sweden took the licent at Windau and Libau under the Truce of Altmark, against dukes who kept their dues
# low to draw trade (Jakovļeva 2012). Douglas seized Mitau and Duke Jakob in 1658, and Karl XII occupied the duchy in
# 1701, partly to end the maritime powers' protests at the blockade of its harbours (Rystad, Scandia 1961); its
# estates paid heavy contributions to his army (Jakovļeva 2022). Its knighthood never capitulated to Sweden, unlike
# Estonia's and Livonia's, so there are no privileges to confirm. Fires whenever Sweden owns Mitau, Windau or Grobin.
swe_baltic.5 = {
	type = country_event
	title = swe_baltic.5.title
	desc = swe_baltic.5.desc
	outcome = neutral
	historical_info = swe_baltic.5.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 3 # Median about two years after Sweden takes Courland
	}

	trigger = {
		has_or_had_tag = SWE
		OR = {
			owns = location:jelgava
			owns = location:ventspils
			owns = location:grobina
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:burghers_estate }
	}

	# Contributions, as under Karl XII
	option = {
		name = swe_baltic.5.a
		historical_option = yes
		ai_chance = {
			base = 40
		}

		change_gold_effect = { scale = 4 }
		custom_tooltip = {
			text = swe_baltic.5.a_tt
			every_owned_location = {
				limit = {
					swe_courland_trigger = yes
					NOT = { has_location_modifier = swe_courland_contributions }
				}
				add_location_modifier = { modifier = swe_courland_contributions years = 10 }
			}
		}
	}

	# The licent, as in the 1630s
	option = {
		name = swe_baltic.5.b
		trigger = {
			OR = {
				owns = location:ventspils
				owns = location:grobina
			}
		}
		ai_chance = {
			base = 35
		}

		custom_tooltip = {
			text = swe_baltic.5.b_tt
			every_owned_location = {
				limit = {
					swe_courland_harbour_trigger = yes
					NOT = { has_location_modifier = swe_riga_licent }
				}
				add_location_modifier = { modifier = swe_riga_licent years = modifier_duration_years_very_long }
			}
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_penalty }
	}

	# The dukes' low dues
	option = {
		name = swe_baltic.5.c
		trigger = {
			OR = {
				owns = location:ventspils
				owns = location:grobina
			}
		}
		ai_chance = {
			base = 25
		}

		custom_tooltip = {
			text = swe_baltic.5.c_tt
			every_owned_location = {
				limit = {
					swe_courland_harbour_trigger = yes
					NOT = { has_location_modifier = swe_courland_low_dues }
				}
				add_location_modifier = { modifier = swe_courland_low_dues years = modifier_duration_years_very_long }
			}
		}
	}
}

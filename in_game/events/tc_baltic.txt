# The Baltic capitulations. As the Livonian Confederation broke up, its German knighthoods and towns submitted to
# Sweden on terms: Estonia and Reval in 1561, Riga in 1621, Livonia in the 1620s, Ösel in 1645. Each event fires when
# Sweden owns the chief town while Baltic Germans still live there. Confirming the privileges speeds integration and
# slows assimilation, as vanilla's lat_compromise does.

# Estonia: the knighthoods of Harrien, Wierland and Jerwen and the town of Reval submitted to Erik XIV in 1561, and he
# confirmed their privileges. Ösel and Wiek (Rotalia) go with it: mainland Wiek was Swedish from the 1580s, and Ösel's
# knighthood kept its Danish-era privileges at Brömsebro (1645).
swe_baltic.1 = {
	type = country_event
	title = swe_baltic.1.title
	desc = swe_baltic.1.desc
	outcome = neutral
	historical_info = swe_baltic.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 15
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:tallinn
		location:tallinn = {
			any_pop = { culture = culture:german_baltic }
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:burghers_estate }
	}

	option = {
		name = swe_baltic.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		custom_tooltip = {
			text = swe_baltic.1.a_tt
			every_owned_location = {
				limit = {
					swe_estonia_trigger = yes
					NOT = { this = location:tallinn }
					NOT = { this = location:narva }
					NOT = { has_location_modifier = swe_baltic_privileges }
				}
				add_location_modifier = { modifier = swe_baltic_privileges years = modifier_duration_years_very_long }
			}
		}
		# Reval and Narva were Hanseatic towns under Lübeck law: their councils, not the knighthood, kept the privileges
		every_owned_location = {
			limit = {
				OR = {
					this = location:tallinn
					this = location:narva
				}
				NOT = { has_location_modifier = swe_baltic_town_privileges }
			}
			add_location_modifier = { modifier = swe_baltic_town_privileges years = modifier_duration_years_very_long }
		}
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_weak_bonus }
	}

	option = {
		name = swe_baltic.1.b
		ai_chance = {
			base = 25
		}

		custom_tooltip = {
			text = swe_baltic.1.b_tt
			every_owned_location = {
				limit = {
					swe_estonia_trigger = yes
					NOT = { has_location_modifier = swe_swedish_law }
				}
				add_location_modifier = { modifier = swe_swedish_law years = 20 }
				every_pop = {
					limit = { pop_type = pop_type:peasants }
					add_pop_satisfaction = pop_satisfaction_weak_bonus
				}
			}
		}
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_penalty }
	}
}

# The Russian staple. Stolbovo (1617) cut Russia off from the Baltic, and Sweden tried to draw Russia's trade with the
# West through Narva, Nyen and Reval; Narva became the main one. Each of the three we hold gets swe_russian_transit and
# the chosen staple swe_russian_staple. Both last even if Russia returns to the Baltic, since a location modifier can't
# check that later.
swe_baltic.4 = {
	type = country_event
	title = swe_baltic.4.title
	desc = {
		# Each option needs its port, so the text names only the ports we hold
		first_valid = {
			triggered_desc = {
				trigger = {
					owns = location:narva
					owns = location:neva
					owns = location:tallinn
				}
				desc = swe_baltic.4.desc
			}
			triggered_desc = {
				trigger = {
					owns = location:narva
					owns = location:neva
				}
				desc = swe_baltic.4.desc_narva_neva
			}
			triggered_desc = {
				trigger = {
					owns = location:narva
					owns = location:tallinn
				}
				desc = swe_baltic.4.desc_narva_tallinn
			}
			triggered_desc = {
				trigger = {
					owns = location:neva
					owns = location:tallinn
				}
				desc = swe_baltic.4.desc_neva_tallinn
			}
			triggered_desc = {
				trigger = {
					owns = location:narva
				}
				desc = swe_baltic.4.desc_narva_only
			}
			triggered_desc = {
				trigger = {
					owns = location:neva
				}
				desc = swe_baltic.4.desc_neva_only
			}
			triggered_desc = {
				desc = swe_baltic.4.desc_tallinn_only
			}
		}
	}
	outcome = positive
	historical_info = swe_baltic.4.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 15
	}

	trigger = {
		has_or_had_tag = SWE
		current_age_or_later = {
			age = age_4_reformation
		}
		OR = {
			owns = location:narva

		if = {
			limit = {
				location:novgorod.owner ?= { NOT = { swe_is_swedish_realm_trigger = yes } }
			}
			location:novgorod.owner = { save_scope_as = swe_russian_hinterland }
		}
		else = {
			location:moscow.owner ?= { save_scope_as = swe_russian_hinterland }
		}
		# The same whichever town becomes the staple
		every_owned_location = {
			limit = {
				swe_russian_transit_port_trigger = yes
				NOT = { has_location_modifier = swe_russian_transit }
			}
			add_location_modifier = { modifier = swe_russian_transit years = modifier_duration_years_very_long }
		}

			owns = location:narva
		}

				change_prosperity = prosperity_weak_penalty
			}
		}
		location:tallinn = {
			if = {
				limit = { owner ?= root }

			}
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }

	option = {

# Silk from Isfahan. Karl XI sent Ludvig Fabritius to the Shah to draw the Persian silk trade through Russia to Narva,

# 1605. Can stack with swe_baltic.4's modifiers at Narva.

	title = swe_baltic.6.title
	desc = swe_baltic.6.desc

	trigger = {
		has_or_had_tag = SWE
		current_age_or_later = {
			age = age_5_absolutism

			NOT = { is_at_war_with = root }
		}
		location:moscow.owner ?= {
			NOT = { is_at_war_with = root }

					}
				}
			}
		}
	}

	option = {
		name = swe_baltic.6.b
		ai_chance = {
			base = 40
		}

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }

# Dominium maris Baltici, a milestone: our realm holds the mouths of the five great rivers into the Baltic and no Baltic
# power outside it has a bigger navy. The navy test is vanilla's flavor_dan.68, widened to the whole Baltic and with no
# navy tradition test. The Sound is left to vanilla's Sound toll at Helsingør. The phrase is first Swedish in the
# Dutch-Swedish treaty of 1614 (Dumont). Other sources: Roberts, The Swedish Imperial Experience; Küng; Gihl.
swe_baltic.7 = {
	type = country_event
	title = swe_baltic.7.title
	desc = swe_baltic.7.desc
	outcome = positive
	historical_info = swe_baltic.7.historical_info

	fire_only_once = yes
	major = yes
	major_trigger = {
		capital ?= {
			OR = {
				region = region:scandinavian_region
				region = region:baltic_region
				region = region:north_german_region
				region = region:russian_region
			}
		}
	}
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 10
	}

	trigger = {
		has_or_had_tag = SWE
		custom_tooltip = {
			text = swe_baltic.7.neva_tt
			location:neva.owner ?= { swe_is_swedish_realm_trigger = yes }
		}
		custom_tooltip = {
			text = swe_baltic.7.narva_tt
			location:narva.owner ?= { swe_is_swedish_realm_trigger = yes }
		}
		custom_tooltip = {
			text = swe_baltic.7.riga_tt
			location:riga.owner ?= { swe_is_swedish_realm_trigger = yes }
		}
		custom_tooltip = {
			text = swe_baltic.7.gdansk_tt
			location:gdansk.owner ?= { swe_is_swedish_realm_trigger = yes }
		}
		custom_tooltip = {
			text = swe_baltic.7.stettin_tt
			location:stettin.owner ?= { swe_is_swedish_realm_trigger = yes }
		}
		navy_size >= 50
		custom_tooltip = {
			text = swe_baltic.7.navy_tt
			area:inner_baltic_sea_area = {
				any_adjacent_ports_to_area = {
					count = all
					swe_baltic_port_no_rival_trigger = yes
				}
			}
			area:south_baltic_sea_area = {
				any_adjacent_ports_to_area = {
					count = all
					swe_baltic_port_no_rival_trigger = yes
				}
			}
			area:eastern_baltic_sea = {
				any_adjacent_ports_to_area = {
					count = all
					swe_baltic_port_no_rival_trigger = yes
				}
			}
		}
	}

	illustration_tags = {
		10 = happy
		10 = exterior
	}

	immediate = {
		ruler_or_regent ?= { save_scope_as = target_character }
		event_illustration_government_estate_effect = yes
	}

	# The sea closed to foreign warships, as under Roskilde's third article in 1658
	option = {
		name = swe_baltic.7.a
		ai_chance = {
			base = 40
		}

		add_prestige = prestige_severe_bonus
		add_navy_tradition = navy_tradition_extreme_bonus
		# Vanilla's own modifier from flavor_dan.68
		add_country_modifier = {
			modifier = dan_dominium_maris_baltici
			years = 10
			mode = add_and_extend
		}
		change_societal_value = { type = belligerent_vs_conciliatory value = societal_value_move_to_left }
		custom_tooltip = {
			text = swe_baltic.7.a_tt
			swe_list_baltic_powers = yes
			every_in_list = {
				list = swe_baltic_powers
				add_antagonism = {
					target = root
					modifier = swe_closed_baltic
				}
			}
		}
	}

	# The licences, as from Prussia to the Neva from the late 1620s
	option = {
		name = swe_baltic.7.b
		historical_option = yes
		ai_chance = {
			base = 40
		}

		add_prestige = prestige_mild_bonus
		# The first yield for the war-chest; the Licent below is the lasting revenue
		change_gold_effect = { scale = 4 }
		# The trade axis exists only from the Age of Reformation (1537), and this can fire earlier
		if = {
			limit = {

# The Baltic capitulations. As the Livonian Confederation broke up, its German knighthoods and towns
# submitted to Sweden on terms: Estonia and Reval in 1561, Riga in 1621, Livonia in the 1620s, Ösel in
# 1645. Each event fires whenever Sweden owns the chief town while Baltic Germans still live there, not
# by date. Confirming the privileges speeds integration and slows assimilation, as vanilla's
# lat_compromise does for the Greeks; Swedish law speeds assimilation through swe_forsvenskning.
# Nobody's culture changes at once. The variables mark the choice for later content.

# Estonia: the knighthoods of Harrien, Wierland and Jerwen and the town of Reval submitted to Erik XIV
# in 1561, and he confirmed their privileges. Ösel and Wiek (Rotalia) go with it: mainland Wiek was Swedish
# from the 1580s, and the island of Ösel was Danish from 1559 until Brömsebro (1645), when its knighthood kept
# the privileges it had held under Denmark.
swe_baltic.1 = {
	type = country_event
	title = swe_baltic.1.title
	desc = swe_baltic.1.desc
	outcome = neutral
	historical_info = swe_baltic.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 3 # Median about two years after Sweden takes Reval
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:tallinn
		location:tallinn = {
			any_pop = { culture = culture:german_baltic }
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:burghers_estate }
	}

	option = {
		name = swe_baltic.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		custom_tooltip = {
			text = swe_baltic.1.a_tt
			every_owned_location = {
				limit = {
					swe_estonia_trigger = yes
					NOT = { this = location:tallinn }
					NOT = { this = location:narva }
					NOT = { has_location_modifier = swe_baltic_privileges }
				}
				add_location_modifier = { modifier = swe_baltic_privileges years = modifier_duration_years_very_long }
			}
		}
		# Reval and Narva were Hanseatic towns under Lübeck law: their councils, not the knighthood, kept the privileges
		every_owned_location = {
			limit = {
				OR = {
					this = location:tallinn
					this = location:narva
				}
				NOT = { has_location_modifier = swe_baltic_town_privileges }
			}
			add_location_modifier = { modifier = swe_baltic_town_privileges years = modifier_duration_years_very_long }
		}
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_weak_bonus }
	}

	option = {
		name = swe_baltic.1.b
		ai_chance = {
			base = 25
		}

		custom_tooltip = {
			text = swe_baltic.1.b_tt
			every_owned_location = {
				limit = {
					swe_estonia_trigger = yes
					NOT = { dominant_culture = culture:swedish }
					NOT = { has_location_modifier = swe_forsvenskning }
				}
				add_location_modifier = { modifier = swe_forsvenskning years = 20 }
			}
		}
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_penalty }
	}
}

# The Russian staple. Stolbovo (1617) cut Russia off from the Baltic, and Sweden tried to draw Russia's
# trade with the West through Narva, Nyen and Reval; Narva became the main one. Fires from the age of
# the Reformation, while Sweden holds one of the three and whoever owns Novgorod has no Baltic port.
# Whichever of them Sweden holds then get swe_russian_transit, and the chosen staple also gets swe_russian_staple.
# Both run for 50 years even if Russia returns to the Baltic: a location modifier can't check that later.
# The prosperity of flavor_tre.5 (the trade port of Kerasous left to prosper). The ports passed over lose the weak step
# of it, since a staple takes their cargo. Holding only one of the three leaves the event free, as the reward for
# shutting Russia out.
swe_baltic.4 = {
	type = country_event
	title = swe_baltic.4.title
	desc = {
		# Each option needs its port, so the text names only the ports we hold
		first_valid = {
			triggered_desc = {
				trigger = {
					owns = location:narva
					owns = location:neva
					owns = location:tallinn
				}
				desc = swe_baltic.4.desc
			}
			triggered_desc = {
				trigger = {
					owns = location:narva
					owns = location:neva
				}
				desc = swe_baltic.4.desc_narva_neva
			}
			triggered_desc = {
				trigger = {
					owns = location:narva
					owns = location:tallinn
				}
				desc = swe_baltic.4.desc_narva_tallinn
			}
			triggered_desc = {
				trigger = {
					owns = location:neva
					owns = location:tallinn
				}
				desc = swe_baltic.4.desc_neva_tallinn
			}
			triggered_desc = {
				trigger = {
					owns = location:narva

		if = {
			limit = {
				location:novgorod.owner ?= { NOT = { swe_is_swedish_realm_trigger = yes } }
			}
			location:novgorod.owner = { save_scope_as = swe_russian_hinterland }

# Silk from Isfahan. Karl XI sent Ludvig Fabritius to the Shah three times (1679-1700) to draw the Persian silk trade
# through Russia to Narva. On 27 September 1687 the Crown contracted with the Armenian merchants of New Julfa: two years
# free of toll, a hostel and warehouse at Narva built with royal support, and transit dues of 2 and 1 per cent
# (Bakhchinyan, Hokkaido Slavic Research Center; the Armenian foreign ministry's history of Armenian-Swedish ties; en.wiki
# Ludvig Fabritius, citing Matthee). From the age of absolutism, since New Julfa dates from 1605, while Sweden owns
# Narva and is at peace with whoever owns Isfahan and Moscow. Can stack with swe_baltic.4's modifiers at Narva.
swe_baltic.6 = {
	type = country_event
	title = swe_baltic.6.title
	desc = swe_baltic.6.desc
	outcome = neutral
	historical_info = swe_baltic.6.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 3 # Median about two years after the conditions are met
	}

	trigger = {
		has_or_had_tag = SWE
		current_age_or_later = {
			age = age_5_absolutism
		}
		owns = location:narva
		location:isfahan.owner ?= {
			NOT = { is_at_war_with = root }
		}
		location:moscow.owner ?= {
			NOT = { is_at_war_with = root }
		}
	}

	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_silk.dds"

	option = {
		name = swe_baltic.6.a
		historical_option = yes
		ai_chance = {
			base = 60
		}

		# The hostel, and more for the two years free of toll, as the description says
		change_gold_effect = { scale = -2 }
		location:narva = {
			add_location_modifier = { modifier = swe_julfa_factory years = 30 }
			# Burghers don't stay in a rural settlement: a 2026-11-08 test at a rural Narva kept 0.005 of 0.03 after a
			# month. So the merchants only settle in a town, as swe_gotaland.7 requires for Alingsås.
			if = {
				limit = {
					OR = {
						location_rank = location_rank:town
						location_rank = location_rank:city
						location_rank = location_rank:megalopolis
					}
				}
				custom_tooltip = {
					text = swe_baltic.6.a_tt
					add_pop = {
						culture = culture:armenian_culture
						religion = religion:miaphysite
						type = pop_type:burghers
						size = 0.03
					}
				}
			}
		}
	}

	option = {
		name = swe_baltic.6.b
		ai_chance = {
			base = 40
		}

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
	}
}

# Dominium maris Baltici. A milestone for the player (AGENTS.md, "Events must actually fire"): it fires when our realm
# holds the mouths of the five great rivers into the Baltic and no Baltic power outside it has a bigger navy. The navy
# test is vanilla's flavor_dan.68 (navy_size >= 50, no port on the sea held by a bigger navy), widened from
# eastern_baltic_sea to the whole Baltic and without its navy tradition of 60. The Sound is left to vanilla's Sound
# toll, which goes to whoever holds Helsingør (default.map).
# The phrase is first Swedish in the Dutch-Swedish treaty of The Hague, 5 April 1614, art. VIII ("Dominio Maris
# Balthici", Dumont V, no. CXXXVII), renewed in 1640 (art. IX, Aitzema, book 20); both also promise free navigation.
# Gustav II Adolf in the riksråd, December 1628, on "den rättighet och tutel, som Sveriges konungar och krona af
# hedenhös hafver haft" (Ahnlund 1918, pp. 105-106); to Oxenstierna, 4 December 1629, "fienden Östersiöns herre, och
# fölljachteligen och wår öfwerman" (Styffe pp. 538-539). Licences in Prussia from 1627 and from Riga to the Neva from
# 1629 (Küng, Hansische Geschichtsblätter 133, 2015, pp. 148-152); "the whole yield of the licences was devoted to the
# payment of the armies in Germany" and they were "immediately indispensable to the war-chest" (Roberts, The Swedish
# Imperial Experience, 1979, pp. 35, 39). 814,000 riksdaler in 1634 (Wendt 1933, via Hvidegaard 2000) is single-source,
# so not in the text. Karl X Gustav to Dureel, 16 December 1656, on the Dutch treaty of 1640: "ett fundament och en
# rätt, den oss odisputerlig tillhörde" (Carlbom 1901, pp. 181-182). Roskilde 1658, art. 3, against "främmande
# fientliga flottor" (Gihl 1913, p. 29; Fridericia 1894, p. 424); the Battle of the Sound, 29 October 1658 (lex.dk);
# the Concert of The Hague, 11 May 1659, art. II, struck it out (Gihl pp. 147-149), and it lapsed at Copenhagen in 1660
# (Roberts p. 132). After 1648 Sweden held "the estuary of every major river from the Neva to the Weser, with the single
# exception of the Vistula" (Roberts p. 101). Notes: .research/milestones-2026-11-10/dominium.md.
swe_baltic.7 = {
	type = country_event
	title = swe_baltic.7.title
	desc = swe_baltic.7.desc
	outcome = positive
	historical_info = swe_baltic.7.historical_info

	fire_only_once = yes
	major = yes
	major_trigger = {
		capital ?= {
			OR = {
				region = region:scandinavian_region
				region = region:baltic_region
				region = region:north_german_region
				region = region:russian_region
			}
		}
	}
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 10 # Median about seven months after the conditions are met
	}

	trigger = {
		has_or_had_tag = SWE
		custom_tooltip = {
			text = swe_baltic.7.neva_tt
			location:neva.owner ?= { swe_is_swedish_realm_trigger = yes }
		}
		custom_tooltip = {
			text = swe_baltic.7.narva_tt
			location:narva.owner ?= { swe_is_swedish_realm_trigger = yes }
		}
		custom_tooltip = {
			text = swe_baltic.7.riga_tt
			location:riga.owner ?= { swe_is_swedish_realm_trigger = yes }
		}
		custom_tooltip = {
			text = swe_baltic.7.gdansk_tt

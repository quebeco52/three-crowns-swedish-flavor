namespace = swe_scania

# Försvenskningen: after Roskilde (1658) Sweden first promised Skåneland its own law, then from
# the 1680s brought in Swedish law, church books and priests. Lund University (1666) trained them.
# Fires whenever Sweden holds all of Skåneland while Danish culture remains there, not from 1658:
# in 1337 Skåneland is already a Swedish fiefdom. It speeds up assimilation rather than
# changing anyone's culture.
# Balanced against flavor_dan.40 (Denmark's reconquest of Scania, Denmark only, so both never fire).
swe_scania.1 = {
	type = country_event
	title = swe_scania.1.title
	desc = swe_scania.1.desc
	outcome = neutral
	historical_info = swe_scania.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 3 # Median about two years after Sweden holds all of Skåneland
	}

	trigger = {
		has_or_had_tag = SWE
		own_entire_area = area:skaneland_area
		area:skaneland_area = {
			any_location_in_area = { dominant_culture = culture:danish }
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:peasants_estate }
	}

	option = {
		name = swe_scania.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		custom_tooltip = {
			text = swe_scania.1.a_tt
			area:skaneland_area = {
				every_location_in_area = {
					limit = { dominant_culture = culture:danish }
					add_location_modifier = { modifier = swe_forsvenskning years = 20 }
				}
			}
		}
		# Lund, where the new university trained the Swedish priests. Development rather than a
		# university building, which vanilla can block (clergy_lecture_oaths) or the date may not allow
		location:lund = {
			change_development = development_mild_bonus
			change_prosperity = prosperity_mild_bonus
		}
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_penalty }
	}

	option = {
		name = swe_scania.1.b
		ai_chance = {
			base = 25
		}

		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_weak_bonus }
	}
}

# The Malmö conspiracy of 1658-59 (Rasmussen, Scandia 70:2, and Malmö stad). Executions per Scandia: Mikkelsen,
# Jørgensen and Brun, 16 December 1659. Fires on Sweden's own rule, not a subject's, whenever Malmö is still mainly
# Danish, as with swe_scania.1, so the text doesn't name a date.
swe_scania.6 = {
	type = country_event
	title = swe_scania.6.title
	desc = swe_scania.6.desc
	outcome = neutral
	historical_info = swe_scania.6.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 3 # Median about two years after Sweden itself owns Malmö
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:malmo
		location:malmo = { dominant_culture = culture:danish }
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
	}

	option = {
		name = swe_scania.6.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = 2 }
		location:malmo = {
			add_location_modifier = { modifier = swe_malmo_plot_crushed years = 25 }
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_penalty }
	}

	option = {
		name = swe_scania.6.b
		ai_chance = {
			base = 25
		}

		location:malmo = {
			add_location_modifier = { modifier = swe_malmo_mercy years = 25 }
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
	}
}

namespace = swe_scania

# Försvenskningen. After Roskilde (1658) Sweden first promised Skåneland its own law, then from the 1680s brought in
# Swedish law, church books and priests trained at Lund University (1666). Fires whenever Sweden holds all of
# Skåneland while Danish culture remains, not from 1658, since in 1337 Skåneland is already a Swedish fiefdom.
swe_scania.1 = {
	type = country_event
	title = swe_scania.1.title
	desc = swe_scania.1.desc
	outcome = neutral
	historical_info = swe_scania.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 15
	}

	trigger = {
		has_or_had_tag = SWE
		own_entire_area = area:skaneland_area
		area:skaneland_area = {
			any_location_in_area = { dominant_culture ?= culture:danish }
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:peasants_estate }
	}

	option = {
		name = swe_scania.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		custom_tooltip = {
			text = swe_scania.1.a_tt
			area:skaneland_area = {
				every_location_in_area = {
					limit = { dominant_culture = culture:danish }
					add_location_modifier = { modifier = swe_forsvenskning years = 20 }
				}
			}
		}
		# Lund, where the university trained the priests. Development rather than a university building, which
		# clergy_lecture_oaths or the date can block
		location:lund = {
			change_development = development_mild_bonus
			change_prosperity = prosperity_mild_bonus
		}
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_penalty }
	}

	option = {
		name = swe_scania.1.b
		ai_chance = {
			base = 25
		}

		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_weak_bonus }
	}
}

# The Malmö conspiracy of 1658-59 (Rasmussen, Scandia 70:2; Malmö stad). Fires whenever Malmö is ours and still
# mainly Danish, so the text names no date.
swe_scania.6 = {
	type = country_event
	title = swe_scania.6.title
	desc = swe_scania.6.desc
	outcome = neutral
	historical_info = swe_scania.6.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 15
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:malmo
		location:malmo = { dominant_culture = culture:danish }
	}

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
	}

	option = {
		name = swe_scania.6.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = 2 }
		# The fines come out of the burghers' purse, as in flavor_ara.110.b
		add_gold_to_estate = {
			estate_type = estate_type:burghers_estate
			value = { value = root.monthly_income_trade_and_tax multiply = -2 }
		}
		location:malmo = {
			add_location_modifier = { modifier = swe_malmo_plot_crushed years = 10 }
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_penalty }
	}

	option = {
		name = swe_scania.6.b
		ai_chance = {
			base = 25
		}

		location:malmo = {
			add_location_modifier = { modifier = swe_malmo_mercy years = 10 }
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
	}
}

# Karlskrona's privileges, 10 August 1680 (Sjöhistoriska museet, rapport 2018:1). Vanilla's Swedish dialect already
# names bregne Karlskrona; the description avoids the name because the town doesn't exist yet. Laborers move because a
# rural Ronneby has almost no burghers.
swe_scania.7 = {
	type = country_event
	title = swe_scania.7.title
	desc = swe_scania.7.desc
	outcome = neutral
	historical_info = swe_scania.7.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1678.1.1
		to = 1720.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_scania_7
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:bregne
		owns = location:ronneby
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
		ruler_or_regent ?= { save_scope_as = target_character }
	}

	option = {
		name = swe_scania.7.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -6 }
		location:bregne = {
			add_location_modifier = { modifier = swe_karlskrona_station years = 50 }
		}
		location:ronneby = {
			change_prosperity = prosperity_mild_penalty
			add_location_modifier = { modifier = swe_ronneby_privileges_revoked years = 10 }
		}
		add_migration = {
			owner = this
			from_location = location:ronneby
			to_location = location:bregne
			type = pop_type:laborers
			amount = 0.05
			months = 24
		}
		# All of Ronneby's pops, since a rural location holds almost no burghers
		custom_tooltip = {
			text = swe_scania.7.a_tt
			swe_all_pops_satisfaction_in = { location = location:ronneby value = pop_satisfaction_mild_penalty }
		}
	}

	option = {
		name = swe_scania.7.b
		ai_chance = {
			base = 25
		}

		custom_tooltip = {
			text = swe_scania.7.b_tt
			swe_all_pops_satisfaction_in = { location = location:ronneby value = pop_satisfaction_weak_bonus }
		}
	}
}

# The Scanian homage. Magnus Eriksson took Skåne in pawn from the counts of Holstein in 1332 (SBL, Kumlien). In 1339
# he pledged the revenues of Skanör and Falsterbo to Albrekt of Mecklenburg, and at Helsingborg on 28 June 1340 the
# Scanians swore fealty and had their privileges confirmed (SBL; Kroman). Option c rests on the letter's "prefectus
# noster", since Bengt Algotsson as governor is disputed (Nordberg 1995). Fires while the fiefdom SKE is ours.
swe_scania.8 = {
	type = country_event
	title = swe_scania.8.title
	desc = swe_scania.8.desc
	outcome = neutral
	historical_info = swe_scania.8.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1338.1.1
		to = 1360.12.31
		# High, to fire before flavor_dan_skane_war.1, whose chain can end the fiefdom
		monthly_chance = 5
	}

	trigger = {
		has_or_had_tag = SWE
		c:SKE ?= { is_subject_of = root }
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:nobles_estate }
		ruler_or_regent ?= { save_scope_as = target_character }
		c:SKE = { save_scope_as = swe_scania }
	}

	# Confirm the liberties they held under the Danish kings, as at Helsingborg in 1340
	option = {
		name = swe_scania.8.a
		historical_option = yes
		ai_chance = {
			base = 60
		}

		scope:swe_scania = {
			add_opinion = { target = root modifier = opinion_granted_country_privileges }
			add_country_modifier = { modifier = swe_scanian_privileges years = 20 }
		}
	}

	# Pledge the dues of Skanör and Falsterbo, as to Albrekt of Mecklenburg in 1339
	option = {
		name = swe_scania.8.b
		ai_chance = {
			base = 25
		}

		# Guarded because the Historical Events panel previews options without running immediate
		if = {
			limit = { exists = scope:swe_scania }
			add_gold = {
				value = scope:swe_scania.monthly_income_trade_and_tax
				multiply = 6
			}
		}
		scope:swe_scania = {
			add_country_modifier = { modifier = swe_scanian_pledge years = 10 }
		}
		c:MKL ?= {
			if = {
				limit = { country_exists = this }
				add_opinion = { target = root modifier = opinion_pleased }
			}
		}
	}

	# Leave the charters unconfirmed and let our governor rule
	option = {
		name = swe_scania.8.c
		ai_chance = {
			base = 15
		}

		scope:swe_scania = {
			add_country_modifier = { modifier = swe_scanian_governor years = 10 }
			add_liberty_desire = liberty_desire_mild_plus
			add_opinion = { target = root modifier = swe_scanian_governor_resented }
		}
	}
}

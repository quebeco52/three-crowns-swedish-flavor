namespace = swe_lappmark

# Content for Västerbotten, the Lappmarks and the Sámi. Sweden starts here with only Umeå, Luleå and the colony
# at Åsele, and SMI and KVE are pop countries that own no land. So .1 to .5 only need Luleå or Umeå and should
# fire in nearly every game. .6 and .7 wait for Sweden to own the location.

# Archbishop Hemming sails to Torneå in the mid-1340s. An inquest of 1374 (SDHK 10584) records that he met the
# Bishop of Åbo at Kompanasund and baptised about 20 Sámi and Karelians in a vat carried into the chapel.
# Umeå, Luleå and Åsele have Sámi and Kven pops of the old religions, so the conversion counts from the start.
swe_lappmark.1 = {
	type = country_event
	title = swe_lappmark.1.title
	desc = swe_lappmark.1.desc
	outcome = neutral
	historical_info = swe_lappmark.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1344.1.1
		to = 1365.1.1
		monthly_chance = 3 # Median late 1345
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:lulea
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
	}

	option = {
		name = swe_lappmark.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -2 }
		custom_tooltip = {
			text = swe_lappmark.1.a_tt
			every_owned_location = {
				limit = {
					swe_far_north_trigger = yes
					any_pop = { NOT = { religion = root.religion } }
				}
				add_location_modifier = { modifier = swe_uppsala_mission years = 20 }
				# The inquest of 1374 records about 20 baptised, so a small share at once, and the modifier does the rest
				every_pop = {
					limit = { NOT = { religion = root.religion } }
					split_pop = {
						fraction = 0.05
						religion = root.religion
					}
				}
			}
		}
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }
	}

	option = {
		name = swe_lappmark.1.b
		ai_chance = {
			base = 25
		}

		location:abo = {
			if = {
				limit = { owner ?= root }
				add_location_modifier = { modifier = swe_abo_diocese_north years = 25 }
			}
		}
	}
}

# In the 14th century coastal farmers paid the Crown for sole fishing rights at Gråträsk, and settlers followed.
# The migration pull is on gratesk, which starts with no owner, so it's skipped if a foreign realm holds it.

	type = country_event
	title = swe_lappmark.3.title
	desc = swe_lappmark.3.desc
	outcome = neutral
	historical_info = swe_lappmark.3.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1340.1.1
		to = 1400.1.1
		monthly_chance = 2 # About four years
	}

	trigger = {
		has_or_had_tag = SWE
		OR = {
			owns = location:lulea
			owns = location:umea
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
	}

	option = {
		name = swe_lappmark.3.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = 2 }
		location:gratesk = {
			if = {
				limit = {
					OR = {
						NOT = { exists = owner }
						owner ?= root
					}
				}
				add_location_modifier = { modifier = swe_gratrask_fishing_rights years = 50 }
			}
		}
	}

	option = {
		name = swe_lappmark.3.b
		ai_chance = {
			base = 25
		}

		location:lulea = {
			if = {
				limit = { owner ?= root }
				add_location_modifier = { modifier = swe_sami_fishing_waters years = 25 }
			}
		}
		location:umea = {
			if = {
				limit = { owner ?= root }
				add_location_modifier = { modifier = swe_sami_fishing_waters years = 25 }
			}
		}
	}
}

# The mountain Sámi turn from hunting to reindeer herding. By the 1600s and 1700s many had switched entirely,
# and a register of 1602 already counted each taxpayer's reindeer. Vanilla has no reindeer good, so livestock
# is the closest (swe_reindeer_herds). The change happens whichever option is chosen, so both options show it.
# The tax only touches locations with Sámi pops, so Swedish parishes don't get its unrest.
swe_lappmark.4 = {
	type = country_event
	title = swe_lappmark.4.title
	desc = swe_lappmark.4.desc
	outcome = neutral
	historical_info = swe_lappmark.4.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1600.1.1
		to = 1700.1.1
		monthly_chance = 3 # Median late 1601
	}

	trigger = {
		has_or_had_tag = SWE
		OR = {
			owns = location:lulea
			owns = location:umea
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
	}

	option = {
		name = swe_lappmark.4.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		custom_tooltip = {
			text = swe_lappmark.4.herds_tt
			swe_reindeer_herds = yes
		}
		change_gold_effect = { scale = 2 }
		custom_tooltip = {
			text = swe_lappmark.4.a_tt
			every_owned_location = {
				limit = {
					swe_far_north_trigger = yes
					any_pop = { culture = culture:sapmi }
				}
				add_location_modifier = { modifier = swe_reindeer_tax years = 20 }
			}
		}
	}

	option = {
		name = swe_lappmark.4.b
		ai_chance = {
			base = 25
		}

		custom_tooltip = {
			text = swe_lappmark.4.herds_tt
			swe_reindeer_herds = yes
		}
	}
}

# The silver road: Sámi were forced to haul ore and supplies from Nasafjäll to the coast. Vanilla's flavor_swe.39
# only runs 1630-35 and needs Sweden to own Silbojock by then, so this event opens the mine if that one never
# did (swe_open_silbojock), and .c lets the player refuse it again. It waits for Sweden to own the location, so it fires at whatever
# date the land is settled.
swe_lappmark.6 = {
	type = country_event
	title = swe_lappmark.6.title
	desc = swe_lappmark.6.desc
	outcome = neutral
	historical_info = swe_lappmark.6.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1635.1.1
		to = 1750.1.1
		monthly_chance = 3 # Median about two years once Sweden owns Silbojock
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:silbojock
	}

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:burghers_estate }
	}

	option = {
		name = swe_lappmark.6.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		swe_open_silbojock = yes
		custom_tooltip = {
			text = swe_lappmark.6.a_tt
			every_owned_location = {
				limit = { province_definition = province_definition:pite_lappmark_province }
				add_location_modifier = { modifier = swe_silver_road years = 20 }
			}
		}
		# The decree of 1640 moved the Lais and Gran lappbyar from Ume Lappmark and Tuorpon from Lule Lappmark to provide
		# more carriers. They went to Arjeplog, which may not be ours, so they move to the mine.
		if = {
			limit = {
				any_owned_location = { province_definition = province_definition:ume_lappmark_province }
			}
			add_migration = {
				owner = this
				from = province_definition:ume_lappmark_province
				to_location = location:silbojock
				culture = culture:sapmi
				amount = 0.05
				months = 24
			}
		}
		if = {
			limit = {
				any_owned_location = { province_definition = province_definition:lule_lappmark_province }
			}
			add_migration = {
				owner = this
				from = province_definition:lule_lappmark_province
				to_location = location:silbojock
				culture = culture:sapmi
				amount = 0.05
				months = 24
			}
		}
	}

	option = {
		name = swe_lappmark.6.b
		ai_chance = {
			base = 25
		}

		change_gold_effect = { scale = -4 }
		swe_open_silbojock = yes
	}

	# The same refusal as flavor_swe.39.b, hidden once the mine is open
	option = {
		name = swe_lappmark.6.c
		trigger = {
			location:silbojock = {
				NOT = { raw_material = goods:silver }
			}
		}
		ai_chance = {
			base = 10
		}

		add_stability = stability_mild_bonus
	}
}

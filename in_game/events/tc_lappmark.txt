# Västerbotten, the Lappmarks and the Sámi. Sweden starts here with only Umeå, Luleå and the colony at Åsele, and SMI
# and KVE are pop countries that own no land. There is no .2.

# Archbishop Hemming sails to Torneå in the mid-1340s. An inquest of 1374 (SDHK 10584) records that he baptised about
# 20 Sámi and Karelians. Umeå, Luleå and Åsele have Sámi and Kven pops of the old religions, so the conversion counts
# from the start.
swe_lappmark.1 = {
	type = country_event
	title = swe_lappmark.1.title
	desc = swe_lappmark.1.desc
	outcome = neutral
	historical_info = swe_lappmark.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1344.1.1
		to = 1365.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_lappmark_1
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:lulea
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
	}

	option = {
		name = swe_lappmark.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -2 }
		custom_tooltip = {
			text = swe_lappmark.1.a_tt
			every_owned_location = {
				limit = {
					swe_far_north_trigger = yes
					any_pop = { NOT = { religion = root.religion } }
				}
				add_location_modifier = { modifier = swe_uppsala_mission years = 20 }
				# About 20 were baptised, so a small share at once, and the modifier does the rest
				every_pop = {
					limit = { NOT = { religion = root.religion } }
					split_pop = {
						fraction = 0.05
						religion = root.religion
					}
				}
			}
		}
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }
	}

	option = {
		name = swe_lappmark.1.b
		ai_chance = {
			base = 25
		}

		location:abo = {
			if = {
				limit = { owner ?= root }
				add_location_modifier = { modifier = swe_abo_diocese_north years = 25 }
			}
		}
	}
}

# In the 14th century coastal farmers paid the Crown for sole fishing rights at Gråträsk, and settlers followed. The
# migration pull goes on gratesk, which starts with no owner and is skipped if a foreign realm holds it. gratesk is

	type = country_event
	title = swe_lappmark.3.title

	outcome = neutral
	historical_info = swe_lappmark.3.historical_info

		tag = SWE
		tag = SCA
		from = 1340.1.1
		to = 1400.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_lappmark_3
	}

	trigger = {
		has_or_had_tag = SWE
		OR = {
			owns = location:lulea
			owns = location:umea
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
	}

	option = {
		name = swe_lappmark.3.a

			base = 75
		}

		change_gold_effect = { scale = 2 }
		location:gratesk = {

				every_pop = {

			}
		}
	}

		name = swe_lappmark.3.b
		ai_chance = {
			base = 25
		}

		location:lulea = {
			if = {
				limit = { owner ?= root }
				add_location_modifier = { modifier = swe_sami_fishing_waters years = 15 }
			}
		}
		location:umea = {
			if = {
				limit = { owner ?= root }

# The silver road: Sámi were forced to haul ore and supplies from Nasafjäll to the coast. Vanilla's flavor_swe.39 only
# runs 1630-35 and needs Sweden to own Silbojock by then, so this event opens the mine if that one never did, and
# option c refuses it again.
swe_lappmark.6 = {
	type = country_event
	title = swe_lappmark.6.title
	desc = {
		first_valid = {
			# Vanilla's flavor_swe.39.a may have opened the mine already, which hides option c
			triggered_desc = {
				trigger = {
					location:silbojock = { raw_material = goods:silver }
				}
				desc = swe_lappmark.6.desc_open
			}
			triggered_desc = {
				desc = swe_lappmark.6.desc
			}
		}
	}
	outcome = neutral
	historical_info = swe_lappmark.6.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1635.1.1
		to = 1750.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_lappmark_6
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:silbojock
	}

	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_silver.dds"

	option = {
		name = swe_lappmark.6.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		swe_open_silbojock = yes
		custom_tooltip = {
			text = swe_lappmark.6.a_tt
			every_owned_location = {
				limit = { province_definition = province_definition:pite_lappmark_province }
				add_location_modifier = { modifier = swe_silver_road years = 20 }
			}
		}
		# The decree of 1640 moved lappbyar from Ume and Lule Lappmark to Arjeplog as carriers. Arjeplog may not be
		# ours, so they move to the mine. Small amounts, since these lappmarks usually hold only a pop or so of Sámi
		# by then.
		if = {
			limit = {
				any_owned_location = { province_definition = province_definition:ume_lappmark_province }
			}
			add_migration = {
				owner = this
				from = province_definition:ume_lappmark_province
				to_location = location:silbojock
				culture = culture:sapmi
				amount = 0.01
				months = 24
			}
		}
		if = {
			limit = {
				any_owned_location = { province_definition = province_definition:lule_lappmark_province }
			}
			add_migration = {
				owner = this
				from = province_definition:lule_lappmark_province
				to_location = location:silbojock
				culture = culture:sapmi
				amount = 0.01
				months = 24
			}
		}
	}

	option = {
		name = swe_lappmark.6.b
		ai_chance = {
			base = 25
		}

		change_gold_effect = { scale = -4 }
		swe_open_silbojock = yes
	}

	# The same refusal as flavor_swe.39.b, hidden once the mine is open
	option = {
		name = swe_lappmark.6.c
		trigger = {
			location:silbojock = {
				NOT = { raw_material = goods:silver }
			}
		}
		ai_chance = {
			base = 10
		}

		add_stability = stability_mild_bonus
	}
}

# The land rise moves Luleå (1649, Länsstyrelsen Norrbotten) and Piteå (by 1668, after the fire of 1666, sv.wiki);
# Piteå Museum names land rise as the cause of both. One event for the two, around Luleå's date, with the modifiers of
# swe_finland_west.7. It needs both towns, as the text names both. Each moves inside its location, so no rename.
swe_lappmark.8 = {
	type = country_event
	title = swe_lappmark.8.title
	desc = swe_lappmark.8.desc
	outcome = neutral
	historical_info = swe_lappmark.8.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1647.1.1
		to = 1700.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_lappmark_8
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:lulea
		owns = location:pitea
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
	}

	option = {
		name = swe_lappmark.8.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		change_gold_effect = { scale = -4 }
		location:lulea = {
			change_prosperity = prosperity_mild_penalty
			add_location_modifier = { modifier = swe_town_moved years = 20 }
		}
		location:pitea = {
			change_prosperity = prosperity_mild_penalty
			add_location_modifier = { modifier = swe_town_moved years = 20 }
		}
		# The townsfolk ordered out, of whatever kind, since a rural location holds almost no burghers
		custom_tooltip = {
			text = swe_lappmark.8.a_tt
			swe_all_pops_satisfaction_in = { location = location:lulea value = pop_satisfaction_weak_penalty }
			swe_all_pops_satisfaction_in = { location = location:pitea value = pop_satisfaction_weak_penalty }
		}
	}

	option = {
		name = swe_lappmark.8.b
		ai_chance = {
			base = 25
		}

		location:lulea = {
			add_location_modifier = { modifier = swe_silted_harbour years = 20 }
		}
		location:pitea = {
			add_location_modifier = { modifier = swe_silted_harbour years = 20 }
		}
		custom_tooltip = {
			text = swe_lappmark.8.b_tt

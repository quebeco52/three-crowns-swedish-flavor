namespace = swe_oland

# The peasants of Öland complain about the royal deer park (swe_olands_djurgard), as their deputies did
# at the Riksdag of 1800. Waits until the park is 20 years old, then needs unhappy peasants or a realm
# leaning toward free subjects. Refusing is final: the park then stays for good.
swe_oland.1 = {
	type = country_event
	title = swe_oland.1.title
	desc = swe_oland.1.desc
	outcome = neutral
	historical_info = swe_oland.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 2 # About four years once the trigger holds
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:oland
		location:oland = { has_location_modifier = swe_olands_djurgard }
		NOT = { has_variable = swe_olands_djurgard_new }
		OR = {
			estate_satisfaction:peasants_estate < 0.5
			societal_value:serfdom_vs_free_subjects > 25
		}
	}

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:nobles_estate }
	}

	option = {
		name = swe_oland.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		location:oland = {
			remove_location_modifier = swe_olands_djurgard
		}
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_bonus }
	}

	option = {
		name = swe_oland.1.b
		ai_chance = {
			base = 25
		}

		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_penalty }
		custom_tooltip = swe_oland.1.b_tt
	}
}

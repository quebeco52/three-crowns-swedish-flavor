# The castle at Faxeholm burns. The Vitalian Brothers built it in 1395/96, the Crown held it by 1398, and the
# Hälsingar stormed and burned it in 1434.
swe_norrland.1 = {
	type = country_event
	title = swe_norrland.1.title
	desc = swe_norrland.1.desc
	outcome = neutral
	historical_info = swe_norrland.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1432.1.1
		to = 1500.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_norrland_1
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:soderala
	}

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:nobles_estate }
	}

	option = {
		name = swe_norrland.1.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		custom_tooltip = {
			text = swe_norrland.1.a_tt
			every_owned_location = {
				limit = {
					OR = {
						province_definition = province_definition:halsingland_province
						province_definition = province_definition:angermanland_province
					}
				}
				add_location_modifier = { modifier = swe_faxeholm_ruins years = 20 }
			}
		}
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_weak_bonus }
		add_prestige = prestige_mild_penalty
	}

	option = {
		name = swe_norrland.1.b
		ai_chance = {
			base = 25
		}

		change_gold_effect = { scale = -4 }
		custom_tooltip = {
			text = swe_norrland.1.b_tt
			every_owned_location = {
				limit = {
					OR = {
						province_definition = province_definition:halsingland_province
						province_definition = province_definition:angermanland_province

# Gustav Vasa gives the fishermen of Gefle the Norrland coast, 1557, for every tenth barrel of salted herring. The

		tag = SCA

		to = 1620.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_norrland_2
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:gefle
		OR = {
			owns = location:hudiksvall
			owns = location:harnosand
		}
	}

	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_fish.dds"

	option = {

# Döda fallet, June 1796. The Indalsälven broke through a timber channel and Ragundasjön emptied in about four hours.
# Both options turn the lakebed into millet, vanilla's grain for most Swedish farmland. Ragunda starts Norwegian, so
# this needs Jämtland taken. Option b pays for the flood, so Selånger keeps its prosperity.
swe_norrland.3 = {
	type = country_event
	title = swe_norrland.3.title
	desc = swe_norrland.3.desc
	outcome = neutral
	historical_info = swe_norrland.3.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1794.1.1
		to = 1837.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_norrland_3
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:ragunda
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
	}

	option = {
		name = swe_norrland.3.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		location:ragunda = {
			change_raw_material = goods:millet
			# The old lakebed is new farmland, whoever ends up holding it
			if = {
				limit = { is_full_expanded_rgo = no }
				change_max_raw_material_workers = 2
			}
		}
		# The Indalsälven reaches the sea here, and the flood destroyed the sawmills and fisheries on its way
		location:selanger = {
			if = {
				limit = { owner ?= root }
				change_prosperity = prosperity_severe_penalty
			}
		}
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_bonus }
	}

	option = {
		name = swe_norrland.3.b
		ai_chance = {
			base = 25
		}

		change_gold_effect = { scale = -2 }
		location:ragunda = {
			change_raw_material = goods:millet
			# The old lakebed is new farmland, whoever ends up holding it
			if = {
				limit = { is_full_expanded_rgo = no }
				change_max_raw_material_workers = 2
			}
			add_location_modifier = { modifier = swe_ragunda_crown_land years = 50 }
		}
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_penalty }
	}
}

# Hälsingland flax. Olof Broman, rector and later vicar in Hudiksvall, is credited with water-powered scutching
# (Växbo Lin, the only source), and about 1,000 flax barns and mills followed. Flor, in Mo outside Söderhamn, made
# damask for Europe's courts; Söderala stands in for it. Vanilla has no fiber crops here; raw material changes
# follow flavor_swe.38.
swe_norrland.6 = {
	type = country_event
	title = swe_norrland.6.title
	desc = swe_norrland.6.desc
	outcome = neutral
	historical_info = swe_norrland.6.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1698.1.1
		to = 1780.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_norrland_6
	}

	trigger = {
		has_or_had_tag = SWE
		owns = location:hudiksvall
		owns = location:alfta
		owns = location:soderala
	}

	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_fiber_crops.dds"

	option = {
		name = swe_norrland.6.a
		historical_option = yes
		ai_chance = {
			base = 75
		}

		location:alfta = {
			change_raw_material = goods:fiber_crops
		}
		location:hogdal = {
			if = {
				limit = { owner ?= root }
				change_raw_material = goods:fiber_crops
			}
		}
		add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_bonus }
	}

	option = {
		name = swe_norrland.6.b
		ai_chance = {
			base = 25
		}

		location:soderala = {
			change_raw_material = goods:fiber_crops
			add_location_modifier = { modifier = swe_flor_manufactory months = -1 }
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
	}
}

# Carved saints. Haaken Gulleson, workshop leader in Enånger, worked across southern Norrland in the early 16th
# century. His signed works include the Anna själv tredje in Enånger's old church, dated to 1520, and a Madonna at
# Forsa, in hudiksvall (Uppsala University; Hälsinglands museum; en.wiki). The parishes paid for their church art, so
# the free option is historical. Catholic only; the window opens before the usual Reformation.
swe_norrland.7 = {
	type = country_event
	title = swe_norrland.7.title
	desc = swe_norrland.7.desc
	outcome = neutral
	historical_info = swe_norrland.7.historical_info

	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1514.1.1
		to = 1530.1.1
		monthly_chance = 1 # Mostly fired by tc_pulse_norrland_7
	}

	trigger = {
		has_or_had_tag = SWE
		religion = religion:catholic
		owns = location:hudiksvall
	}

	illustration_tags = {
		10 = happy
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:peasants_estate }
		ruler_or_regent ?= { save_scope_as = target_character }
	}

	option = {
		name = swe_norrland.7.a

			base = 75

			base = 25

		change_gold_effect = { scale = 2 }

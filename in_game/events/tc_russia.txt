# Content for the Russian lands beyond Ingria and Kexholm. Sweden held Novgorod in 1611-17 and never Moscow, so .1 is a
# milestone for the player (AGENTS.md, "Events must actually fire"): it fires whenever Sweden or a Swedish subject
# holds each city. .2 fires on whoever holds Novgorod and borders us, so AI games see it; .3 needs Archangel, which only
# a player takes.

# Novgorod and Moscow. Under the Treaty of Vyborg (1609) Vasily Shuisky hired a mercenary corps from Karl IX, paid by
# Russia, for Kexholm, and De la Gardie entered Moscow with Skopin-Shuisky in March 1610; the mercenaries went over
# to the Poles at Klushino on 24 June 1610 (SBL, Jakob De la Gardie). De la Gardie stormed Novgorod on 16 July 1611
# and held it to 1617; Widekindi (1671) has him promise to keep its "ancient religion of the Greek rite" inviolate, St
# Sophia kept its services, and he demanded 6,000 roubles of Metropolitan Isidor in 1616 (A. A. Selin, Novgorod 1611-17,
# from RGADA and Riksarkivet). He came home with sables and church bells, which he said he had bought from soldiers who
# took them "med krigets rätt", and left the doors thought to be Sigtuna's "out of regard for opinion in Novgorod"
# (SBL); the Novgorod archives are in Riksarkivet. Gustav II Adolf promised Munich safety and its religion for 300,000
# Reichstaler in 1632; it raised about a third and gave 42 hostages (Altmann 2018). Prague's castle treasures went to
# Stockholm in 1648 (Uppsala University Library, Codex Argenteus). Karl XII meant, in Piper's words, to make the peace
# "i själva Moskva" (Åberg, Svensk Tidskrift 1959, after Haintz) and turned south at Tatarsk in September 1708 (SBL,
# Karl XII). The Kremlin treasury really was looted in 1610-12, but by the Polish-Lithuanian garrison (Kozlyakov), so
# option a rests on Munich, Prague and Novgorod, not on that. Notes: .research/milestones-2026-11-10/russia.md.
swe_russia.1 = {
	type = country_event
	title = swe_russia.1.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:swe_moscow_holder }
				desc = swe_russia.1.desc_subject
			}
			triggered_desc = {
				desc = swe_russia.1.desc
			}
		}
	}
	outcome = positive
	historical_info = swe_russia.1.historical_info

	fire_only_once = yes
	major = yes
	major_trigger = {
		capital ?= {
			OR = {
				region = region:russian_region
				region = region:ruthenia_region
				region = region:baltic_region
				region = region:scandinavian_region
			}
		}
	}
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 10 # Median about seven months after our realm holds both cities
	}

	trigger = {
		has_or_had_tag = SWE
		custom_tooltip = {
			text = swe_russia.1.novgorod_tt
			location:novgorod.owner ?= { swe_is_swedish_realm_trigger = yes }
		}
		custom_tooltip = {
			text = swe_russia.1.moscow_tt
			location:moscow.owner ?= { swe_is_swedish_realm_trigger = yes }
		}
	}

	illustration_tags = {
		10 = armed
		10 = exterior
	}

	immediate = {
		ruler_or_regent ?= { save_scope_as = target_character }
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:clergy_estate }
		# A subject holding either city, saved once if it holds both
		location:moscow.owner ?= {
			if = {
				limit = { NOT = { this = root } }
				save_scope_as = swe_moscow_holder
			}
		}
		location:novgorod.owner ?= {
			if = {
				limit = {
					NOT = { this = root }
					NOT = { owns = location:moscow }
				}
				save_scope_as = swe_novgorod_holder
			}
		}
		# The gold of options a and b follows Moscow's own wealth, as vanilla's conquest_mission_events.4 does
		location:moscow = {
			province = { save_scope_as = swe_moscow_province }
		}
	}

	# The treasury to Stockholm, as from Prague in 1648. Vanilla's looting of a conquered capital
	# (conquest_mission_events.4.a): four times the province's tax base, and mild prosperity loss across it.
	option = {
		name = swe_russia.1.a
		ai_chance = {
			base = 40
		}

		add_prestige = prestige_extreme_bonus
		add_gold = {
			value = scope:swe_moscow_province.province_tax_base
			multiply = 4
		}
		custom_tooltip = {
			text = swe_russia.1.a_prosperity_tt
			scope:swe_moscow_province = {
				every_location_in_province = {
					limit = {
						owner ?= { swe_is_swedish_realm_trigger = yes }
					}
					change_prosperity = prosperity_mild_penalty
				}
			}
		}
		scope:swe_moscow_holder ?= {
			add_opinion = {
				target = root
				modifier = opinion_insulted
			}
			if = {
				limit = { is_subject_of = root }
				add_liberty_desire = liberty_desire_mild_plus
			}
		}
		custom_tooltip = {
			text = swe_russia.1.a_tt
			every_owned_location = {
				limit = { region = region:russian_region }
				every_pop = {
					limit = { religion = religion:orthodox }
					add_pop_satisfaction = pop_satisfaction_weak_penalty
				}
			}
		}
	}

	# A ransom, as Munich paid in 1632. Twice the province's tax base, as conquest_mission_events.4.b.
	option = {
		name = swe_russia.1.b
		ai_chance = {
			base = 30
		}

		add_prestige = prestige_severe_bonus
		add_gold = {
			value = scope:swe_moscow_province.province_tax_base
			multiply = 2
		}
		custom_tooltip = {
			text = swe_russia.1.b_tt
			location:moscow = {
				every_pop = {
					limit = { pop_type = pop_type:burghers }
					add_pop_satisfaction = pop_satisfaction_mild_penalty
				}
			}
		}
	}

	# The churches left alone, as St Sophia was in 1611-17
	option = {
		name = swe_russia.1.c
		ai_chance = {
			base = 30
		}

		add_prestige = prestige_mild_bonus
		scope:swe_moscow_holder ?= {
			add_opinion = {
				target = root
				modifier = opinion_pleased
			}
			if = {
				limit = { is_subject_of = root }
				add_liberty_desire = liberty_desire_mild_minus
			}
		}
		scope:swe_novgorod_holder ?= {
			add_opinion = {
				target = root
				modifier = opinion_pleased
			}
			if = {
				limit = { is_subject_of = root }
				add_liberty_desire = liberty_desire_mild_minus
			}
		}
		custom_tooltip = {
			text = swe_russia.1.c_tt
			every_owned_location = {
				limit = { region = region:russian_region }
				every_pop = {
					limit = { religion = religion:orthodox }
					add_pop_satisfaction = pop_satisfaction_weak_bonus
				}
			}
		}
	}
}

# The governor in Novgorod. After Moscow took Novgorod in 1478, Sweden's business with Russia went through the governor
# there. Ivan IV would not treat with Gustav I, an elected king, as an equal; Gustav opened talks through Prince Mikhail
# Glinsky in 1556, and the forty-year truce, sworn on the cross in Novgorod on 2 April 1557, referred Swedish envoys to
# Novgorod for good (en.wiki Treaty of Novgorod (1557), after de Madariaga 2006 and Heininen & Heikkilä 2002; en.wiki
# Russo-Swedish War (1554-1557), after Shkvarov 2012 and Eriksson 2007). Johan III's embassy of 1569 under Bishop Paul
# Juusten was kept three months in Novgorod by a governor who insisted they treat with him, then held at Murom; Johan
# demanded to treat with the tsar himself, and Ivan answered in 1571 that he should be grateful to deal with the governor
# (en.wiki Correspondence between John III and Ivan IV, after Ericson 2004 and Larsson 2005). Sweden had a resident of its
# own in Moscow from March 1631 (Taimre, Tuna 1/2012, p. 6; en.wiki Catharina Stopia), so the window closes then. The
# opened letter of 1639 is Taimre's alone (p. 9, from the resident's letters). Fires once while a Russian monarchy that
# rules Novgorod from elsewhere borders us, at peace and with no truce. The truce is SCALED_TRUCE_YEARS (10, 00_defines);
# the prestige cost is king_in_prussia.5's for accepting a lower rank; refusing costs what a slight costs in
# random_event.18, the CB for them. No border-raiding option: whose raids began the war of 1554 is unclear, and the
# player can make war by hand.
swe_russia.2 = {
	type = country_event
	title = swe_russia.2.title
	desc = swe_russia.2.desc
	outcome = neutral
	historical_info = swe_russia.2.historical_info
	image = "gfx/interface/illustrations/government/diplomacy_illustration.dds" # As vanilla's flavor_rus.21

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1478.1.1
		to = 1631.3.1
		monthly_chance = 3 # Median about two years after such a neighbour appears
	}

	trigger = {
		has_or_had_tag = SWE
		is_subject = no
		location:novgorod = { is_capital = no }
		location:novgorod.owner ?= {
			NOT = { this = root }
			NOT = { is_subject_of = root }
			government_type = government_type:monarchy
			culture ?= { has_culture_group = culture_group:russian_group }
			is_neighbor_of = root
			NOT = { is_at_war_with = root }
			NOT = { has_truce_with = root }
		}
	}

	immediate = {
		# Our ruler first, then theirs: two portraits
		ruler_or_regent ?= { save_scope_as = target_character }
		location:novgorod.owner = {
			save_scope_as = swe_tsar
			ruler_or_regent ?= { save_scope_as = swe_tsar_ruler }
		}
	}

	# A truce sworn on the cross before the governor, as in 1557
	option = {
		name = swe_russia.2.a
		historical_option = yes
		ai_chance = {
			base = 60
		}

		add_truce_with = {
			target = scope:swe_tsar
			mutual = yes
			years = 10
		}
		add_prestige = prestige_mild_penalty
		scope:swe_tsar = {
			add_opinion = {
				target = root
				modifier = opinion_pleased
			}
		}
	}

	# Johan III's answer: a king treats with the prince himself
	option = {
		name = swe_russia.2.b
		ai_chance = {
			base = 40
		}

		add_prestige = prestige_mild_bonus
		scope:swe_tsar = {
			add_opinion = {
				target = root
				modifier = opinion_insulted
			}
			add_casus_belli = {
				target = root
				type = casus_belli:cb_insulted_us
			}
		}
	}
}

# The Dvina mouth. From 1553 western ships reached Russia up the Northern Dvina, and Archangel (1584) became Russia's
# gate to western Europe. Sweden wanted that trade through its own ports: Teusina (1595) channelled the foreigners'
# trade through Viborg and Reval (histdoc.net translation; en.wiki Treaty of Teusina); Philipp Crusius, later von
# Krusenstjerna, urged it in memoranda from 1643 to 1660 and the customs of 1648 cut the transit duty at Reval, Narva and
# Nyen (en.wiki after Soom, SBL; Nyström 1937), his memorandum of 1648 proposing to "dismember" the Archangel voyage
# (Nyström, n. 64). After the embassy of 1673-74 Jochum Lillienhoff told the Board of Trade the only course left was to
# take Archangel and block its harbour mouth (Nyström §8; single source, so not in the text). In 1701 seven ships under
# Commodore Carl Henrik von Löwe sailed from Göteborg to destroy the port; pilots taken on the way ran two aground under
# the new fort of Novodvinsk (de.wiki Angriff auf Archangelsk, after Anderson 1969 and Hoffmann 2010; historyofwar.org).
# Fires from the age of the Reformation while we own Arkhangelsk and one of Narva, Nyen and Reval, and Russia has no
# Baltic port (swe_russia_shut_out_trigger, as swe_baltic.4). Closing the bar cuts the trade embark efficiency that
# Arkhangelsk's harbour gives it; the cargo goes to the Baltic gates (swe_white_sea_cargo). Option b's licent at
# Archangel is our inference from the Baltic licent, not history.
swe_russia.3 = {
	type = country_event
	title = swe_russia.3.title
	desc = swe_russia.3.desc
	outcome = neutral
	historical_info = swe_russia.3.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 3 # Median about two years after we hold the Dvina mouth and a Baltic gate
	}

	trigger = {
		has_or_had_tag = SWE
		current_age_or_later = {
			age = age_4_reformation
		}
		owns = location:arkhangelsk
		OR = {
			owns = location:narva
			owns = location:neva
			owns = location:tallinn
		}
		swe_russia_shut_out_trigger = yes
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:nobles_estate }
		# The same "Russia" as swe_baltic.4
		if = {
			limit = {
				location:novgorod.owner ?= { NOT = { swe_is_swedish_realm_trigger = yes } }
			}
			location:novgorod.owner = { save_scope_as = swe_russian_hinterland }
		}
		else = {
			location:moscow.owner ?= { save_scope_as = swe_russian_hinterland }
		}
	}

	# Close the bar, as Lillienhoff proposed in 1674 and the squadron of 1701 set out to do
	option = {
		name = swe_russia.3.a
		historical_option = yes
		ai_chance = {
			base = 40
		}

		location:arkhangelsk = {
			add_location_modifier = { modifier = swe_dvina_closed years = 20 }
			change_prosperity = prosperity_weak_penalty
		}
		custom_tooltip = {
			text = swe_russia.3.a_tt
			every_owned_location = {
				limit = {
					swe_russian_transit_port_trigger = yes
					NOT = { has_location_modifier = swe_white_sea_cargo }
				}
				add_location_modifier = { modifier = swe_white_sea_cargo years = 20 }
			}
		}
	}

	# The licent, as at Riga and in Prussia
	option = {
		name = swe_russia.3.b
		ai_chance = {
			base = 35
		}

		location:arkhangelsk = {
			add_location_modifier = { modifier = swe_riga_licent years = modifier_duration_years_very_long }
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_penalty }
	}

	# Leave the convoys alone
	option = {
		name = swe_russia.3.c
		ai_chance = {
			base = 25
		}

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
	}
}

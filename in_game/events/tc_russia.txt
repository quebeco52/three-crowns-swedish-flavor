namespace = swe_russia

# Content for the Russian lands beyond Ingria and Kexholm. Sweden held Novgorod in 1611-17 and never Moscow, so this is a
# milestone for the player (AGENTS.md, "Events must actually fire"): it fires whenever Sweden or a Swedish subject
# holds each city.

# Novgorod and Moscow. Under the Treaty of Vyborg (1609) Vasily Shuisky hired a mercenary corps from Karl IX, paid by
# Russia, for Kexholm, and De la Gardie entered Moscow with Skopin-Shuisky in March 1610; the mercenaries went over
# to the Poles at Klushino on 24 June 1610 (SBL, Jakob De la Gardie). De la Gardie stormed Novgorod on 16 July 1611
# and held it to 1617; Widekindi (1671) has him promise to keep its "ancient religion of the Greek rite" inviolate, St
# Sophia kept its services, and he demanded 6,000 roubles of Metropolitan Isidor in 1616 (A. A. Selin, Novgorod 1611-17,
# from RGADA and Riksarkivet). He came home with sables and church bells, which he said he had bought from soldiers who
# took them "med krigets rätt", and left the doors thought to be Sigtuna's "out of regard for opinion in Novgorod"
# (SBL); the Novgorod archives are in Riksarkivet. Gustav II Adolf promised Munich safety and its religion for 300,000
# Reichstaler in 1632; it raised about a third and gave 42 hostages (Altmann 2018). Prague's castle treasures went to
# Stockholm in 1648 (Uppsala University Library, Codex Argenteus). Karl XII meant, in Piper's words, to make the peace
# "i själva Moskva" (Åberg, Svensk Tidskrift 1959, after Haintz) and turned south at Tatarsk in September 1708 (SBL,
# Karl XII). The Kremlin treasury really was looted in 1610-12, but by the Polish-Lithuanian garrison (Kozlyakov), so
# option a rests on Munich, Prague and Novgorod, not on that. Notes: .research/milestones-2026-11-10/russia.md.
swe_russia.1 = {
	type = country_event
	title = swe_russia.1.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:swe_moscow_holder }
				desc = swe_russia.1.desc_subject
			}
			triggered_desc = {
				desc = swe_russia.1.desc
			}
		}
	}
	outcome = positive
	historical_info = swe_russia.1.historical_info

	fire_only_once = yes
	major = yes
	major_trigger = {
		capital ?= {
			OR = {
				region = region:russian_region
				region = region:ruthenia_region
				region = region:baltic_region
				region = region:scandinavian_region
			}
		}
	}
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 10 # Median about seven months after our realm holds both cities
	}

	trigger = {
		has_or_had_tag = SWE
		custom_tooltip = {
			text = swe_russia.1.novgorod_tt
			location:novgorod.owner ?= { swe_is_swedish_realm_trigger = yes }
		}
		custom_tooltip = {
			text = swe_russia.1.moscow_tt
			location:moscow.owner ?= { swe_is_swedish_realm_trigger = yes }
		}
	}

	illustration_tags = {
		10 = armed
		10 = exterior
	}

	immediate = {
		ruler_or_regent ?= { save_scope_as = target_character }
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:clergy_estate }
		# A subject holding either city, saved once if it holds both
		location:moscow.owner ?= {
			if = {
				limit = { NOT = { this = root } }
				save_scope_as = swe_moscow_holder
			}
		}
		location:novgorod.owner ?= {
			if = {
				limit = {
					NOT = { this = root }
					NOT = { owns = location:moscow }
				}
				save_scope_as = swe_novgorod_holder
			}
		}
		# The gold of options a and b follows Moscow's own wealth, as vanilla's conquest_mission_events.4 does
		location:moscow = {
			province = { save_scope_as = swe_moscow_province }
		}
	}

	# The treasury to Stockholm, as from Prague in 1648. Vanilla's looting of a conquered capital
	# (conquest_mission_events.4.a): four times the province's tax base, and mild prosperity loss across it.
	option = {
		name = swe_russia.1.a
		ai_chance = {
			base = 40
		}

		add_prestige = prestige_extreme_bonus
		add_gold = {
			value = scope:swe_moscow_province.province_tax_base
			multiply = 4
		}
		custom_tooltip = {
			text = swe_russia.1.a_prosperity_tt
			scope:swe_moscow_province = {
				every_location_in_province = {
					limit = {
						owner ?= { swe_is_swedish_realm_trigger = yes }
					}
					change_prosperity = prosperity_mild_penalty
				}
			}
		}
		scope:swe_moscow_holder ?= {
			add_opinion = {
				target = root
				modifier = opinion_insulted
			}
			if = {
				limit = { is_subject_of = root }
				add_liberty_desire = liberty_desire_mild_plus
			}
		}
		custom_tooltip = {
			text = swe_russia.1.a_tt
			every_owned_location = {
				limit = { region = region:russian_region }
				every_pop = {
					limit = { religion = religion:orthodox }
					add_pop_satisfaction = pop_satisfaction_weak_penalty
				}
			}
		}
	}

	# A ransom, as Munich paid in 1632. Twice the province's tax base, as conquest_mission_events.4.b.
	option = {
		name = swe_russia.1.b
		ai_chance = {
			base = 30
		}

		add_prestige = prestige_severe_bonus
		add_gold = {
			value = scope:swe_moscow_province.province_tax_base
			multiply = 2
		}
		custom_tooltip = {
			text = swe_russia.1.b_tt
			location:moscow = {
				every_pop = {
					limit = { pop_type = pop_type:burghers }
					add_pop_satisfaction = pop_satisfaction_mild_penalty
				}
			}
		}
	}

	# The churches left alone, as St Sophia was in 1611-17
	option = {
		name = swe_russia.1.c
		ai_chance = {
			base = 30
		}

		add_prestige = prestige_mild_bonus
		scope:swe_moscow_holder ?= {
			add_opinion = {
				target = root
				modifier = opinion_pleased
			}
			if = {
				limit = { is_subject_of = root }
				add_liberty_desire = liberty_desire_mild_minus
			}
		}
		scope:swe_novgorod_holder ?= {
			add_opinion = {
				target = root
				modifier = opinion_pleased
			}
			if = {
				limit = { is_subject_of = root }
				add_liberty_desire = liberty_desire_mild_minus
			}
		}
		custom_tooltip = {
			text = swe_russia.1.c_tt
			every_owned_location = {
				limit = { region = region:russian_region }
				every_pop = {
					limit = { religion = religion:orthodox }
					add_pop_satisfaction = pop_satisfaction_weak_bonus
				}
			}
		}
	}
}

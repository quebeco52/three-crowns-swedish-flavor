namespace = swe_russia

# Content for the Russian lands beyond Ingria and Kexholm.

# Novgorod and Moscow, a milestone: our realm holds both cities. Sweden held Novgorod in 1611-17, where St Sophia kept
# its services, and never Moscow (SBL, Jakob De la Gardie; Selin). Option a rests on Munich (1632), Prague (1648) and
# Novgorod, not on the Kremlin treasury, which the Polish-Lithuanian garrison looted in 1610-12 (Kozlyakov).
swe_russia.1 = {
	type = country_event
	title = swe_russia.1.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:swe_moscow_holder }
				desc = swe_russia.1.desc_subject
			}
			triggered_desc = {
				desc = swe_russia.1.desc
			}
		}
	}
	outcome = positive
	historical_info = swe_russia.1.historical_info

	fire_only_once = yes
	major = yes
	major_trigger = {
		capital ?= {
			OR = {
				region = region:russian_region
				region = region:ruthenia_region
				region = region:baltic_region
				region = region:scandinavian_region
			}
		}
	}
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 10
	}

	trigger = {
		has_or_had_tag = SWE
		custom_tooltip = {
			text = swe_russia.1.novgorod_tt
			location:novgorod.owner ?= { swe_is_swedish_realm_trigger = yes }
		}
		custom_tooltip = {
			text = swe_russia.1.moscow_tt
			location:moscow.owner ?= { swe_is_swedish_realm_trigger = yes }
		}
	}

	illustration_tags = {
		10 = armed
		10 = exterior
	}

	immediate = {
		ruler_or_regent ?= { save_scope_as = target_character }
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:clergy_estate }
		# A subject holding either city, saved once if it holds both
		location:moscow.owner ?= {
			if = {
				limit = { NOT = { this = root } }
				save_scope_as = swe_moscow_holder
			}
		}
		location:novgorod.owner ?= {
			if = {
				limit = {
					NOT = { this = root }
					NOT = { owns = location:moscow }
				}
				save_scope_as = swe_novgorod_holder
			}
		}
		# The gold of options a and b follows Moscow's own wealth, as in vanilla's conquest_mission_events.4
		location:moscow = {
			province = { save_scope_as = swe_moscow_province }
		}
	}

	# The treasury to Stockholm, as from Prague in 1648. Vanilla's looting of a conquered capital
	# (conquest_mission_events.4.a).
	option = {
		name = swe_russia.1.a
		ai_chance = {
			base = 40
		}

		add_prestige = prestige_extreme_bonus
		# Guarded because the Historical Events panel previews options without running immediate
		if = {
			limit = { exists = scope:swe_moscow_province }
			add_gold = {
				value = scope:swe_moscow_province.province_tax_base
				multiply = 4
			}
		}
		custom_tooltip = {
			text = swe_russia.1.a_prosperity_tt
			scope:swe_moscow_province = {
				every_location_in_province = {
					limit = {
						owner ?= { swe_is_swedish_realm_trigger = yes }
					}
					change_prosperity = prosperity_mild_penalty
				}
			}
		}
		scope:swe_moscow_holder ?= {
			add_opinion = {
				target = root
				modifier = opinion_insulted
			}
			if = {
				limit = { is_subject_of = root }
				add_liberty_desire = liberty_desire_mild_plus
			}
		}
		custom_tooltip = {
			text = swe_russia.1.a_tt
			every_owned_location = {
				limit = { region = region:russian_region }
				every_pop = {
					limit = { religion = religion:orthodox }
					add_pop_satisfaction = pop_satisfaction_weak_penalty
				}
			}
		}
	}

	# A ransom, as Munich paid in 1632 (vanilla's conquest_mission_events.4.b)
	option = {
		name = swe_russia.1.b
		ai_chance = {
			base = 30
		}

		add_prestige = prestige_severe_bonus
		if = {
			limit = { exists = scope:swe_moscow_province }
			add_gold = {
				value = scope:swe_moscow_province.province_tax_base
				multiply = 2
			}
		}
		custom_tooltip = {
			text = swe_russia.1.b_tt
			location:moscow = {
				every_pop = {
					limit = { pop_type = pop_type:burghers }
					add_pop_satisfaction = pop_satisfaction_mild_penalty
				}
			}
		}
	}

	# The churches left alone, as St Sophia was in 1611-17
	option = {
		name = swe_russia.1.c
		ai_chance = {
			base = 30
		}

		add_prestige = prestige_mild_bonus
		scope:swe_moscow_holder ?= {
			add_opinion = {
				target = root
				modifier = opinion_pleased
			}
			if = {
				limit = { is_subject_of = root }
				add_liberty_desire = liberty_desire_mild_minus
			}
		}
		scope:swe_novgorod_holder ?= {
			add_opinion = {
				target = root
				modifier = opinion_pleased
			}
			if = {
				limit = { is_subject_of = root }
				add_liberty_desire = liberty_desire_mild_minus
			}
		}
		custom_tooltip = {
			text = swe_russia.1.c_tt
			every_owned_location = {
				limit = { region = region:russian_region }
				every_pop = {
					limit = { religion = religion:orthodox }
					add_pop_satisfaction = pop_satisfaction_weak_bonus
				}
			}
		}
	}
}

# The governor in Novgorod. After 1478 Sweden dealt with Russia through the governor there: Ivan IV would not treat
# with Gustav I, an elected king, as an equal, and when Johan III demanded to treat with the tsar himself, Ivan
# answered in 1571 that he should be grateful to deal with the governor (en.wiki Treaty of Novgorod; Correspondence
# between John III and Ivan IV). Sweden had a resident in Moscow from 1631 (Taimre), so the window closes then.
swe_russia.2 = {
	type = country_event
	title = swe_russia.2.title
	desc = swe_russia.2.desc
	outcome = neutral
	historical_info = swe_russia.2.historical_info
	image = "gfx/interface/illustrations/government/diplomacy_illustration.dds" # As vanilla's flavor_rus.21

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1478.1.1
		to = 1631.3.1
		monthly_chance = 1 # Mostly fired by tc_pulse_russia_2
	}

	trigger = {
		has_or_had_tag = SWE
		is_subject = no
		location:novgorod = { is_capital = no }
		location:novgorod.owner ?= {
			NOT = { this = root }
			NOT = { is_subject_of = root }
			government_type = government_type:monarchy
			culture ?= { has_culture_group = culture_group:russian_group }
			is_neighbor_of = root
			NOT = { is_at_war_with = root }
			NOT = { has_truce_with = root }
		}
	}

	immediate = {
		# Our ruler first, then theirs: two portraits
		ruler_or_regent ?= { save_scope_as = target_character }
		location:novgorod.owner = {
			save_scope_as = swe_tsar
			ruler_or_regent ?= { save_scope_as = swe_tsar_ruler }
		}
	}

	# A truce sworn on the cross before the governor, as in 1557
	option = {
		name = swe_russia.2.a
		historical_option = yes
		ai_chance = {
			base = 60
		}

		add_truce_with = {
			target = scope:swe_tsar
			mutual = yes
			years = 10
		}
		add_prestige = prestige_mild_penalty
		scope:swe_tsar = {
			add_opinion = {
				target = root
				modifier = opinion_pleased
			}
		}
	}

	# Johan III's answer: a king treats with the prince himself
	option = {
		name = swe_russia.2.b
		ai_chance = {
			base = 40
		}

		add_prestige = prestige_mild_bonus
		scope:swe_tsar = {
			add_opinion = {
				target = root
				modifier = opinion_insulted
			}
			add_casus_belli = {
				target = root
				type = casus_belli:cb_insulted_us
			}
		}
	}
}

# The Dvina mouth. Archangel (1584) became Russia's gate to western Europe, and Sweden wanted that trade through its own
# ports (en.wiki Treaty of Teusina; Nyström 1937). In 1701 seven ships under Carl Henrik von Löwe sailed to destroy the
# port, but two ran aground under the new fort of Novodvinsk (de.wiki Angriff auf Archangelsk). Option b's licent at
# Archangel is our inference from the Baltic licent, not history.
swe_russia.3 = {
	type = country_event
	title = swe_russia.3.title
	desc = swe_russia.3.desc
	outcome = neutral
	historical_info = swe_russia.3.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = SWE
		tag = SCA
		from = 1337.1.1
		to = 1837.1.1
		monthly_chance = 15
	}

	trigger = {
		has_or_had_tag = SWE
		current_age_or_later = {
			age = age_4_reformation
		}
		owns = location:arkhangelsk
		OR = {
			owns = location:narva
			owns = location:neva
			owns = location:tallinn
		}
		swe_russia_shut_out_trigger = yes
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:nobles_estate }
		# The same "Russia" as swe_baltic.4
		if = {
			limit = {
				location:novgorod.owner ?= { NOT = { swe_is_swedish_realm_trigger = yes } }
			}
			location:novgorod.owner = { save_scope_as = swe_russian_hinterland }
		}
		else = {
			location:moscow.owner ?= { save_scope_as = swe_russian_hinterland }
		}
	}

	# Close the bar, as Lillienhoff proposed in 1674 and the squadron of 1701 set out to do
	option = {
		name = swe_russia.3.a
		historical_option = yes
		ai_chance = {
			base = 40
		}

		location:arkhangelsk = {
			add_location_modifier = { modifier = swe_dvina_closed years = 20 }
			change_prosperity = prosperity_weak_penalty
		}
		custom_tooltip = {
			text = swe_russia.3.a_tt
			every_owned_location = {
				limit = {
					swe_russian_transit_port_trigger = yes
					NOT = { has_location_modifier = swe_white_sea_cargo }
				}
				add_location_modifier = { modifier = swe_white_sea_cargo years = 20 }
			}
		}
	}

	# The licent, as at Riga and in Prussia
	option = {
		name = swe_russia.3.b
		ai_chance = {
			base = 35
		}

		location:arkhangelsk = {
			add_location_modifier = { modifier = swe_riga_licent years = modifier_duration_years_very_long }
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_penalty }
	}

	# Leave the convoys alone
	option = {
		name = swe_russia.3.c
		ai_chance = {
			base = 25
		}

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_weak_bonus }
	}
}

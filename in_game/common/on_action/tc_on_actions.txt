# Hooks into vanilla on_actions without replacing them (same pattern as ai_personalities_setup.txt)
on_game_start = {
	on_actions = { tc_on_game_start }
}

on_ruler_death_delayed = {
	on_actions = { tc_on_ruler_death_delayed }
}

on_new_ruler = {
	on_actions = { tc_on_new_ruler tc_on_new_ruler_regalia }
}

yearly_country_pulse = {
	on_actions = { tc_yearly_country_pulse }
}

# Gustav Vasa's line: a third a year while he has fewer than two living sons (swe_gustav_vasa_sons).
# The cheap outer trigger follows vanilla's advice in country_yearly.txt.
tc_yearly_country_pulse = {
	trigger = {
		has_or_had_tag = SWE
		exists = character:swe_gustav_vasa
	}

	effect = {
		random = {
			chance = 33
			swe_gustav_vasa_sons = yes
		}
	}
}

# A new Swedish monarch under Konungabalken is asked to ride the Eriksgata (swe_eriksgata.1), after any
# funeral. Delayed like vanilla's culture_religion_events.2, which still reads scope:new_ruler when it fires.
tc_on_new_ruler = {
	trigger = {
		has_or_had_tag = SWE
		government_type = government_type:monarchy
		has_estate_privilege = estate_privilege:swe_konungabalken
		# Not the wave of rulers set at game start (vanilla's guard in succession.100)
		game_is_initialized = yes
		# Not when the same monarch is set again, e.g. under joint rule (vanilla culture_religion_events)
		trigger_if = {
			limit = {
				exists = scope:old_ruler
				exists = scope:new_ruler
			}
			NOT = { scope:old_ruler = scope:new_ruler }
		}
	}

	events = {
		delay = { days = { 60 120 } }
		swe_eriksgata.1
	}
}

# The first Swedish monarch to succeed from 1544, when the crown became hereditary (swe_vasa.3), until 1650 is offered
# new regalia for the coronation (swe_stockholm.6, which fires only once). A regency has no ruler, so a child's
# accession passes it on to the next reign. Guards as in tc_on_new_ruler.
tc_on_new_ruler_regalia = {
	trigger = {
		has_or_had_tag = SWE
		government_type = government_type:monarchy
		current_date >= 1544.1.1
		current_date < 1650.1.1
		game_is_initialized = yes
		trigger_if = {
			limit = {
				exists = scope:old_ruler
				exists = scope:new_ruler
			}
			NOT = { scope:old_ruler = scope:new_ruler }
		}
	}

	events = {
		delay = { days = { 30 60 } }
		swe_stockholm.6
	}
}

# A Swedish monarch dies: Tre Kronor may burn while the body lies in state there, as it did in 1697
# (swe_stockholm.1), and the funeral follows a month or two later, after any fire (swe_stockholm.2)
tc_on_ruler_death_delayed = {
	trigger = {
		has_or_had_tag = SWE
	}

	random_events = {
		chance_to_happen = 25
		100 = swe_stockholm.1
	}

	effect = {
		trigger_event_non_silently = { id = swe_stockholm.2 days = { 30 60 } }
	}
}

tc_on_game_start = {
	effect = {
		# Sweden starts with Konungabalken. Granted here rather than in setup/1337
		# so the mod never redefines vanilla's SWE country block.
		c:SWE ?= {
			# Only for a start before the Hereditary Union of 1544 (swe_vasa.3), which ended it
			if = {
				limit = { current_date < 1544.1.1 }
				unlock_estate_privilege_effect = { type = swe_konungabalken }
				if = {
					limit = { NOT = { has_estate_privilege = estate_privilege:swe_konungabalken } }
					grant_estate_privilege = estate_privilege:swe_konungabalken
				}
			}
			# The Birkarlar already trade with the Sámi in 1337; the town charters of 1621 ended them
			if = {
				limit = {
					current_date < 1621.1.1
					NOT = { has_estate_privilege = estate_privilege:swe_birkarl_rights }
				}
				grant_estate_privilege = estate_privilege:swe_birkarl_rights
			}
		}
		# The salmon fisheries belong to the location, not to Sweden: Torneå starts as Kven land in 1337
		location:tornea ?= {
			if = {
				limit = { NOT = { has_location_modifier = swe_northern_salmon_fishery } }
				add_location_modifier = { modifier = swe_northern_salmon_fishery months = -1 }
			}
		}
		# Kemi and Oulu lived on salmon until the timber trade of the 19th century, so they produce
		# fish instead of vanilla's lumber. Only changed while they still have vanilla's lumber.
		location:kiemi ?= {
			if = {
				limit = { raw_material = goods:lumber }
				change_raw_material = goods:fish
			}
			if = {
				limit = { NOT = { has_location_modifier = swe_northern_salmon_fishery } }
				add_location_modifier = { modifier = swe_northern_salmon_fishery months = -1 }
			}
		}
		location:uleaborg ?= {
			if = {
				limit = { raw_material = goods:lumber }
				change_raw_material = goods:fish
			}
			if = {
				limit = { NOT = { has_location_modifier = swe_northern_salmon_fishery } }
				add_location_modifier = { modifier = swe_northern_salmon_fishery months = -1 }
			}
		}
		# Västerås lies on the Västmanland plain, the farmland on Lake Mälaren that fed the mines of
		# Bergslagen. Millet like neighbouring Enköping, vanilla's grain for most Swedish farmland;
		# the mining locations keep no food, as Bergslagen bought its grain.
		location:vasteras ?= {
			if = {
				limit = { raw_material = goods:lumber }
				change_raw_material = goods:millet
			}
		}
		# Färnebo härad was Filipstads bergslag, "Jernberghet i Wærmelande" in 1413: the iron mines of
		# Persberg, Nordmark and Långban, and Filipstad's town privileges in 1611
		location:farnebo ?= {
			if = {
				limit = { raw_material = goods:lumber }
				change_raw_material = goods:iron
			}
		}
		# The Garpenberg copper mine east of Hedemora, worked since the 1200s (charter of 1354)
		location:hedemora ?= {
			if = {
				limit = { raw_material = goods:horses }
				change_raw_material = goods:copper
			}
		}
		# Göksholm's meadows on the Hjälmaren shore suit livestock; the lake was only lowered for
		# farmland in 1878-87
		location:goksholm ?= {
			if = {
				limit = { raw_material = goods:medicaments }
				change_raw_material = goods:livestock
			}
		}
		# Älvkarleby, at the mouth of the Dalälven, lies in tierp, which already produces fish
		location:tierp ?= {
			if = {
				limit = { NOT = { has_location_modifier = swe_alvkarleby_salmon_fishery } }
				add_location_modifier = { modifier = swe_alvkarleby_salmon_fishery months = -1 }
			}
		}
		# The iron of Bergslagen belongs to the location, like the salmon fisheries. Färnebo is iron
		# from the change above.
		location:nora ?= {
			if = {
				limit = { NOT = { has_location_modifier = swe_bergslagen_iron } }
				add_location_modifier = { modifier = swe_bergslagen_iron months = -1 }
			}
		}
		location:skinskatteberg ?= {
			if = {
				limit = { NOT = { has_location_modifier = swe_bergslagen_iron } }
				add_location_modifier = { modifier = swe_bergslagen_iron months = -1 }
			}
		}
		location:grythyttan ?= {
			if = {
				limit = { NOT = { has_location_modifier = swe_bergslagen_iron } }
				add_location_modifier = { modifier = swe_bergslagen_iron months = -1 }
			}
		}
		location:avesta ?= {
			if = {
				limit = { NOT = { has_location_modifier = swe_bergslagen_iron } }
				add_location_modifier = { modifier = swe_bergslagen_iron months = -1 }
			}
		}
		location:grange ?= {
			if = {
				limit = { NOT = { has_location_modifier = swe_bergslagen_iron } }
				add_location_modifier = { modifier = swe_bergslagen_iron months = -1 }
			}
		}
		location:farnebo ?= {
			if = {
				limit = { NOT = { has_location_modifier = swe_bergslagen_iron } }
				add_location_modifier = { modifier = swe_bergslagen_iron months = -1 }
			}
		}
		# Half the mayors and councillors of a Swedish town had to be Germans, a rule written into
		# Magnus Eriksson's town law around 1350 and struck out in 1471 (swe_svenska_radman_decision).
		# Vanilla gives Kalmar only Swedish burghers, so it gets its German community here;
		# Stockholm already starts with more German burghers than Swedish ones.
		location:kalmar ?= {
			if = {
				limit = {
					NOT = {
						any_pop = {
							pop_type = pop_type:burghers
							culture = { has_culture_group = culture_group:german_group }
						}
					}
				}
				add_pop = {
					culture = culture:lower_saxon
					religion = religion:catholic
					type = pop_type:burghers
					size = 0.6
				}
			}
		}
		if = {
			limit = { current_date < 1471.1.1 }
			location:stockholm ?= {
				if = {
					limit = {
						owner ?= c:SWE
						NOT = { has_location_modifier = swe_tyska_radet }
					}
					add_location_modifier = { modifier = swe_tyska_radet months = -1 }
				}
			}
			location:kalmar ?= {
				if = {
					limit = {
						owner ?= c:SWE
						NOT = { has_location_modifier = swe_tyska_radet }
					}
					add_location_modifier = { modifier = swe_tyska_radet months = -1 }
				}
			}
		}
		# Tre Kronor, the royal castle of Stockholm, has stood since the 13th century. Its walls are
		# vanilla's castle there; the building is the royal residence, which burned in 1697 (swe_stockholm.1).
		# Set as a level like vanilla's banking.1, not constructed, so no construction is reported.
		if = {
			limit = { current_date < 1697.5.7 }
			location:stockholm ?= {
				if = {
					limit = { NOT = { has_building = building_type:swe_tre_kronor } }
					change_building_level_in_location = {
						building = building_type:swe_tre_kronor
						value = 1
						owner = owner
					}
				}
			}
		}
	}
}

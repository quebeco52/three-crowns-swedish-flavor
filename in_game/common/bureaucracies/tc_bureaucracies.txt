# The colleges of the Instrument of Government (regeringsform) of 1634, as Swedish bureaucracies. Shaped like
# D009's Spanish councils: shown once Sweden can hold any bureaucracy (the generic gate), allowed once
# swe_regeringsform_decision has set swe_regeringsform. Each is a generic bureaucracy (generic.txt) with a
# Swedish twist, not a stronger copy, and the generic ones stay open beside them. Svea hovrätt, the first
# college, is vanilla's court_of_appeals (flavor_swe.15), and Kanslikollegium is left out. Bergskollegium
# stays a policy in the mining law (tc_laws.txt), as a bureaucracy's modifiers are country-wide.

# Kammarkollegium: the Chamber College kept the accounts of every crown revenue. Based on vanilla's
# board_of_revenue_bureaucracy, with tax efficiency in place of credit when funded.
swe_kammarkollegium_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}

	potential = {
		has_or_had_tag = SWE
		OR = {
			has_embraced_institution = institution:confessionalism
			has_advance = generic_bureaucracy_4_advance
		}
	}

	allow = {
		custom_tooltip = {
			text = swe_regeringsform_colleges_tt
			has_variable = swe_regeringsform
		}
	}

	estates_that_like = {
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	neutral_modifier = {
		monthly_towards_centralization = societal_value_minor_monthly_move
	}

	positive_modifier = {
		scale = { value = scope:maintenance }
		tax_income_efficiency = medium_tax_income_efficiency_bonus
	}

	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		estate_enrichment = 0.1
	}
}

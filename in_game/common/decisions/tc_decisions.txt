# Bottniska handelstvånget: the trade ordinances of 1614 and 1617 bind the Bothnian towns to
# Stockholm, and the crown takes the Sámi trade from the Birkarlar (swe_birkarl_rights -> swe_bottniska_handelstvanget).
# Needs a town on the Bothnian coast, since a staple needs towns to bind. None start as towns,
# so one has to be built first.
swe_bottniska_handelstvanget_decision = {
	decision_category = country_specific_decisions
	historical_info = swe_bottniska_handelstvanget_decision.historical_info
	only_once = yes

	potential = {
		has_or_had_tag = SWE
		current_age_or_later = {
			age = age_4_reformation
		}
		NOT = { has_unlocked_estate_privilege_trigger = { type = swe_bottniska_handelstvanget } }
	}

	allow = {
		owns = location:stockholm
		any_owned_location = {
			swe_bothnian_coast_trigger = yes
			OR = {
				location_rank = location_rank:town
				location_rank = location_rank:city
				location_rank = location_rank:megalopolis
			}
		}
	}

	ai_will_do = { value = 100 }

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:peasants_estate }
	}

	option = {
		name = swe_bottniska_handelstvanget_decision.a
		ai_chance = { value = 100 }
		effect = {
			unlock_estate_privilege_effect = { type = swe_bottniska_handelstvanget }
			if = {
				limit = { has_estate_privilege = estate_privilege:swe_birkarl_rights }
				revoke_estate_privilege = estate_privilege:swe_birkarl_rights
			}
			grant_estate_privilege = estate_privilege:swe_bottniska_handelstvanget
			add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_penalty }
		}
	}
}

# Svenska rådmän: in 1471, after Brunkeberg, the town law's rule that half the councillors be Germans
# was struck out. Opens once the Swedish burghers of Stockholm outnumber the Germans, so assimilation
# and migration set the date. Ends swe_tyska_radet in every town at once, as it was one law. The Hansa
# takes vanilla's swe_denies_hsa, as in flavor_swe.55.
swe_svenska_radman_decision = {
	decision_category = country_specific_decisions
	historical_info = swe_svenska_radman_decision.historical_info
	only_once = yes

	potential = {
		has_or_had_tag = SWE
		OR = {
			location:stockholm = { has_location_modifier = swe_tyska_radet }
			location:kalmar = { has_location_modifier = swe_tyska_radet }
		}
	}

	allow = {
		owns = location:stockholm
		location:stockholm = {
			custom_tooltip = {
				text = swe_svenska_radman_decision.majority_tt
				swe_german_burgher_lead < 0
			}
		}
	}

	# The AI waits until swe_council.1's window (1387-1420) has closed, since that event needs the
	# German council still in place. Form from vanilla generic_age_decisions.txt
	ai_will_do = {
		value = 0
		add = {
			if = {
				limit = { current_date >= 1420.1.1 }
				add = 100
			}
		}
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
	}

	option = {
		name = swe_svenska_radman_decision.a
		ai_chance = { value = 100 }
		effect = {
			location:stockholm = {
				if = {
					limit = { has_location_modifier = swe_tyska_radet }
					remove_location_modifier = swe_tyska_radet
				}
			}
			location:kalmar = {
				if = {
					limit = { has_location_modifier = swe_tyska_radet }
					remove_location_modifier = swe_tyska_radet
				}
			}
			c:HSA ?= { add_opinion = { target = root modifier = swe_denies_hsa } }
		}
	}
}

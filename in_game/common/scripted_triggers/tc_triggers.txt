# Province groups used in more than one place, kept here so every copy stays the same.
# All are location scope unless noted. No vanilla area matches them (norrland_area mixes them all).

# The valleys and lappmarks of the Birkarl trade (swe_birkarl_rights, swe_bottniska_handelstvanget)
swe_birkarl_lands_trigger = {
	OR = {
		province_definition = province_definition:vasterbotten_province
		province_definition = province_definition:pite_lappmark_province
		province_definition = province_definition:lule_lappmark_province
		province_definition = province_definition:torne_lappmark_province
		province_definition = province_definition:kemi_lappmark_province
	}
}

# The East and West Bothnian coast that the trade ordinances of 1614 and 1617 bound to Stockholm
swe_bothnian_coast_trigger = {
	OR = {
		province_definition = province_definition:halsingland_province
		province_definition = province_definition:angermanland_province
		province_definition = province_definition:vasterbotten_province
		province_definition = province_definition:norra_osterbotten_province
		province_definition = province_definition:sodra_osterbotten_province
	}
}

# Västerbotten and the five lappmarks, where the swe_lappmark events act on Sweden's own locations
swe_far_north_trigger = {
	OR = {
		province_definition = province_definition:vasterbotten_province
		province_definition = province_definition:ume_lappmark_province
		province_definition = province_definition:pite_lappmark_province
		province_definition = province_definition:lule_lappmark_province
		province_definition = province_definition:torne_lappmark_province
		province_definition = province_definition:kemi_lappmark_province
	}
}

# Jämtland, plus Idre, which vanilla puts in dalarna_province though it was Norwegian land too (swe_jamtland)
swe_jamtland_lands_trigger = {
	OR = {
		province_definition = province_definition:jamtland_province
		this = location:idre
	}
}

# Kexholm county and Ingria, ceded by Russia at Stolbovo (swe_ingria.1)
swe_kexholm_ingria_trigger = {
	OR = {
		province_definition = province_definition:korela_province
		province_definition = province_definition:lopskaya_province
		province_definition = province_definition:vodskaya_province
	}
}

# The German lands the Wismar Tribunal judged as the Crown's imperial fiefs (swe_wismar_tribunal_decision):
# north Germany without the Low Countries (Pomerania, Mecklenburg, Bremen-Verden and their neighbours), and
# only imperial land while the Empire exists. Same HRE guard as vanilla's hre_buildings.txt.
swe_german_provinces_trigger = {
	region = region:north_german_region
	NOT = { area = area:holland_area }
	NOT = { area = area:brabant_area }
	NOT = { area = area:flanders_area }
	NOT = { area = area:wallonia_area }
	trigger_if = {
		limit = { exists = international_organization:hre }
		is_owned_by_international_organization = international_organization:hre
	}
}

# Character scope. A councillor who can head a Förmyndarregering (swe_council_regency). Foreigners are out, after the
# council's decision of 1319 barring them from its seats, which Albrekt had to respect in 1371 (one secondary text,
# Sverige 1266-1397; the assurance itself is DS X nr 75). Accepted cultures count as the realm's own, not only Swedish.
# Same culture check as vanilla's manpower_buildings.txt, which runs it from a location. From a character, prev is that
# character: by console on 2026-11-05, Blanche of Namur (Champenois) gave false and Knut Jonsson (Swedish) gave true.
swe_council_regent_trigger = {
	can_become_a_regent = yes
	owner ?= { has_primary_or_accepted_culture = prev.culture }
}

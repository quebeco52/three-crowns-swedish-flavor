# so the right now shifts power to the peasants and nobles. Enfranchisement stands in for the freeholders' voice
# at the same value vanilla gives the Bergslag and tar rights. Applies to every Scandinavian culture, not just Sweden.

	kept_at_conquest = no
	location_modifier = {
		local_migration_attraction = 0.3
		local_monthly_prosperity = 0.001
		local_peasant_enfranchisment = 0.05
		local_peasants_estate_power = 0.25
		local_nobles_estate_power = 0.25
		local_crown_estate_power = -0.25
	}

	color = goods_amber
}

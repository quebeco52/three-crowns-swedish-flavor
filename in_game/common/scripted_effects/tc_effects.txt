# Effects shared by events in several regions. Each comment lists the events that use it.

# A market for a location: a market village while it is rural, a marketplace level once it is a town or city,
# as market villages are rural buildings and marketplaces town ones. Set as a level like vanilla's banking.1,
# not constructed. Location scope. Used by swe_gotaland.13-15, swe_towns.8 and swe_svealand.1.
swe_grant_market = {
	if = {
		limit = { location_rank = location_rank:rural_settlement }
		change_building_level_in_location = {
			building = building_type:market_village
			value = 1
			owner = owner
		}
	}
	else = {
		change_building_level_in_location = {
			building = building_type:marketplace
			value = 1
			owner = owner
		}
	}
}

# Craftsmen ordered into a town: laborers move from one of our locations to another, about a quarter of a pop
# in all. Laborers, since rural locations hold almost no burghers. add_migration's owner is the source's owner,
# so only from a location we own. Country scope; $from$ and $to$ are locations. Used by swe_gotaland.13 and .15.
swe_craftsmen_move_in = {
	if = {
		limit = { owns = $from$ }
		add_migration = {
			owner = $from$.owner
			from_location = $from$
			to_location = $to$
			type = pop_type:laborers
			amount = 0.01
			months = 24
		}
	}
}

# Changes the satisfaction of one pop type in one location, if we own it, so that anger stays where it belongs. Country
# scope; $location$ is a location, $type$ a pop type key, $value$ a pop satisfaction value. Used by swe_gotaland.6, .13-15,
# swe_towns.1, swe_finland_west.5 and swe_germany.1.
swe_pop_satisfaction_in = {
	$location$ = {
		if = {
			limit = { owner ?= root }
			every_pop = {
				limit = { pop_type = pop_type:$type$ }
				add_pop_satisfaction = $value$
			}
		}
	}
}

# Changes the satisfaction of every pop in one location, if we own it: for a whole town ordered to move, since a rural
# location holds almost no burghers. Country scope; $location$ is a location, $value$ a pop satisfaction value. Used by
# swe_finland_west.7, swe_lappmark.8 and swe_scania.7.
swe_all_pops_satisfaction_in = {
	$location$ = {
		if = {
			limit = { owner ?= root }
			every_pop = {
				add_pop_satisfaction = $value$
			}
		}
	}
}

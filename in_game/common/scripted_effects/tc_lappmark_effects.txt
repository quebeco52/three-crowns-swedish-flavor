# Opens the Nasafjäll mine if vanilla's flavor_swe.39 never did. From 1635 to 1659 it yielded only 981 kg of silver, so
# it gets the same cap as Sala in flavor_swe.38.
swe_open_silbojock = {
	location:silbojock = {
		if = {
			limit = { NOT = { raw_material = goods:silver } }
			change_raw_material = goods:silver
			if = {
				limit = { rgo_level > 1 }
				change_max_raw_material_workers = {
					value = rgo_level
					subtract = 1
					multiply = -1
				}
			}
		}
	}
}

# The mountain Sámi turn from hunting to reindeer herding (swe_lappmark.4). Vanilla has no reindeer good, so these
# locations keep their goods and gain a modifier. Silbojock stands for the mountain Sámi of Pite Lappmark and keeps
# the modifier's food once the mine opens. Tärna is left out, as Ume Lappmark's mountain Sámi were counted under
# Pite or Norway in 1602, and so are the forest locations, whose Sámi kept a mixed living. Foreign-held land is
# left alone.
swe_reindeer_herds_location = {
	if = {
		limit = {
			OR = {
				NOT = { exists = owner }
				owner ?= root
			}
		}
		add_location_modifier = { modifier = swe_reindeer_herding months = -1 }
	}
}

swe_reindeer_herds = {
	location:sorsele = { swe_reindeer_herds_location = yes }
	location:silbojock = { swe_reindeer_herds_location = yes }
	location:kvikkjock = { swe_reindeer_herds_location = yes }
	location:sarek = { swe_reindeer_herds_location = yes }
	location:nikkaluokta = { swe_reindeer_herds_location = yes }
	location:kasivarren = { swe_reindeer_herds_location = yes }
	location:abisko = { swe_reindeer_herds_location = yes }
	location:raunala = { swe_reindeer_herds_location = yes }
}

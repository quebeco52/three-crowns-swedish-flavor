# The Orthodox leave for Russia and settlers from Savolax take over their farms (swe_ingria.1), in each province
# where Sweden holds Orthodox land. Amounts are per month; the exodus is sized like vanilla's flavor_fra.12 (the
# Huguenots). No culture filter on the settlers: they were Finns, but the game usually turns Savolax Swedish by
# the mid-1500s, and a filter would then move nobody. Peasants only, since farmers left and farmers took over;
# otherwise the garrisons would leave too.
swe_orthodox_exodus = {
	if = {
		limit = {
			any_owned_location = {
				province_definition = province_definition:$province$
				dominant_religion = religion:orthodox
			}
		}
		add_migration = {
			owner = this
			from = province_definition:$province$
			to = location:moscow.province_definition
			religion = religion:orthodox
			type = pop_type:peasants
			amount = 0.1
			months = 36
		}
		if = {
			limit = {
				any_owned_location = { province_definition = province_definition:savolax_province }
			}
			add_migration = {
				owner = this
				from = province_definition:savolax_province
				to = province_definition:$province$
				type = pop_type:peasants
				amount = 0.005
				months = 240
			}
		}
	}
}

# Adds Swedish policies to vanilla laws without replacing them (same pattern as D009's INJECT:legal_code_law)

INJECT:mining_law = {

	# Bergskollegium (swe_bergskollegium_decision): the Board of Mines ran every mine and ironworks
	# of the realm for the Crown, so the Crown gains power in the mines rather than output, which
	# falun_base, mulltimmer_masugn and polhem_machina_nova already cover.
	# Balanced against zimbabwe_mining_law (same mine expansion cost, without its output) and
	# nobles_mining_law (-0.1 rgo build time everywhere): no estate satisfaction or societal drift,
	# and the rest only in mining locations.
	swe_bergskollegium = {
		unique = yes
		potential = {
			has_or_had_tag = SWE
			has_unlocked_policy_trigger = { type = swe_bergskollegium }
		}
		country_modifier = {
			expand_rgo_mining_cost_modifier = -0.2
		}
		# Every good vanilla mines (method = mining in 00_raw_materials.txt)
		location_modifier = {
			potential_trigger = {
				custom_tooltip = {
					text = swe_bergskollegium_mines_tt
					OR = {
						raw_material = goods:iron
						raw_material = goods:copper
						raw_material = goods:silver
						raw_material = goods:goods_gold
						raw_material = goods:lead
						raw_material = goods:tin
						raw_material = goods:coal
						raw_material = goods:alum
						raw_material = goods:saltpeter
						raw_material = goods:mercury
						raw_material = goods:gems
						raw_material = goods:stone
						raw_material = goods:marble
					}
				}
			}
			local_crown_estate_power = 0.1
			local_rgo_build_time = -0.1
		}
		estate_preferences = {
			crown_estate
		}
		years = 2
	}
}

INJECT:legal_code_law = {

	# Landslagen (swe_landslag_decision): Magnus Eriksson's national law code of about 1350 replaced
	# the provincial laws. Its King's Chapter is the swe_konungabalken privilege Sweden holds from 1337.
	# Balanced against dan_regional_law (three minor value moves, small satisfaction) and vanilla
	# flavor_swe.53, which only moves centralization once and needs Magnus Eriksson himself.
	swe_landslag = {
		unique = yes
		potential = {
			has_or_had_tag = SWE
			has_unlocked_policy_trigger = { type = swe_landslag }
		}
		country_modifier = {
			monthly_towards_centralization = societal_value_minor_monthly_move
			monthly_towards_free_subjects = societal_value_minor_monthly_move
			global_estate_target_satisfaction = small_permanent_target_satisfaction
		}
		# Sveriges rikes lag (swe_rikes_lag_decision): the code of 1734 for town and countryside alike.
		# Kept inside this policy rather than as a policy of its own. civil_law_policy's legislative
		# efficiency on top of the base. No proximity: swedish_royal_post_office already gives +0.1.
		country_modifier = {
			potential_trigger = {
				custom_tooltip = {
					text = swe_rikes_lag_country_bonus
					has_variable = swe_rikes_lag
				}
			}
			legislative_efficiency = 0.1
		}
		estate_preferences = {
			crown_estate
			peasants_estate
		}
		years = 2
	}
}

INJECT:distribution_of_power_law = {

	# Riksdagsordningen: the Riksdag ordinance of 1617 fixed the four estates of the realm, with the
	# peasants as the fourth. No decision: gated by vanilla's kung_och_riksdag and the peasants'
	# seat (peasants_represented_in_parliament), the way dop_law_estates_general is gated by its advance.
	# Balanced against dop_law_english_parliament (-0.33 crown power, +0.05 legitimacy, +0.1 estate
	# max tax, +0.1 nobles and burghers agenda impact, -0.15 political influence): a smaller crown
	# penalty and no legitimacy, with the agenda weight on the peasants.
	swe_riksdagsordningen = {
		unique = yes
		potential = {
			has_or_had_tag = SWE
			has_advance = kung_och_riksdag
		}
		allow = {
			current_age_or_later = {
				age = age_4_reformation
			}
			has_estate_privilege = estate_privilege:peasants_represented_in_parliament
		}
		country_modifier = {
			global_crown_estate_power = -0.2
			global_estate_max_tax = 0.1
			peasants_estate_agenda_impact = 0.2
			burghers_estate_agenda_impact = 0.1
			monthly_political_influence_gain_modifier = -0.1
		}
		# The Age of Liberty (vanilla swe_age_of_freedom): the Secret Committee of the nobles, clergy
		# and burghers ran foreign affairs, finance and defence, and kept the peasants out
		country_modifier = {
			potential_trigger = {
				custom_tooltip = {
					text = swe_riksdagsordningen_liberty_bonus
					has_reform = government_reform:swe_age_of_freedom
				}
			}
			nobles_estate_agenda_impact = 0.25
			peasants_estate_agenda_impact = -0.2
		}
		estate_preferences = {
			peasants_estate
			burghers_estate
		}
		years = 2
	}
}

INJECT:cultural_traditions_law = {

	# Göticism (swe_council.4): the belief that the Goths of antiquity came from Sweden, made the Vasa
	# kings' own by Johannes Magnus's history (1554) and given an office of state in 1630, when Gustav II
	# Adolf made Johannes Bureus riksantikvarie. Shaped like plc_sarmatism, Poland's origin myth in this
	# same law, unlocked by an event (flavor_plc.4): one bonus, one value move, the nobles, 5 years.
	# Taking it gives up martial_society or civil_society, which are about as strong (army tradition
	# 0.05 or peasants' max tax 0.025, each with one value move).
	# Prestige 0.1 is vanilla's most common monthly_prestige in laws; the drift to traditionalist is the
	# reverence for the forefathers, at minor speed (0.05), as the plain move would more than double the cost of
	# embracing institutions within 50 years, just as the scientific revolution arrives.
	swe_goticism = {
		unique = yes
		potential = {
			has_or_had_tag = SWE
			has_unlocked_policy_trigger = { type = swe_goticism }
		}
		country_modifier = {
			monthly_prestige = 0.1
			monthly_towards_traditionalist = societal_value_minor_monthly_move
		}
		estate_preferences = {
			nobles_estate
		}
		years = 5
	}
}

# The colleges of the Instrument of Government are bureaucracies (tc_bureaucracies.txt), not
# policies; Bergskollegium stays in the mining law above, as its effect is limited to the mines.

# Kronans jord: who holds the land of the realm. Every Swedish farm was crown, freehold or noble land
# (krono, skatte, frälse). The Crown granted land and rents away in donations to pay for its wars
# (1611-54), took them back in the Quarter Reduction of 1655 and the Great Reduction from 1680, and
# from 1701 sold crown farms to the peasants on them as freehold (skatteköp: royal letter of 20 March
# 1701, ordinance of 19 September 1723). After the Great Reduction each kind held about a third.
# Sources: Högman, "The Reduction" (hhogman.se); prop. 1930:246; Carlsson in Scandia.
# A law of its own, not policies in nobles_rights_laws, since swe_riddarhuset holds that slot and the
# House of Nobility (1626) stood beside the reductions. Shaped like sco_clan_holdings (a tag-gated
# law) and teu_amber_monopoly_law (one policy for each estate). Vanilla's dated reduction events
# (flavor_swe.17, .18, .20) still fire; this is the standing choice beside them.
# The law starts with no policy: vanilla gives its country laws one in setup or by event.
swe_kronans_jord_law = {
	unique = yes
	law_category = estates

	potential = {
		has_or_had_tag = SWE
		current_age_or_later = {
			age = age_4_reformation
		}
	}

	# Crown farms and rents granted to officers, generals and creditors. Between
	# rural_nobles_authority_policy (+0.05 power, small satisfaction) and strengthen_nobles_rights_policy
	# (+0.1 power, medium satisfaction, levy size 0.2), with half the levy for the service it bought.
	swe_donationer = {
		country_modifier = {
			nobles_estate_target_satisfaction = small_permanent_target_satisfaction
			global_nobles_estate_power = 0.1
			global_crown_estate_power = -0.05
			nobles_estate_levy_size = 0.1
			monthly_towards_aristocracy = societal_value_minor_monthly_move
		}
		estate_preferences = {
			nobles_estate
		}
		years = 2
	}

	# The Crown takes the land back. Vanilla's crown-land reversion is Portugal's por_lei_mental
	# (-0.25 noble power, medium penalty, +0.005 monthly control); the mod never hands out control,
	# so this takes half of absolute_rule_policy's +0.2 crown power instead, and less from the nobles.
	swe_reduktion = {
		country_modifier = {
			nobles_estate_target_satisfaction = medium_permanent_target_satisfaction_penalty
			global_nobles_estate_power = -0.15
			global_crown_estate_power = 0.1
			monthly_towards_centralization = societal_value_minor_monthly_move
		}
		estate_preferences = {
			crown_estate
		}
		years = 2
	}

	# Crown tenants buy their farms as freehold, from the letter of 1701 (age 5). The peasants'
	# mirror of swe_donationer, as teu_amber_monopoly_law's peasant policy mirrors its burgher one.
	swe_skattekop = {
		potential = {
			current_age_or_later = {
				age = age_5_absolutism
			}
		}
		country_modifier = {
			peasants_estate_target_satisfaction = small_permanent_target_satisfaction
			global_peasants_estate_power = 0.1
			global_crown_estate_power = -0.05
			monthly_towards_free_subjects = societal_value_minor_monthly_move
		}
		estate_preferences = {
			peasants_estate
		}
		years = 2
	}
}

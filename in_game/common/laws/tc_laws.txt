# Adds Swedish policies to vanilla laws without replacing them (same pattern as D009's INJECT:legal_code_law)

INJECT:mining_law = {

	# Bergskollegium (swe_bergskollegium_decision): the Board of Mines ran every mine and ironworks
	# of the realm for the Crown, so the Crown gains power in the mines rather than output, which
	# falun_base, mulltimmer_masugn and polhem_machina_nova already cover.
	# Balanced against burghers_mining_law (same mine expansion cost) and nobles_mining_law
	# (-0.1 rgo build time everywhere): no estate satisfaction or societal drift, and the rest
	# only in mining locations.
	swe_bergskollegium = {
		unique = yes
		potential = {
			has_or_had_tag = SWE
			has_unlocked_policy_trigger = { type = swe_bergskollegium }
		}
		country_modifier = {
			expand_rgo_mining_cost_modifier = -0.1
		}
		# Every good vanilla mines (method = mining in 00_raw_materials.txt)
		location_modifier = {
			potential_trigger = {
				custom_tooltip = {
					text = swe_bergskollegium_mines_tt
					OR = {
						raw_material = goods:iron
						raw_material = goods:copper
						raw_material = goods:silver
						raw_material = goods:goods_gold
						raw_material = goods:lead
						raw_material = goods:tin
						raw_material = goods:coal
						raw_material = goods:alum
						raw_material = goods:saltpeter
						raw_material = goods:mercury
						raw_material = goods:gems
						raw_material = goods:stone
						raw_material = goods:marble
					}
				}
			}
			local_crown_estate_power = 0.1
			local_rgo_build_time = -0.1
		}
		estate_preferences = {
			crown_estate
		}
		years = 2
	}
}

INJECT:legal_code_law = {

	# Landslagen (swe_landslag_decision): Magnus Eriksson's national law code of about 1350 replaced
	# the provincial laws. Its King's Chapter is the swe_konungabalken privilege Sweden holds from 1337.
	# Balanced against dan_regional_law (three minor value moves, small satisfaction) and vanilla
	# flavor_swe.53, which only moves centralization once and needs Magnus Eriksson himself.
	swe_landslag = {
		unique = yes
		potential = {
			has_or_had_tag = SWE
			has_unlocked_policy_trigger = { type = swe_landslag }
		}
		country_modifier = {
			monthly_towards_centralization = societal_value_minor_monthly_move
			monthly_towards_free_subjects = societal_value_minor_monthly_move
			global_estate_target_satisfaction = small_permanent_target_satisfaction
		}
		# Sveriges rikes lag (swe_rikes_lag_decision): the code of 1734 for town and countryside alike.
		# Kept inside this policy rather than as a policy of its own. civil_law_policy's legislative
		# efficiency on top of the base. No proximity: swedish_royal_post_office already gives +0.1.
		country_modifier = {
			potential_trigger = {
				custom_tooltip = {
					text = swe_rikes_lag_country_bonus
					has_variable = swe_rikes_lag
				}
			}
			legislative_efficiency = 0.1
		}
		# Wismar Tribunal (swe_wismar_tribunal_decision): from 1653 the Crown's own high court at Wismar
		# judged its German lands under their own law, so no appeal went to the imperial courts.
		# Kept inside this policy rather than as a privilege of its own. Integration and the slower
		# assimilation are those of lat_compromise_with_greek_populace (German law and language kept);
		# the crown power is that of Bottniska handelstvånget, the reverse of justicia_de_aragon.
		location_modifier = {
			potential_trigger = {
				custom_tooltip = {
					text = swe_wismar_tribunal_location_bonus
					owner = { has_variable = swe_wismar_tribunal }
					swe_german_provinces_trigger = yes
				}
			}
			local_integration_speed = 0.10
			local_crown_estate_power = 0.1
			local_pop_assimilation_speed_modifier = -0.25
		}
		estate_preferences = {
			crown_estate
			peasants_estate
		}
		years = 2
	}
}

INJECT:distribution_of_power_law = {

	# Riksdagsordningen: the Riksdag ordinance of 1617 fixed the four estates of the realm, with the
	# peasants as the fourth. No decision: gated by vanilla's kung_och_riksdag and the peasants'
	# seat (peasants_represented_in_parliament), the way dop_law_estates_general is gated by its advance.
	# Balanced against dop_law_english_parliament (-0.33 crown power, +0.05 legitimacy, +0.1 estate
	# max tax, +0.1 nobles and burghers agenda impact, -0.15 political influence): a smaller crown
	# penalty and no legitimacy, with the agenda weight on the peasants.
	swe_riksdagsordningen = {
		unique = yes
		potential = {
			has_or_had_tag = SWE
			has_advance = kung_och_riksdag
		}
		allow = {
			current_age_or_later = {
				age = age_4_reformation
			}
			has_estate_privilege = estate_privilege:peasants_represented_in_parliament
		}
		country_modifier = {
			global_crown_estate_power = -0.2
			global_estate_max_tax = 0.1
			peasants_estate_agenda_impact = 0.2
			burghers_estate_agenda_impact = 0.1
			monthly_political_influence_gain_modifier = -0.1
		}
		# The Age of Liberty (vanilla swe_age_of_freedom): the Secret Committee of the nobles, clergy
		# and burghers ran foreign affairs, finance and defence, and kept the peasants out
		country_modifier = {
			potential_trigger = {
				custom_tooltip = {
					text = swe_riksdagsordningen_liberty_bonus
					has_reform = government_reform:swe_age_of_freedom
				}
			}
			nobles_estate_agenda_impact = 0.25
			peasants_estate_agenda_impact = -0.2
		}
		estate_preferences = {
			peasants_estate
			burghers_estate
		}
		years = 2
	}
}

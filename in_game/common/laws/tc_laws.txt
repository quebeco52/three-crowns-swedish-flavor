# Adds Swedish policies to vanilla laws without replacing them.

INJECT:mining_law = {

	# Bergskollegium (swe_bergskollegium_decision): the Board of Mines ran the realm's mines and ironworks
	# for the Crown, so it gives crown power in mining locations rather than output. Mine expansion cost
	# from vanilla's zimbabwe_mining_law.
	swe_bergskollegium = {
		unique = yes
		potential = {
			has_or_had_tag = SWE
			has_unlocked_policy_trigger = { type = swe_bergskollegium }
		}
		country_modifier = {
			expand_rgo_mining_cost_modifier = -0.2
		}
		# Every mined good (method = mining in vanilla's 00_raw_materials.txt)
		location_modifier = {
			potential_trigger = {
				custom_tooltip = {
					text = swe_bergskollegium_mines_tt
					OR = {
						raw_material = goods:iron
						raw_material = goods:copper
						raw_material = goods:silver
						raw_material = goods:goods_gold
						raw_material = goods:lead
						raw_material = goods:tin
						raw_material = goods:coal
						raw_material = goods:alum
						raw_material = goods:saltpeter
						raw_material = goods:mercury
						raw_material = goods:gems
						raw_material = goods:stone
						raw_material = goods:marble
					}
				}
			}
			local_crown_estate_power = 0.1
			local_rgo_build_time = -0.1
		}
		estate_preferences = {
			crown_estate
		}
		years = 2
	}
}

INJECT:legal_code_law = {

	# Landslagen (swe_landslag_decision): Magnus Eriksson's national law code of about 1350, which replaced
	# the provincial laws. Its King's Chapter is the swe_konungabalken privilege, held from 1337. Balanced
	# against vanilla's dan_regional_law.
	swe_landslag = {
		unique = yes
		potential = {
			has_or_had_tag = SWE
			has_unlocked_policy_trigger = { type = swe_landslag }
		}
		country_modifier = {
			monthly_towards_centralization = societal_value_minor_monthly_move
			monthly_towards_free_subjects = societal_value_minor_monthly_move
			global_estate_target_satisfaction = small_permanent_target_satisfaction
		}
		# Sveriges rikes lag (swe_rikes_lag_decision): the code of 1734 for town and countryside alike.
		# Legislative efficiency from vanilla's civil_law_policy.
		country_modifier = {
			potential_trigger = {
				custom_tooltip = {
					text = swe_rikes_lag_country_bonus
					has_variable = swe_rikes_lag
				}
			}
			legislative_efficiency = 0.1
		}
		estate_preferences = {
			crown_estate
			peasants_estate
		}
		years = 2
	}
}

INJECT:nobles_rights_laws = {

	# Riddarhusordningen (swe_riddarhusordningen_decision): the House of Nobility ordinance of 1626 gave the
	# nobles their own house at the Riksdag and tied their rank to the service of the Crown. Balanced against
	# vanilla's rural_nobles_authority_policy. Excludes swe_sekreterare (see there).
	swe_riddarhuset = {
		unique = yes
		potential = {
			has_or_had_tag = SWE
			has_unlocked_policy_trigger = { type = swe_riddarhuset }
		}
		allow = {
			NOT = { has_policy = swe_sekreterare } # Bare key, as in vanilla laws
		}
		country_modifier = {
			monthly_towards_aristocracy = societal_value_minor_monthly_move
			nobles_estate_target_satisfaction = small_permanent_target_satisfaction
			global_nobles_estate_power = 0.1
			country_cabinet_efficiency = 0.05
		}
		estate_preferences = {
			nobles_estate
		}
		years = 2
	}
}

INJECT:distribution_of_power_law = {

	# Riksdagsordningen: the Riksdag ordinance of 1617 fixed the four estates of the realm, with the peasants
	# as the fourth. Gated by vanilla's kung_och_riksdag and the peasants' seat in parliament
	# (peasants_represented_in_parliament), not by a decision. Balanced against vanilla's
	# dop_law_english_parliament.
	swe_riksdagsordningen = {
		unique = yes
		potential = {
			has_or_had_tag = SWE
			has_advance = kung_och_riksdag
		}
		allow = {
			current_age_or_later = {
				age = age_4_reformation
			}
			has_estate_privilege = estate_privilege:peasants_represented_in_parliament
		}
		country_modifier = {
			global_crown_estate_power = -0.2
			global_estate_max_tax = 0.1
			peasants_estate_agenda_impact = 0.2
			burghers_estate_agenda_impact = 0.1
			monthly_political_influence_gain_modifier = -0.1
		}
		# The Age of Liberty (vanilla swe_age_of_freedom): the Secret Committee of the nobles, clergy
		# and burghers ran foreign affairs, finance and defence, and kept the peasants out.
		country_modifier = {
			potential_trigger = {
				custom_tooltip = {
					text = swe_riksdagsordningen_liberty_bonus
					has_reform = government_reform:swe_age_of_freedom
				}
			}
			nobles_estate_agenda_impact = 0.25
			peasants_estate_agenda_impact = -0.2
		}
		estate_preferences = {
			peasants_estate
			burghers_estate
		}
		years = 2
	}
}

INJECT:cabinet_laws = {

	# Sekreterarregementet: Gustav I ran his chancery with Swedish and German secretaries, and from the
	# regiment of 1539-40 his sons kept high-noble chancellors in name but took advice from secretaries of
	# non-noble birth, until Gustav II Adolf's assurance of 1611 (SBL Conrad von Pyhy; Asker, Hur riket
	# styrdes; Nilsson, Scandia 1937). The secular twin of vanilla's administrative_clergy_policy, with the
	# burghers in the clergy's place. Excludes swe_riddarhuset both ways, as the two would stack cabinet
	# efficiency; swe_riddarhusordningen_decision removes it.
	swe_sekreterare = {
		unique = yes
		potential = {
			has_or_had_tag = SWE
			government_type = government_type:monarchy
			current_age_or_later = {
				age = age_4_reformation
			}
		}
		allow = {
			NOT = { has_policy = swe_riddarhuset } # Bare key, as in vanilla laws
		}
		country_modifier = {
			country_cabinet_efficiency = 0.1
			monthly_political_influence_gain_modifier = 0.05
			burghers_estate_target_satisfaction = small_permanent_target_satisfaction
			nobles_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
			monthly_towards_centralization = societal_value_minor_monthly_move
		}
		estate_preferences = {
			crown_estate
			burghers_estate
		}
		years = 2
	}
}

	}
}

# The other colleges of the Instrument of Government are bureaucracies, in tc_bureaucracies.txt.

# Kronans jord: who holds the land of the realm. Every Swedish farm was crown, freehold or noble land (krono,
# skatte, frälse). The Crown granted land and rents away in donations to pay for its wars (1611-54), took them
# back in the Quarter Reduction of 1655 and the Great Reduction from 1680, and from 1701 sold crown farms to
# their tenants as freehold (Högman, "The Reduction"; prop. 1930:246; Carlsson in Scandia). A law of its own,
# as swe_riddarhuset holds nobles_rights_laws. Vanilla's reduction events (flavor_swe.17, .18, .20) still fire
# beside it. It starts with no policy; vanilla gives its country laws one in setup or by event.
swe_kronans_jord_law = {
	unique = yes
	law_category = estates

	potential = {
		has_or_had_tag = SWE
		current_age_or_later = {
			age = age_4_reformation
		}
	}

	# Crown farms and rents granted to officers, generals and creditors. Pitched between vanilla's
	# rural_nobles_authority_policy and strengthen_nobles_rights_policy.
	swe_donationer = {
		country_modifier = {
			nobles_estate_target_satisfaction = small_permanent_target_satisfaction
			global_nobles_estate_power = 0.1
			global_crown_estate_power = -0.05
			nobles_estate_levy_size = 0.1
			monthly_towards_aristocracy = societal_value_minor_monthly_move
		}
		estate_preferences = {
			nobles_estate
		}
		years = 2
	}

	# The Crown takes the land back. Crown power is half of vanilla's absolute_rule_policy.
	swe_reduktion = {
		country_modifier = {
			nobles_estate_target_satisfaction = medium_permanent_target_satisfaction_penalty
			global_nobles_estate_power = -0.15
			global_crown_estate_power = 0.1
			monthly_towards_centralization = societal_value_minor_monthly_move
		}
		estate_preferences = {
			crown_estate
		}
		years = 2
	}

	# Crown tenants buy their farms as freehold, from the letter of 1701 (age 5). The peasants' mirror of
	# swe_donationer.
	swe_skattekop = {
		potential = {
			current_age_or_later = {
				age = age_5_absolutism
			}
		}
		country_modifier = {
			peasants_estate_target_satisfaction = small_permanent_target_satisfaction
			global_peasants_estate_power = 0.1
			global_crown_estate_power = -0.05
			monthly_towards_free_subjects = societal_value_minor_monthly_move
		}
		estate_preferences = {
			peasants_estate
		}
		years = 2
	}
}

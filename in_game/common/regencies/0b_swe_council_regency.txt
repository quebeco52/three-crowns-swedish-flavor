# Förmyndarregering: when the Swedish crown passes to a child, the council of the realm governs through one of its
# own, as it did for Magnus Eriksson (1319-32), Kristina (1632-44) and Karl XI (1660-72). Named 0b to sort before
# 1_nobles_regency.txt and 10_consort_regency.txt, since regencies have no priority key.
# The regent is the head of the cabinet if one has been promoted, otherwise any councillor, and must pass
# swe_council_regent_trigger, which bars foreigners. With no eligible councillor a vanilla type takes over.
# Cabinet efficiency is twice vanilla's cabinet_head_regency (0.05); crown power as iro_assertive_council_of_fifty (-0.1).
# Not a consort's regency, so vanilla's flavor_swe.16 and .19 can still fire.
swe_council_regency = {
	allow = {
		has_or_had_tag = SWE
		government_type = government_type:monarchy
		any_cabinet_character = {
			swe_council_regent_trigger = yes
		}
	}

	modifier = {
		country_cabinet_efficiency = 0.1
		global_crown_estate_power = -0.1
	}

	start_effect = {
		if = {
			limit = {
				any_cabinet_character = {
					has_character_modifier = head_of_the_cabinet_modifier
					swe_council_regent_trigger = yes
				}
			}
			random_cabinet_character = {
				limit = {
					has_character_modifier = head_of_the_cabinet_modifier
					swe_council_regent_trigger = yes
				}
				save_scope_as = swe_council_regent
			}
		}
		else = {
			random_cabinet_character = {
				limit = { swe_council_regent_trigger = yes }
				save_scope_as = swe_council_regent
			}
		}
		set_regent = scope:swe_council_regent
	}
}

# Uppsala mötes beslut: the Uppsala Synod of 1593 bound the Swedish church to the Augsburg Confession and every
# future king to uphold it. Granted through swe_church.1. Tiers below: Kyrkolagen (swe_kyrkolagen_decision) and
# Uppsala's charter (swe_svealand.5). Based on vanilla's clergy_enforced_unity, with less power and conversion
# of heretics only.
swe_uppsala_mote = {
	estate = clergy_estate

	potential = {
		has_or_had_tag = SWE
		religion = religion:lutheran
		has_unlocked_estate_privilege_trigger = { type = swe_uppsala_mote }
	}

	can_revoke = {
		"estate_power(estate_type:crown_estate)" >= 0.5
	}

	country_modifier = {
		clergy_estate_target_satisfaction = small_privilege_target_satisfaction
		global_clergy_estate_power = 0.2
		global_heretic_pop_conversion_speed_modifier = 0.1
	}

	# Kyrkolagen (swe_kyrkolagen_decision): the Church Law of 1686 made every Swede learn to read. Its household
	# examinations reached every home, so it raises commoners' maximum literacy rather than its growth. Below
	# vanilla's basic_religious_education, as it takes no policy slot and stacks with any education policy. The
	# drift to spiritualist is the law binding every subject to the Augsburg Confession (sv.wiki 1686 års
	# kyrkolag). Not global_monthly_literacy, which shows two decimals, so small values read +0.00%.
	country_modifier = {
		potential_trigger = {
			custom_tooltip = {
				text = swe_kyrkolagen_country_bonus
				has_variable = swe_kyrkolagen
			}
		}
		global_peasants_max_literacy = 3
		global_soldiers_max_literacy = 3
		global_laborers_max_literacy = 3
		monthly_towards_spiritualist = societal_value_tiny_monthly_move
	}

	# Uppsala's charter (swe_svealand.5): the privileges of 15 March 1595 had the university examine schoolmasters
	# and keep every school to the same plan, and Sigismund's confirmation of the clergy's privileges in 1594 bound
	# the Crown to keep the university (Annerstedt 1877). Half of vanilla's clerical_institutions, as it takes no
	# policy slot and stacks. Clergy only; the burghers' schools are swe_church.2's.
	country_modifier = {
		potential_trigger = {
			custom_tooltip = {
				text = swe_uppsala_charter_country_bonus
				has_variable = swe_uppsala_charter
			}
		}
		global_clergy_max_literacy = 5
	}
}

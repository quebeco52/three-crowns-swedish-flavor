# Uppsala mötes beslut: the Uppsala Synod of 1593 bound the Swedish church to the Augsburg
# Confession and every future king to uphold it. Granted through swe_church.1.
# Balanced against clergy_enforced_unity (+0.25 clergy power, +10% conversion of all faiths):
# less power, and the conversion bonus only works on heretics (same as Castile's advance).
swe_uppsala_mote = {
	estate = clergy_estate

	potential = {
		has_or_had_tag = SWE
		religion = religion:lutheran
	}

	allow = {
		has_unlocked_estate_privilege_trigger = { type = swe_uppsala_mote }
	}

	can_revoke = {
		"estate_power(estate_type:crown_estate)" >= 0.5
	}

	country_modifier = {
		clergy_estate_target_satisfaction = small_privilege_target_satisfaction
		global_clergy_estate_power = 0.2
		global_heretic_pop_conversion_speed_modifier = 0.1
	}

	# Kyrkolagen (swe_kyrkolagen_decision): the Church Law of 1686 made every Swede learn to read.
	# A quarter of the smallest vanilla monthly literacy bonus (0.01), since it adds up over centuries.
	country_modifier = {
		potential_trigger = {
			custom_tooltip = {
				text = swe_kyrkolagen_country_bonus
				has_variable = swe_kyrkolagen
			}
		}
		global_monthly_literacy = 0.0025
	}
}

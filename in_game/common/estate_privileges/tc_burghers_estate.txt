# Bottniska handelstvånget: from the trade ordinances of 1614 and 1617 the East and West
# Bothnian towns could trade abroad only through Stockholm. Abolished in 1765.
# Replaces swe_birkarl_rights through swe_bottniska_handelstvanget_decision: the Sámi trade goes on at half strength,
# but the crown takes the tax back from the Birkarlar.
# Balanced against swe_birkarl_rights: its -0.1 crown power goes away in the lappmarks and becomes
# +0.1 along the Bothnian coast (Västerbotten is in both).
swe_bottniska_handelstvanget = {
	estate = burghers_estate

	potential = {
		has_or_had_tag = SWE
	}

	allow = {
		has_unlocked_estate_privilege_trigger = { type = swe_bottniska_handelstvanget }
		NOT = { has_estate_privilege = estate_privilege:swe_birkarl_rights }
	}

	# Lower than the other Swedish privileges (0.5): the Riksdag abolished the staple in 1765, when the
	# crown was weak, and revoking a privilege already costs stability
	can_revoke = {
		"estate_power(estate_type:crown_estate)" >= 0.3
	}

	# The old birkarl lands (same provinces as swe_birkarl_rights)
	location_modifier = {
		potential_trigger = {
			swe_birkarl_lands_trigger = yes
		}
		local_fur_output_modifier = 0.1
		local_wild_game_output_modifier = 0.1
		local_market_access = 0.1
	}

	# The East and West Bothnian coast cannot trade abroad, and the crown taxes what it sells
	location_modifier = {
		potential_trigger = {
			swe_bothnian_coast_trigger = yes
		}
		local_crown_estate_power = 0.1
		local_trades_per_burgher = -0.25
	}

	# Tjärkompaniet (swe_tar_company_decision): from 1648 the Tar Company bought all the tar of Österbotten
	# for export through Stockholm. Kept inside this privilege rather than as a privilege of its own.
	# Above the single-town scandinavian_tar_privileges (+0.2 tar, with an efficiency penalty)
	# because Österbotten is thinly settled subpolar forest, and it only pays where tar kilns are built.
	location_modifier = {
		potential_trigger = {
			custom_tooltip = {
				text = swe_tar_company_location_bonus
				owner = { has_variable = swe_tar_company }
				swe_osterbotten_trigger = yes
			}
		}
		local_tar_output_modifier = 0.25
		local_tar_establishment_speed = 0.25
		local_lumber_output_modifier = 0.1
	}

	# The staple. Kept small: Stockholm is already the main market of the north
	location_modifier = {
		potential_trigger = {
			this = location:stockholm
		}
		local_merchant_capacity_modifier = 0.1
		local_trades_per_burgher = 0.25
	}

	country_modifier = {
		burghers_estate_target_satisfaction = small_privilege_target_satisfaction
		global_burghers_estate_power = 0.1
	}
}
